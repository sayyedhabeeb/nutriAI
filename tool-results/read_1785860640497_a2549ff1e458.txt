     1→'use client';
     2→
     3→import React, { useState, useEffect, useCallback } from 'react';
     4→import { format, parseISO } from 'date-fns';
     5→import { motion, AnimatePresence } from 'framer-motion';
     6→import { toast } from 'sonner';
     7→import {
     8→  BarChart, Bar, AreaChart, Area, PieChart, Pie, Cell,
     9→  XAxis, YAxis, Tooltip, ResponsiveContainer, ReferenceLine,
    10→} from 'recharts';
    11→import { Button } from '@/components/ui/button';
    12→import { Card } from '@/components/ui/card';
    13→import { Input } from '@/components/ui/input';
    14→import { Skeleton } from '@/components/ui/skeleton';
    15→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
    16→import {
    17→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogClose,
    18→} from '@/components/ui/dialog';
    19→import { Label } from '@/components/ui/label';
    20→import {
    21→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    22→} from '@/components/ui/select';
    23→import {
    24→  TrendingUp, Flame, Dumbbell, Scale, Droplets, Plus, Minus,
    25→  BarChart3, Download, Lightbulb, Trophy, Lock,
    26→} from 'lucide-react';
    27→import { apiFetch } from './api';
    28→import { PIE_COLORS, fadeIn } from './constants';
    29→
    30→export function ProgressView() {
    31→  const [tab, setTab] = useState('weekly');
    32→  const [weeklyData, setWeeklyData] = useState<Record<string, unknown>[]>([]);
    33→  const [summary, setSummary] = useState<Record<string, unknown> | null>(null);
    34→  const [weightLogs, setWeightLogs] = useState<Record<string, unknown>[]>([]);
    35→  const [waterGlasses, setWaterGlasses] = useState(0);
    36→  const [loading, setLoading] = useState(true);
    37→  const [weightInput, setWeightInput] = useState('');
    38→  const [weightNotes, setWeightNotes] = useState('');
    39→  const [justFilledIndex, setJustFilledIndex] = useState<number | null>(null);
    40→
    41→  // Achievements state
    42→  const [achievements, setAchievements] = useState<{ id: string; name: string; description: string; icon: string; earned: boolean; earnedDate?: string }[]>([]);
    43→
    44→  // Export dialog state
    45→  const [exportOpen, setExportOpen] = useState(false);
    46→  const [exportFormat, setExportFormat] = useState('csv');
    47→  const [exportDays, setExportDays] = useState('7');
    48→  const [exporting, setExporting] = useState(false);
    49→
    50→  const fetchProgress = useCallback(async () => {
    51→    setLoading(true);
    52→    try {
    53→      const [weekData, sumData, wLogs, waterData] = await Promise.all([
    54→        apiFetch('/api/progress/weekly'),
    55→        apiFetch(`/api/progress/summary?period=${tab === 'monthly' ? 'month' : 'week'}`),
    56→        apiFetch('/api/weight-log?limit=30'),
    57→        apiFetch('/api/water-log'),
    58→      ]);
    59→      setWeeklyData(weekData || []);
    60→      setSummary(sumData?.summary || null);
    61→      setWeightLogs(wLogs || []);
    62→      setWaterGlasses(waterData?.glassesConsumed || 0);
    63→
    64→      // Fetch achievements
    65→      try {
    66→        const achData = await apiFetch('/api/achievements');
    67→        setAchievements(achData.achievements || []);
    68→      } catch {
    69→        // Silent fail
    70→      }
    71→    } catch { toast.error('Failed to load progress'); }
    72→    finally { setLoading(false); }
    73→  }, [tab]);
    74→
    75→  useEffect(() => { fetchProgress(); }, [fetchProgress]);
    76→
    77→  const handleAddWater = async (delta: number) => {
    78→    const newCount = waterGlasses + delta;
    79→    if (newCount < 0) return;
    80→    if (delta > 0) {
    81→      setJustFilledIndex(waterGlasses);
    82→      setTimeout(() => setJustFilledIndex(null), 600);
    83→      try {
    84→        await apiFetch('/api/water-log', { method: 'POST', body: JSON.stringify({ glasses: delta }) });
    85→        setWaterGlasses(newCount);
    86→        toast.success(`\uD83D\uDCA7 +${delta} glass${delta > 1 ? 'es' : ''} of water`);
    87→      } catch { toast.error('Failed to log water'); }
    88→    } else {
    89→      setWaterGlasses(Math.max(0, newCount));
    90→    }
    91→  };
    92→
    93→  const handleLogWeight = async () => {
    94→    if (!weightInput || Number(weightInput) <= 0) return;
    95→    try {
    96→      await apiFetch('/api/weight-log', { method: 'POST', body: JSON.stringify({ weightKg: Number(weightInput), notes: weightNotes || null }) });
    97→      toast.success('Weight logged!');
    98→      setWeightInput('');
    99→      setWeightNotes('');
   100→      fetchProgress();
   101→    } catch (err) { toast.error((err as Error).message); }
   102→  };
   103→
   104→  const handleExport = async () => {
   105→    setExporting(true);
   106→    try {
   107→      const token = localStorage.getItem('nutriai_session');
   108→      const url = `/api/export?format=${exportFormat}&days=${exportDays}`;
   109→      const res = await fetch(url, {
   110→        headers: { Authorization: `Bearer ${token}` },
   111→      });
   112→      if (!res.ok) throw new Error('Export failed');
   113→      const blob = await res.blob();
   114→      const ext = exportFormat === 'csv' ? 'csv' : 'json';
   115→      const contentType = exportFormat === 'csv' ? 'text/csv' : 'application/json';
   116→      const blobUrl = URL.createObjectURL(new Blob([blob], { type: contentType }));
   117→      const a = document.createElement('a');
   118→      a.href = blobUrl;
   119→      a.download = `nutriai-export-${exportDays}days.${ext}`;
   120→      document.body.appendChild(a);
   121→      a.click();
   122→      document.body.removeChild(a);
   123→      URL.revokeObjectURL(blobUrl);
   124→      toast.success('Data exported successfully!');
   125→      setExportOpen(false);
   126→    } catch {
   127→      toast.error('Failed to export data');
   128→    } finally {
   129→      setExporting(false);
   130→    }
   131→  };
   132→
   133→  const chartData = weeklyData.map((d) => {
   134→    const consumed = d.consumed as Record<string, number>;
   135→    const targets = d.targets as Record<string, number> | null;
   136→    return {
   137→      date: format(parseISO(d.date as string), 'EEE'),
   138→      calories: consumed?.calories || 0,
   139→      target: targets?.calories || 0,
   140→    };
   141→  });
   142→
   143→  const allCaloriesZero = chartData.every((d) => d.calories === 0 && d.target === 0);
   144→
   145→  const macroData = [
   146→    { name: 'Protein', value: summary?.avgProtein || 0, color: PIE_COLORS[0] },
   147→    { name: 'Carbs', value: summary?.avgCarbs || 0, color: PIE_COLORS[1] },
   148→    { name: 'Fat', value: summary?.avgFat || 0, color: PIE_COLORS[2] },
   149→  ];
   150→
   151→  const weightChartData = weightLogs.slice().reverse().slice(-7).map((w) => ({
   152→    date: format(parseISO(w.logDate as string), 'MMM d'),
   153→    weight: w.weightKg as number,
   154→  }));
   155→
   156→  const noWeightData = weightChartData.length < 2;
   157→
   158→  // Calorie breakdown table data - filter out zero-eaten days
   159→  const breakdownData = weeklyData
   160→    .map((d) => {
   161→      const consumed = d.consumed as Record<string, number>;
   162→      const targets = d.targets as Record<string, number> | null;
   163→      const consumedCal = consumed?.calories || 0;
   164→      const targetCal = targets?.calories || 0;
   165→      const diff = consumedCal - targetCal;
   166→      const pct = targetCal > 0 ? Math.round((consumedCal / targetCal) * 100) : 0;
   167→      return {
   168→        date: d.date as string,
   169→        consumedCal,
   170→        targetCal,
   171→        diff,
   172→        pct: Math.min(pct, 150),
   173→      };
   174→    })
   175→    .filter((d) => d.consumedCal > 0);
   176→
   177→  const weeklyTotalConsumed = breakdownData.reduce((sum, d) => sum + d.consumedCal, 0);
   178→  const weeklyTotalTarget = breakdownData.reduce((sum, d) => sum + d.targetCal, 0);
   179→  const weeklyTotalDiff = weeklyTotalConsumed - weeklyTotalTarget;
   180→
   181→  const statCards = [
   182→    { label: 'Avg Daily Calories', value: `${Math.round(summary?.avgCalories || 0)}`, icon: Flame, iconBg: 'bg-orange-100', iconColor: 'text-orange-600' },
   183→    { label: 'Avg Protein', value: `${Math.round(summary?.avgProtein || 0)}g`, icon: Dumbbell, iconBg: 'bg-blue-100', iconColor: 'text-blue-600' },
   184→    { label: 'Total Days', value: `${summary?.totalDays || 0}`, icon: TrendingUp, iconBg: 'bg-emerald-100', iconColor: 'text-emerald-600' },
   185→    { label: 'Current Weight', value: summary?.currentWeight ? `${summary.currentWeight}kg` : 'N/A', icon: Scale, iconBg: 'bg-purple-100', iconColor: 'text-purple-600' },
   186→  ];
   187→
   188→  if (loading) {
   189→    return (
   190→      <div className="p-4 max-w-lg mx-auto space-y-4">
   191→        <Skeleton className="h-8 w-32 rounded-xl" />
   192→        <Skeleton className="h-64 w-full rounded-2xl" />
   193→        <div className="grid grid-cols-2 gap-4"><Skeleton className="h-28 w-full rounded-2xl" /><Skeleton className="h-28 w-full rounded-2xl" /></div>
   194→      </div>
   195→    );
   196→  }
   197→
   198→  return (
   199→    <motion.div {...fadeIn} className="p-4 max-w-lg mx-auto space-y-5">
   200→      {/* Header */}
   201→      <div className="flex items-center justify-between">
   202→        <div className="flex items-center gap-2">
   203→          <div className="w-9 h-9 rounded-xl bg-emerald-100 flex items-center justify-center">
   204→            <TrendingUp className="h-5 w-5 text-emerald-600" />
   205→          </div>
   206→          <h1 className="text-xl font-bold text-gray-900">Progress</h1>
   207→        </div>
   208→        <button
   209→          onClick={() => setExportOpen(true)}
   210→          className="border border-gray-200 dark:border-gray-700 bg-white dark:bg-gray-800 shadow-sm rounded-lg px-3 py-1.5 text-xs font-medium text-gray-600 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-700 transition-colors flex items-center gap-1.5"
   211→        >
   212→          <Download className="h-3.5 w-3.5" />
   213→          <span className="hidden sm:inline">Export</span>
   214→        </button>
   215→      </div>
   216→
   217→      {/* Export Dialog */}
   218→      <Dialog open={exportOpen} onOpenChange={setExportOpen}>
   219→        <DialogContent className="sm:max-w-sm rounded-2xl">
   220→          <DialogHeader>
   221→            <DialogTitle>Export Data</DialogTitle>
   222→          </DialogHeader>
   223→          <div className="space-y-4 py-2">
   224→            <div className="space-y-1.5">
   225→              <Label className="text-sm font-medium">Format</Label>
   226→              <Select value={exportFormat} onValueChange={setExportFormat}>
   227→                <SelectTrigger className="h-11 rounded-xl"><SelectValue /></SelectTrigger>
   228→                <SelectContent>
   229→                  <SelectItem value="csv">CSV Spreadsheet</SelectItem>
   230→                  <SelectItem value="json">JSON</SelectItem>
   231→                </SelectContent>
   232→              </Select>
   233→            </div>
   234→            <div className="space-y-1.5">
   235→              <Label className="text-sm font-medium">Period</Label>
   236→              <Select value={exportDays} onValueChange={setExportDays}>
   237→                <SelectTrigger className="h-11 rounded-xl"><SelectValue /></SelectTrigger>
   238→                <SelectContent>
   239→                  <SelectItem value="7">Last 7 days</SelectItem>
   240→                  <SelectItem value="14">Last 14 days</SelectItem>
   241→                  <SelectItem value="30">Last 30 days</SelectItem>
   242→                </SelectContent>
   243→              </Select>
   244→            </div>
   245→          </div>
   246→          <DialogFooter className="gap-2 sm:gap-0">
   247→            <DialogClose asChild>
   248→              <Button variant="outline" className="rounded-xl">Cancel</Button>
   249→            </DialogClose>
   250→            <Button className="bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl" onClick={handleExport} disabled={exporting}>
   251→              {exporting ? 'Exporting...' : 'Download'}
   252→            </Button>
   253→          </DialogFooter>
   254→        </DialogContent>
   255→      </Dialog>
   256→
   257→      <Tabs value={tab} onValueChange={setTab}>
   258→        <TabsList className="w-full rounded-xl">
   259→          <TabsTrigger value="weekly" className="flex-1 rounded-lg data-[state=active]:shadow-sm">Weekly</TabsTrigger>
   260→          <TabsTrigger value="monthly" className="flex-1 rounded-lg text-gray-500 dark:text-gray-400 data-[state=active]:text-foreground data-[state=active]:shadow-sm">Monthly</TabsTrigger>
   261→        </TabsList>
   262→      </Tabs>
   263→
   264→      {/* Calorie Chart */}
   265→      <Card className="p-5 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   266→        <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100 mb-3">Calorie Intake</h3>
   267→        <div className="h-48">
   268→          {allCaloriesZero ? (
   269→            <div className="h-full flex flex-col items-center justify-center py-8 text-gray-400">
   270→              <BarChart3 className="h-12 w-12 mb-3 text-gray-200" />
   271→              <p className="text-sm font-medium">Start logging meals to see your calorie trends!</p>
   272→            </div>
   273→          ) : (
   274→            <ResponsiveContainer width="100%" height="100%">
   275→              <BarChart data={chartData} barCategoryGap="20%">
   276→                <XAxis dataKey="date" tick={{ fontSize: 11, fill: '#6b7280' }} axisLine={false} tickLine={false} />
   277→                <YAxis tick={{ fontSize: 11, fill: '#6b7280' }} axisLine={false} tickLine={false} />
   278→                <Tooltip
   279→                  contentStyle={{ borderRadius: '12px', border: '1px solid #e5e7eb', boxShadow: '0 4px 6px -1px rgba(0,0,0,0.1)' }}
   280→                />
   281→                <defs>
   282→                  <linearGradient id="barGrad" x1="0" y1="0" x2="0" y2="1">
   283→                    <stop offset="0%" stopColor="#10b981" stopOpacity={1} />
   284→                    <stop offset="100%" stopColor="#34d399" stopOpacity={0.8} />
   285→                  </linearGradient>
   286→                </defs>
   287→                <Bar dataKey="calories" fill="url(#barGrad)" radius={[4, 4, 0, 0]} name="Consumed" />
   288→                <Bar dataKey="target" fill="#e5e7eb" radius={[4, 4, 0, 0]} name="Target" />
   289→                {!allCaloriesZero && chartData[0]?.target > 0 && (
   290→                  <ReferenceLine y={chartData[0].target} stroke="#10b981" strokeDasharray="6 3" strokeOpacity={0.6} />
   291→                )}
   292→              </BarChart>
   293→            </ResponsiveContainer>
   294→          )}
   295→        </div>
   296→      </Card>
   297→
   298→      {/* Macro Pie + Weight Trend */}
   299→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   300→        <Card className="p-5 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   301→          <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100 mb-2">Macro Breakdown</h3>
   302→          <div className="h-44 relative">
   303→            <ResponsiveContainer width="100%" height="100%">
   304→              <PieChart>
   305→                <Pie data={macroData} cx="50%" cy="50%" innerRadius={38} outerRadius={60} dataKey="value" strokeWidth={2}>
   306→                  {macroData.map((entry, i) => <Cell key={i} fill={entry.color} />)}
   307→                </Pie>
   308→                <Tooltip contentStyle={{ borderRadius: '12px', border: '1px solid #e5e7eb' }} />
   309→              </PieChart>
   310→            </ResponsiveContainer>
   311→            <div className="absolute inset-0 flex flex-col items-center justify-center pointer-events-none">
   312→              <span className="text-lg font-bold text-gray-800 dark:text-gray-100 tabular-nums">{Math.round(summary?.avgCalories || 0)}</span>
   313→              <span className="text-[10px] text-gray-400 dark:text-gray-500 font-medium">avg kcal</span>
   314→            </div>
   315→          </div>
   316→          <div className="flex justify-center gap-4 mt-1">
   317→            {macroData.map((m) => (
   318→              <div key={m.name} className="flex items-center gap-1.5">
   319→                <div className="w-2.5 h-2.5 rounded-full" style={{ backgroundColor: m.color }} />
   320→                <span className="text-xs text-gray-600 dark:text-gray-400 font-medium">{m.name} {Math.round(m.value)}g</span>
   321→              </div>
   322→            ))}
   323→          </div>
   324→        </Card>
   325→        <Card className="p-5 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   326→          <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100 mb-2">Weight Trend</h3>
   327→          {noWeightData ? (
   328→            <div className="h-44 flex flex-col items-center justify-center py-8 text-gray-400">
   329→              <Scale className="h-10 w-10 mb-2 text-gray-200" />
   330→              <p className="text-sm font-medium">Log your weight daily to track your progress!</p>
   331→            </div>
   332→          ) : (
   333→            <div className="h-44">
   334→              <ResponsiveContainer width="100%" height="100%">
   335→                <AreaChart data={weightChartData}>
   336→                  <defs>
   337→                    <linearGradient id="weightGrad" x1="0" y1="0" x2="0" y2="1">
   338→                      <stop offset="0%" stopColor="#10b981" stopOpacity={0.3} />
   339→                      <stop offset="100%" stopColor="#10b981" stopOpacity={0.02} />
   340→                    </linearGradient>
   341→                  </defs>
   342→                  <XAxis dataKey="date" tick={{ fontSize: 10, fill: '#6b7280' }} axisLine={false} tickLine={false} />
   343→                  <YAxis domain={['dataMin - 1', 'dataMax + 1']} tick={{ fontSize: 10, fill: '#6b7280' }} axisLine={false} tickLine={false} />
   344→                  <Tooltip contentStyle={{ borderRadius: '12px', border: '1px solid #e5e7eb' }} />
   345→                  <Area type="monotone" dataKey="weight" stroke="#10b981" strokeWidth={2.5} fill="url(#weightGrad)" dot={{ r: 4, fill: '#10b981', strokeWidth: 2, stroke: '#fff' }} activeDot={{ r: 5 }} />
   346→                </AreaChart>
   347→              </ResponsiveContainer>
   348→            </div>
   349→          )}
   350→        </Card>
   351→      </div>
   352→
   353→      {/* Weekly Summary Insight */}
   354→      {summary && !allCaloriesZero && (
   355→        <Card className="p-4 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 bg-gradient-to-r from-emerald-50/50 to-teal-50/50 dark:from-emerald-900/10 dark:to-teal-900/10">
   356→          <div className="flex items-start gap-3">
   357→            <div className="w-9 h-9 rounded-xl bg-emerald-100/80 dark:bg-emerald-900/30 flex items-center justify-center shrink-0">
   358→              <Lightbulb className="h-[18px] w-[18px] text-emerald-500" />
   359→            </div>
   360→            <div>
   361→              <h3 className="text-sm font-bold text-gray-800 dark:text-gray-200">Weekly Insight</h3>
   362→              <p className="text-xs text-gray-600 dark:text-gray-400 mt-1 leading-relaxed">
   363→                {(() => {
   364→                  const avgCal = Math.round(summary?.avgCalories || 0);
   365→                  const targetCal = chartData[0]?.target || 0;
   366→                  const avgProtein = Math.round(summary?.avgProtein || 0);
   367→                  const targetProtein = summary?.targetProtein ? Math.round(summary.targetProtein as number) : 0;
   368→                  const proteinPct = targetProtein > 0 ? Math.round((avgProtein / targetProtein) * 100) : 0;
   369→                  const bestDay = weeklyData.reduce((best: Record<string, unknown> | null, d) => {
   370→                    const cal = (d.consumed as Record<string, number>)?.calories || 0;
   371→                    const bestCal = best ? (best.consumed as Record<string, number>)?.calories || 0 : 0;
   372→                    return cal > bestCal ? d : best;
   373→                  }, null);
   374→                  const bestDayName = bestDay ? format(parseISO(bestDay.date as string), 'EEEE') : '';
   375→                  const parts = [`You averaged ${avgCal} kcal/day this week.`];
   376→                  if (proteinPct > 0) parts.push(`Protein intake was ${proteinPct}% of target.`);
   377→                  if (targetCal > 0 && avgCal < targetCal) parts.push(`You're ${targetCal - avgCal} kcal under daily target.`);
   378→                  else if (targetCal > 0 && avgCal > targetCal) parts.push(`You're ${avgCal - targetCal} kcal over daily target.`);
   379→                  if (bestDayName) parts.push(`Best day: ${bestDayName}.`);
   380→                  parts.push('Keep it up!');
   381→                  return parts.join(' ');
   382→                })()}
   383→              </p>
   384→            </div>
   385→          </div>
   386→        </Card>
   387→      )}
   388→
   389→      {/* Stats Cards */}
   390→      <div className="grid grid-cols-2 gap-3">
   391→        {statCards.map((s) => (
   392→          <Card key={s.label} className="p-4 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   393→            <div className="flex items-center gap-2.5 mb-2">
   394→              <div className={`w-10 h-10 ${s.iconBg} rounded-lg flex items-center justify-center`}>
   395→                <s.icon className={`h-5 w-5 ${s.iconColor}`} />
   396→              </div>
   397→              <span className="text-xs text-gray-500 dark:text-gray-400 font-medium leading-tight">{s.label}</span>
   398→            </div>
   399→            <p className="text-2xl font-bold text-gray-900 dark:text-gray-100 pl-[60px] tabular-nums">{s.value}</p>
   400→          </Card>
   401→        ))}
   402→      </div>
   403→
   404→      {/* Achievements Grid */}
   405→      {achievements.length > 0 && (
   406→        <Card className="p-4 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   407→          <div className="flex items-center gap-2 mb-4">
   408→            <Trophy className="h-4 w-4 text-amber-500" />
   409→            <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100">Achievements</h3>
   410→            <span className="text-xs text-gray-400 dark:text-gray-500 ml-auto">{achievements.filter((a) => a.earned).length}/8 Unlocked</span>
   411→          </div>
   412→          <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   413→            {achievements.map((ach) => (
   414→              <div
   415→                key={ach.id}
   416→                className={`relative p-3 rounded-xl border transition-colors ${
   417→                  ach.earned
   418→                    ? 'bg-white dark:bg-gray-900 border-l-4 border-l-emerald-500 border-t-0 border-r-0 border-b-0 border-emerald-200 dark:border-gray-700'
   419→                    : 'bg-gray-50 dark:bg-gray-800/50 border-gray-200 dark:border-gray-700 opacity-50'
   420→                }`}
   421→              >
   422→                <div className="flex items-center gap-2 mb-1">
   423→                  {ach.earned ? (
   424→                    <span className="text-3xl leading-none">{ach.icon}</span>
   425→                  ) : (
   426→                    <Lock className="h-6 w-6 text-gray-400 dark:text-gray-500" />
   427→                  )}
   428→                </div>
   429→                <p className={`text-sm font-semibold ${ach.earned ? 'text-gray-900 dark:text-gray-100' : 'text-gray-400 dark:text-gray-500'}`}>
   430→                  {ach.name}
   431→                </p>
   432→                <p className="text-xs text-gray-500 dark:text-gray-400 mt-0.5 leading-snug">
   433→                  {ach.description}
   434→                </p>
   435→              </div>
   436→            ))}
   437→          </div>
   438→        </Card>
   439→      )}
   440→
   441→      {/* Calorie Breakdown Table */}
   442→      <motion.div
   443→        initial={{ opacity: 0, y: 8 }}
   444→        animate={{ opacity: 1, y: 0 }}
   445→        transition={{ delay: 0.2 }}
   446→      >
   447→        <Card className="p-0 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900 overflow-hidden">
   448→          {/* Table header */}
   449→          <div className="bg-gray-50/80 px-5 py-3 border-b border-gray-100">
   450→            <h3 className="text-base font-semibold text-gray-800">Calorie Breakdown</h3>
   451→            <p className="text-xs text-gray-400 mt-0.5">This week&apos;s daily calorie tracking</p>
   452→          </div>
   453→
   454→          {breakdownData.length === 0 ? (
   455→            <div className="py-10 flex flex-col items-center justify-center text-gray-400">
   456→              <Flame className="h-10 w-10 mb-2 text-gray-200" />
   457→              <p className="text-sm font-medium">No meals logged yet this week</p>
   458→              <p className="text-xs text-gray-300 mt-1">Start logging meals to see your breakdown</p>
   459→            </div>
   460→          ) : (
   461→            <>
   462→              {/* Column headers */}
   463→              <div className="grid grid-cols-12 gap-1 px-5 py-2.5 text-xs font-semibold text-gray-500 uppercase tracking-wider border-b border-gray-100">

   465→                <div className="col-span-3 text-right">Eaten</div>
   466→                <div className="col-span-3 text-right">Target</div>
   467→                <div className="col-span-3 text-right">Diff</div>
   468→              </div>
   469→
   470→              {/* Table rows */}
   471→              <div className="divide-y divide-gray-50">
   472→                {breakdownData.map((row, idx) => {
   473→                  const isOver = row.diff > 0;
   474→                  const isToday = row.date === format(new Date(), 'yyyy-MM-dd');

   477→                      key={row.date}
   478→                      className={`grid grid-cols-12 gap-1 py-3 px-4 items-center text-xs hover:bg-gray-50/50 transition-colors ${

   480→                          ? 'bg-emerald-50/80 border-l-[3px] border-l-emerald-500'

   490→                      </div>
   491→
   492→                      {/* Consumed */}
   493→                      <div className="col-span-3 text-right font-medium text-gray-800">
   494→                        {row.consumedCal.toLocaleString()}

   496→                      </div>
   497→
   498→                      {/* Target */}
   499→                      <div className="col-span-3 text-right text-gray-500">
   500→                        {row.targetCal.toLocaleString()}

   502→
   503→                      {/* Difference + mini bar */}
   504→                      <div className="col-span-3 flex flex-col items-end gap-1">

   507→                        </span>
   508→                        <div className="w-full h-1.5 bg-gray-100 rounded-full overflow-hidden">
   509→                          <motion.div
   510→                            className={`h-full rounded-full ${isOver ? 'bg-red-400' : 'bg-emerald-400'}`}
   511→                            initial={{ width: 0 }}
   512→                            animate={{ width: `${Math.min(row.pct, 100)}%` }}
   513→                            transition={{ duration: 0.6, delay: idx * 0.05 }}
   514→                          />
   515→                        </div>
   516→                      </div>
   517→                    </div>
   518→                  );
   519→                })}
   520→              </div>
   521→
   522→              {/* Weekly total row */}
   523→              <div className="grid grid-cols-12 gap-1 px-5 py-3 bg-gray-50 border-t border-gray-200 text-xs font-bold">
   524→                <div className="col-span-3 text-gray-700">Total</div>
   525→                <div className="col-span-3 text-right text-gray-900">{weeklyTotalConsumed.toLocaleString()}</div>
   526→                <div className="col-span-3 text-right text-gray-500">{weeklyTotalTarget.toLocaleString()}</div>
   527→                <div className={`col-span-3 text-right ${weeklyTotalDiff > 0 ? 'text-red-500' : 'text-emerald-600'}`}>{
   528→                  weeklyTotalDiff > 0 ? '+' : ''}{weeklyTotalDiff}
   529→                </div>
   530→              </div>
   531→            </>
   532→          )}
   533→        </Card>
   534→      </motion.div>
   535→
   536→      {/* Water Tracking */}
   537→      <Card className="p-5 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   538→        <div className="flex items-center justify-between mb-4">
   539→          <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100 flex items-center gap-2">
   540→            <Droplets className="h-4 w-4 text-blue-500" /> Water Intake
   541→          </h3>

   543→        </div>
   544→        <div className="flex items-center justify-center gap-4">
   545→          <Button
   546→            size="icon"
   547→            variant="outline"
   548→            className="h-10 w-10 rounded-xl"
   549→            onClick={() => handleAddWater(-1)}
   550→            disabled={waterGlasses <= 0}
   551→          >
   552→            <Minus className="h-4 w-4" />
   553→          </Button>
   554→          <div className="flex gap-2 flex-wrap justify-center">
   555→            {Array.from({ length: 8 }, (_, i) => {
   556→              const isFilled = i < waterGlasses;
   557→              const isJustFilled = justFilledIndex === i;
   558→              return (
   559→                <motion.div
   560→                  key={i}
   561→                  className="relative"
   562→                  animate={isJustFilled ? { scale: [1, 1.25, 1] } : {}}

   567→                      isFilled
   568→                        ? 'bg-blue-100 border-blue-300'
   569→                        : 'bg-gray-50 border-gray-200'

   573→                    }}
   574→                    transition={{ duration: 0.4 }}

   576→                    <Droplets
   577→                      className={`h-4 w-4 transition-colors ${
   578→                        isFilled ? 'text-blue-500' : 'text-gray-200'
   579→                      }`}
   580→                    />
   581→                  </motion.div>
   582→                  {isJustFilled && (
   583→                    <motion.div
   584→                      className="absolute inset-0 rounded-md bg-blue-400/20"
   585→                      initial={{ opacity: 1, scale: 0.8 }}
   586→                      animate={{ opacity: 0, scale: 1.4 }}
   587→                      transition={{ duration: 0.6 }}
   588→                    />
   589→                  )}
   590→                </motion.div>
   591→              );
   592→            })}
   593→          </div>
   594→          <Button
   595→            size="icon"
   596→            className="h-10 w-10 bg-blue-500 hover:bg-blue-600 rounded-xl"
   597→            onClick={() => handleAddWater(1)}
   598→            disabled={waterGlasses >= 16}
   599→          >
   600→            <Plus className="h-4 w-4" />
   601→          </Button>
   602→        </div>
   603→      </Card>
   604→
   605→      {/* Weight Logging */}
   606→      <Card className="p-5 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   607→        <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100 mb-3 flex items-center gap-2">
   608→          <Scale className="h-4 w-4 text-purple-500" /> Log Weight
   609→        </h3>
   610→        <div className="flex gap-2">
   611→          <Input
   612→            type="number"
   613→            placeholder="e.g., 74.5"
   614→            value={weightInput}
   615→            onChange={(e) => setWeightInput(e.target.value)}
   616→            className="flex-1 rounded-xl h-11"
   617→          />
   618→          <Input
   619→            placeholder="Optional notes..."
   620→            value={weightNotes}
   621→            onChange={(e) => setWeightNotes(e.target.value)}
   622→            className="flex-1 rounded-xl h-11"
   623→          />
   624→          <Button
   625→            className="bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl min-h-[44px] font-semibold px-5"
   626→            onClick={handleLogWeight}
   627→            disabled={!weightInput}
   628→          >
   629→            Log
   630→          </Button>
   631→        </div>
   632→        {weightLogs.length > 0 && (
   633→          <div className="mt-4 space-y-1">
   634→            <p className="text-xs text-gray-500 font-semibold uppercase tracking-wider mb-2">Recent Entries</p>
   635→            {weightLogs.slice(0, 5).map((w) => (
   636→              <div key={w.id as string} className="flex justify-between text-xs text-gray-600 py-2 px-3 border-b border-gray-50 last:border-0 hover:bg-gray-50/50 transition-colors rounded">
   637→                <span>{format(parseISO(w.logDate as string), 'MMM d, yyyy')}</span>
   638→                <span className="font-semibold">{w.weightKg as number} kg</span>
   639→              </div>
   640→            ))}
   641→          </div>
   642→        )}
   643→      </Card>
   644→    </motion.div>
   645→  );
