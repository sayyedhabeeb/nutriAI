     2→

     4→import { format } from 'date-fns';

     6→import { toast } from 'sonner';
     7→import { Button } from '@/components/ui/button';
     8→import { Card } from '@/components/ui/card';
     9→import { Input } from '@/components/ui/input';
    10→import { Label } from '@/components/ui/label';
    11→import { Badge } from '@/components/ui/badge';
    12→import { Skeleton } from '@/components/ui/skeleton';
    13→import { ScrollArea } from '@/components/ui/scroll-area';
    14→import {

    17→import {
    18→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    19→} from '@/components/ui/select';
    20→import {

    23→import {

    25→} from '@/components/ui/collapsible';
    26→import {
    27→  User, Search, Droplets, Camera, Clock, Zap, ChevronRight, Lightbulb, Sparkles, ChevronDown, Loader2,
    28→} from 'lucide-react';
    29→import { apiFetch } from './api';
    30→import {
    31→  SLOTS, SLOT_LABELS, SLOT_ICONS, SLOT_BORDER_COLORS, SLOT_BADGE_COLORS,
    32→  SLOT_GRADIENT_COLORS, fadeIn,
    33→} from './constants';
    34→import { CalorieRing, NutritionFactsLabel } from './shared';
    35→import type { ViewType, NutritionData, MealRecommendation, SearchMeal, MealPlanItemData } from './types';
    36→
    37→export function DashboardView({ onNavigate }: { onNavigate: (v: ViewType) => void }) {
    38→  const [user, setUser] = useState<Record<string, unknown> | null>(null);
    39→  const [nutrition, setNutrition] = useState<NutritionData | null>(null);
    40→  const [recommendations, setRecommendations] = useState<Record<string, MealRecommendation[]>>({});
    41→  const [loading, setLoading] = useState(true);
    42→  const [streak, setStreak] = useState(0);
    43→
    44→  // Log meal dialog state
    45→  const [logDialog, setLogDialog] = useState<{
    46→    open: boolean;
    47→    meal: MealRecommendation;
    48→    slot: string;
    49→    baseNutritionPer100g: { calories: number; proteinG: number; carbsG: number; fatG: number } | null;
    50→    mealName: string;
    51→    cuisine: string;
    52→    mealType: string;
    53→    description?: string | null;
    54→    isVeg?: boolean;
    55→    isVegan?: boolean;
    56→  } | { open: false }>({ open: false });
    57→  const [servingGms, setServingGms] = useState(100);
    58→
    59→  // Meal search dialog state (per-slot)
    60→  const [slotSearchDialog, setSlotSearchDialog] = useState<{ open: boolean; slot: string } | { open: false }>({ open: false });
    61→  const [searchQuery, setSearchQuery] = useState('');
    62→  const [searchResults, setSearchResults] = useState<SearchMeal[]>([]);
    63→  const [searchLoading, setSearchLoading] = useState(false);
    64→  const [searchDebounce, setSearchDebounce] = useState<ReturnType<typeof setTimeout> | null>(null);
    65→
    66→  // Meal detail sheet
    67→  const [detailSheet, setDetailSheet] = useState<{
    68→    open: boolean;
    69→    rec: MealRecommendation;
    70→  } | { open: false }>({ open: false });
    71→
    72→  // Meal plan state (Feature 2)
    73→  const [mealPlan, setMealPlan] = useState<{
    74→    exists: boolean;
    75→    items: MealPlanItemData[];
    76→  } | null>(null);
    77→  const [planLoading, setPlanLoading] = useState(false);
    78→  const [planExpanded, setPlanExpanded] = useState<Record<string, boolean>>({});
    79→
    80→  const fetchData = useCallback(async () => {
    81→    try {
    82→      const [me, nut, summaryData] = await Promise.all([
    83→        apiFetch('/api/auth/me'),
    84→        apiFetch('/api/nutrition/daily'),
    85→        apiFetch('/api/progress/summary?period=week').catch(() => ({ summary: { totalDays: 0 } })),
    86→      ]);
    87→      setUser(me);
    88→      setNutrition(nut);
    89→
    90→      // Calculate streak from weekly data
    91→      const weeklyData = await apiFetch('/api/progress/weekly').catch(() => [] as Record<string, unknown>[]);
    92→      let streakCount = 0;
    93→      for (let i = (weeklyData as Record<string, unknown>[]).length - 1; i >= 0; i--) {
    94→        const day = weeklyData[i];
    95→        const consumed = (day.consumed as Record<string, number>)?.calories || 0;
    96→        if (consumed > 0) streakCount++;
    97→        else break;
    98→      }
    99→      setStreak(streakCount);
   100→
   101→      const recs = await Promise.all(
   102→        SLOTS.map(async (slot) => {
   103→          try {
   104→            const r = await apiFetch(`/api/recommendations?slot=${slot}`);
   105→            return { slot, recs: r.recommendations || [] };
   106→          } catch { return { slot, recs: [] }; }
   107→        })
   108→      );
   109→      const recMap: Record<string, MealRecommendation[]> = {};
   110→      for (const r of recs) recMap[r.slot] = r.recs;
   111→      setRecommendations(recMap);
   112→
   113→      // Fetch meal plan (Feature 2)
   114→      try {
   115→        const plan = await apiFetch('/api/meal-plan');
   116→        setMealPlan({ exists: plan.exists, items: plan.items || [] });
   117→      } catch {
   118→        setMealPlan(null);
   119→      }
   120→    } catch {
   121→      toast.error('Failed to load dashboard');
   122→    } finally {
   123→      setLoading(false);
   124→    }
   125→  }, []);
   126→
   127→  useEffect(() => { fetchData(); }, [fetchData]);
   128→
   129→  const handleLogMeal = async () => {
   130→    if (!logDialog.open) return;
   131→    try {
   132→      await apiFetch('/api/food-logs', {
   133→        method: 'POST',
   134→        body: JSON.stringify({ mealId: logDialog.meal.meal.id, servingGms, mealSlot: logDialog.slot }),
   135→      });
   136→      toast.success(`Logged ${logDialog.mealName}!`);
   137→      setLogDialog({ open: false });
   138→      fetchData();
   139→    } catch (err) { toast.error((err as Error).message); }
   140→  };
   141→
   142→  // Per-slot search with debounce
   143→  const handleSlotSearch = (query: string) => {
   144→    setSearchQuery(query);
   145→    if (searchDebounce) clearTimeout(searchDebounce);
   146→    if (!query.trim()) { setSearchResults([]); return; }
   147→    const timer = setTimeout(async () => {
   148→      setSearchLoading(true);
   149→      try {
   150→        const res = await apiFetch(`/api/meals/search?q=${encodeURIComponent(query)}`);
   151→        setSearchResults((res.meals || []) as SearchMeal[]);
   152→      } catch { toast.error('Search failed'); }
   153→      finally { setSearchLoading(false); }
   154→    }, 350);
   155→    setSearchDebounce(timer);
   156→  };
   157→
   158→  const handleLogFromSearch = (meal: SearchMeal) => {
   159→    const slot = slotSearchDialog.open ? slotSearchDialog.slot : 'lunch';
   160→    setServingGms(meal.baseServingGms || 100);
   161→    setLogDialog({
   162→      open: true,
   163→      meal: {
   164→        meal: { id: meal.id, name: meal.name, mealType: meal.mealType, cuisine: meal.cuisine, imageUrl: null, prepTimeMin: meal.prepTimeMin, isVeg: meal.isVeg, isVegan: meal.isVegan },
   165→        score: 0, recommendedServingGms: meal.baseServingGms || 100, estimatedNutrition: meal.nutrition ? { calories: meal.nutrition.calories, proteinG: meal.nutrition.proteinG, carbsG: meal.nutrition.carbsG, fatG: meal.nutrition.fatG } : null,
   166→        baseNutritionPer100g: meal.nutrition ? { calories: meal.nutrition.calories, proteinG: meal.nutrition.proteinG, carbsG: meal.nutrition.carbsG, fatG: meal.nutrition.fatG } : null,
   167→      },
   168→      slot,
   169→      baseNutritionPer100g: meal.nutrition || null,
   170→      mealName: meal.name,
   171→      cuisine: meal.cuisine,
   172→      mealType: meal.mealType,
   173→      isVeg: meal.isVeg,
   174→      isVegan: meal.isVegan,
   175→    });
   176→    setSlotSearchDialog({ open: false });
   177→    setSearchQuery('');
   178→    setSearchResults([]);
   179→  };
   180→
   181→  const handleWaterAdd = async () => {
   182→    try {
   183→      await apiFetch('/api/water-log', { method: 'POST', body: JSON.stringify({ glasses: 1 }) });
   184→      toast.success('\uD83D\uDCA7 +1 glass of water');
   185→    } catch { toast.error('Failed to log water'); }
   186→  };
   187→
   188→  // Generate meal plan (Feature 2)
   189→  const handleGeneratePlan = async () => {
   190→    setPlanLoading(true);
   191→    try {
   192→      const result = await apiFetch('/api/meal-plan/generate');
   193→      toast.success(result.status === 'existing' ? 'Plan already exists for today!' : "Today's meal plan generated!");
   194→      // Fetch the plan
   195→      const plan = await apiFetch('/api/meal-plan');
   196→      setMealPlan({ exists: plan.exists, items: plan.items || [] });
   197→    } catch (err) { toast.error((err as Error).message); }
   198→    finally { setPlanLoading(false); }
   199→  };
   200→
   201→  const firstName = (user?.profile as Record<string, unknown> | null)?.firstName || (user?.name as string) || 'there';
   202→  const todayStr = format(new Date(), 'EEEE, MMMM d');
   203→
   204→  // Insights logic (Feature 3)
   205→  const targetCal = nutrition?.targets?.calories || 0;
   206→  const consumedCal = nutrition?.consumed?.calories || 0;
   207→  const pctConsumed = targetCal > 0 ? consumedCal / targetCal : 0;
   208→  const targetProtein = nutrition?.targets?.proteinG || 0;
   209→  const consumedProtein = nutrition?.consumed?.proteinG || 0;
   210→  const pctProtein = targetProtein > 0 ? consumedProtein / targetProtein : 0;
   211→  const hour = new Date().getHours();
   212→  const timeOfDayTip = hour < 12
   213→    ? 'Start your day with a protein-rich breakfast!'
   214→    : hour < 17
   215→      ? 'Keep up the good work! Stay consistent with your meals.'
   216→      : 'Plan a balanced dinner to finish strong today.';
   217→
   218→  if (loading) {
   219→    return (
   220→      <div className="p-4 max-w-lg mx-auto space-y-4">
   221→        <Skeleton className="h-8 w-56 rounded-xl" />
   222→        <div className="flex justify-center"><Skeleton className="h-[180px] w-[180px] rounded-full" /></div>
   223→        <div className="space-y-3"><Skeleton className="h-10 w-full rounded-xl" /><Skeleton className="h-10 w-full rounded-xl" /><Skeleton className="h-10 w-full rounded-xl" /></div>
   224→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4"><Skeleton className="h-56 w-full rounded-xl" /><Skeleton className="h-56 w-full rounded-xl" /></div>
   225→      </div>
   226→    );
   227→  }
   228→
   229→  return (
   230→    <motion.div {...fadeIn} className="p-4 max-w-lg mx-auto space-y-4">
   231→      {/* ═══ Hero Section with gradient bg & decorative blobs ═══ */}
   232→      <div className="relative overflow-hidden rounded-2xl bg-gradient-to-r from-emerald-500/5 to-teal-500/5 p-5 border border-emerald-100/50">
   233→        <div className="absolute -top-8 -right-8 w-24 h-24 bg-emerald-200/20 rounded-full blur-2xl" />
   234→        <div className="absolute -bottom-6 -left-6 w-20 h-20 bg-teal-200/20 rounded-full blur-2xl" />
   235→        <div className="absolute top-1/2 right-1/4 w-12 h-12 bg-green-200/10 rounded-full blur-xl" />
   236→        <div className="relative flex items-start justify-between">
   237→          <div>
   238→            <h1 className="text-xl font-bold text-gray-900">Hi, {firstName}! {'\uD83D\uDC4B'}</h1>
   239→            <div className="flex items-center gap-1.5 mt-1 text-sm text-gray-500">
   240→              <CalendarIcon className="h-3.5 w-3.5" />
   241→              <span>{todayStr}</span>
   242→            </div>
   243→          </div>
   244→          <div className="flex items-center gap-2">
   245→            {streak > 0 && (
   246→              <Badge className="bg-amber-50 text-amber-700 border border-amber-200 px-2.5 py-1 gap-1 shadow-sm">
   247→                <Zap className="h-3 w-3" />{streak} day{streak > 1 ? 's' : ''}
   248→              </Badge>
   249→            )}
   250→            <div className="w-10 h-10 bg-emerald-100 rounded-xl flex items-center justify-center shadow-sm">
   251→              <User className="h-5 w-5 text-emerald-600" />
   252→            </div>
   253→          </div>
   254→        </div>
   255→      </div>
   256→
   257→      {/* ═══ Calorie Ring Card ═══ */}
   258→      <Card className="p-6 rounded-xl shadow-sm">
   259→        <div className="flex justify-center">
   260→          <CalorieRing consumed={consumedCal} target={targetCal} />
   261→        </div>
   262→        {pctConsumed >= 1 && (
   263→          <p className="text-center text-sm text-rose-600 font-medium mt-2">
   264→            You&apos;ve reached your calorie goal! Great job! {'\uD83C\uDF89'}
   265→          </p>
   266→        )}
   267→      </Card>
   268→
   269→      {/* ═══ Macro Progress Bars ═══ */}
   270→      <Card className="p-4 rounded-xl shadow-sm space-y-3">
   271→        <h3 className="text-sm font-semibold text-gray-700">Today&apos;s Macros</h3>
   272→        {[
   273→          { label: 'Protein', val: consumedProtein, target: targetProtein, from: '#3b82f6', to: '#60a5fa', unit: 'g' },
   274→          { label: 'Carbs', val: nutrition?.consumed?.carbsG || 0, target: nutrition?.targets?.carbsG || 0, from: '#f59e0b', to: '#fbbf24', unit: 'g' },
   275→          { label: 'Fat', val: nutrition?.consumed?.fatG || 0, target: nutrition?.targets?.fatG || 0, from: '#f43f5e', to: '#fb7185', unit: 'g' },
   276→        ].map((m) => {
   277→          const pct = Math.min(Math.round((m.val / (m.target || 1)) * 100), 100);
   278→          return (
   279→            <div key={m.label}>
   280→              <div className="flex justify-between text-xs mb-1.5">
   281→                <span className="text-gray-600 font-medium">{m.label}</span>
   282→                <span className="text-gray-500">{m.val}{m.unit} / {m.target}{m.unit} &middot; {pct}%</span>
   283→              </div>
   284→              <div className="h-2.5 bg-gray-100 rounded-full overflow-hidden">
   285→                <motion.div
   286→                  className="h-full rounded-full"
   287→                  style={{ background: `linear-gradient(90deg, ${m.from}, ${m.to})` }}
   288→                  initial={{ width: 0 }}
   289→                  animate={{ width: `${pct}%` }}
   290→                  transition={{ duration: 0.8 }}
   291→                />
   292→              </div>
   293→            </div>
   294→          );
   295→        })}
   296→      </Card>
   297→
   298→      {/* ═══ Feature 3: Today's Insights Card ═══ */}
   299→      <motion.div
   300→        initial={{ opacity: 0, y: 8 }}
   301→        animate={{ opacity: 1, y: 0 }}
   302→        transition={{ delay: 0.3 }}
   303→      >
   304→        <Card className={`p-4 rounded-xl shadow-sm border-l-4 ${pctConsumed > 1 ? 'border-l-amber-500' : pctConsumed >= 0.75 ? 'border-l-emerald-500' : 'border-l-blue-500'}`}>
   305→          <div className="flex items-start gap-3">
   306→            <div className={`w-9 h-9 rounded-xl flex items-center justify-center shrink-0 ${pctConsumed > 1 ? 'bg-amber-50' : pctConsumed >= 0.75 ? 'bg-emerald-50' : 'bg-blue-50'}`}>
   307→              <Lightbulb className={`h-4.5 w-4.5 ${pctConsumed > 1 ? 'text-amber-500' : pctConsumed >= 0.75 ? 'text-emerald-500' : 'text-blue-500'}`} />
   308→            </div>
   309→            <div className="flex-1 min-w-0 space-y-2">
   310→              <h3 className="text-sm font-semibold text-gray-900">Today&apos;s Insights</h3>
   311→
   312→              {/* Calorie status message */}
   313→              {pctConsumed > 1 && (
   314→                <p className="text-xs text-amber-700 leading-relaxed">
   315→                  You&apos;ve exceeded your calorie goal by <span className="font-bold">{Math.round(consumedCal - targetCal)} kcal</span>. Consider lighter meals for the rest of the day.
   316→                </p>
   317→              )}
   318→              {pctConsumed >= 0.75 && pctConsumed <= 1 && (
   319→                <p className="text-xs text-emerald-700 leading-relaxed">
   320→                  Great progress! You&apos;re <span className="font-bold">{Math.round(targetCal - consumedCal)} kcal</span> away from your daily goal.
   321→                </p>
   322→              )}
   323→              {pctConsumed < 0.75 && pctConsumed > 0 && (
   324→                <p className="text-xs text-blue-700 leading-relaxed">
   325→                  You still have <span className="font-bold">{Math.round(targetCal - consumedCal)} kcal</span> remaining. Time to fuel up!
   326→                </p>
   327→              )}
   328→              {consumedCal === 0 && (
   329→                <p className="text-xs text-gray-500 leading-relaxed">
   330→                  No meals logged yet today. Start with a balanced breakfast!
   331→                </p>
   332→              )}
   333→
   334→              {/* Protein status */}
   335→              {consumedProtein > 0 && pctProtein >= 0.8 && (
   336→                <p className="text-xs text-emerald-600 font-medium">
   337→                  Protein goal almost reached! {'\uD83C\uDF89'} ({Math.round(consumedProtein)}/{Math.round(targetProtein)}g)
   338→                </p>
   339→              )}
   340→
   341→              {/* Time-of-day tip */}
   342→              <p className="text-xs text-gray-500 italic">{'\uD83D\uDCAC'} {timeOfDayTip}</p>
   343→            </div>
   344→          </div>
   345→        </Card>
   346→      </motion.div>
   347→
   348→      {/* ═══ Feature 2: Meal Plan Generator Banner ═══ */}
   349→      <motion.div
   350→        initial={{ opacity: 0, y: 8 }}
   351→        animate={{ opacity: 1, y: 0 }}
   352→        transition={{ delay: 0.4 }}
   353→      >
   354→        <Card className="p-4 rounded-xl shadow-sm bg-gradient-to-r from-emerald-500/5 to-teal-500/5 border border-emerald-100/50">
   355→          {!mealPlan?.exists ? (
   356→            <div className="flex items-center justify-between">
   357→              <div className="flex items-center gap-3">
   358→                <div className="w-9 h-9 bg-emerald-100 rounded-xl flex items-center justify-center">
   359→                  <Sparkles className="h-4.5 w-4.5 text-emerald-600" />
   360→                </div>
   361→                <div>
   362→                  <h3 className="text-sm font-semibold text-gray-900">Today&apos;s Plan</h3>
   363→                  <p className="text-xs text-gray-500">Generate a personalized meal plan</p>
   364→                </div>
   365→              </div>
   366→              <Button
   367→                size="sm"
   368→                className="bg-emerald-600 hover:bg-emerald-700 text-white rounded-lg font-medium min-h-[36px]"
   369→                onClick={handleGeneratePlan}
   370→                disabled={planLoading}
   371→              >
   372→                {planLoading ? <Loader2 className="h-4 w-4 animate-spin mr-1.5" /> : <Sparkles className="h-4 w-4 mr-1.5" />}
   373→                Generate
   374→              </Button>
   375→            </div>
   376→          ) : (
   377→            <div className="space-y-2">
   378→              <div className="flex items-center justify-between">
   379→                <div className="flex items-center gap-3">
   380→                  <div className="w-9 h-9 bg-emerald-100 rounded-xl flex items-center justify-center">
   381→                    <Sparkles className="h-4.5 w-4.5 text-emerald-600" />
   382→                  </div>
   383→                  <div>
   384→                    <h3 className="text-sm font-semibold text-gray-900">Today&apos;s Meal Plan</h3>
   385→                    <p className="text-xs text-gray-500">AI-generated for your goals</p>
   386→                  </div>
   387→                </div>
   388→                <Badge className="bg-emerald-50 text-emerald-700 border-emerald-200 text-xs">Active</Badge>
   389→              </div>
   390→
   391→              {/* Plan items by slot */}
   392→              {SLOTS.map((slot) => {
   393→                const slotItems = (mealPlan?.items || []).filter((i) => i.mealSlot === slot);
   394→                if (slotItems.length === 0) return null;
   395→                const isExpanded = planExpanded[slot] || false;
   396→                return (
   397→                  <Collapsible
   398→                    key={slot}
   399→                    open={isExpanded}
   400→                    onOpenChange={(open) => setPlanExpanded((prev) => ({ ...prev, [slot]: open }))}
   401→                  >
   402→                    <CollapsibleTrigger asChild>
   403→                      <button className={`w-full flex items-center gap-2 p-2.5 rounded-xl border border-gray-100 hover:bg-white/80 transition-colors text-left ${SLOT_GRADIENT_COLORS[slot].from} ${SLOT_GRADIENT_COLORS[slot].to}`}>
   404→                        <span>{SLOT_ICONS[slot]}</span>
   405→                        <span className="text-xs font-semibold text-gray-700">{SLOT_LABELS[slot]}</span>
   406→                        <span className="text-xs text-gray-500 truncate flex-1">{slotItems[0]?.meal.name}</span>
   407→                        <ChevronDown className={`h-3.5 w-3.5 text-gray-400 transition-transform ${isExpanded ? 'rotate-180' : ''}`} />
   408→                      </button>
   409→                    </CollapsibleTrigger>
   410→                    <AnimatePresence>
   411→                      {isExpanded && (
   412→                        <CollapsibleContent forceMount>
   413→                          <motion.div
   414→                            initial={{ height: 0, opacity: 0 }}
   415→                            animate={{ height: 'auto', opacity: 1 }}
   416→                            exit={{ height: 0, opacity: 0 }}
   417→                            transition={{ duration: 0.2 }}
   418→                            className="overflow-hidden"
   419→                          >
   420→                            <div className="pl-4 pr-1 pt-1 space-y-1.5">
   421→                              {slotItems.map((item) => (
   422→                                <div key={item.id} className="flex items-center justify-between p-2.5 rounded-lg bg-white border border-gray-100">
   423→                                  <div className="flex-1 min-w-0">
   424→                                    <p className="text-xs font-medium text-gray-800 truncate">{item.meal.name}</p>
   425→                                    <div className="flex items-center gap-1.5 mt-0.5">
   426→                                      <Badge variant="outline" className="text-[10px] px-1.5 py-0 h-4 font-medium border-gray-200 text-gray-500">{item.meal.cuisine}</Badge>
   427→                                      <span className="text-[10px] text-gray-400">{item.servingGms}g</span>
   428→                                    </div>
   429→                                    {item.nutrition && (
   430→                                      <div className="flex gap-2 mt-0.5 text-[10px] text-gray-500">
   431→                                        <span className="text-orange-600 font-medium">{item.nutrition.calories} kcal</span>
   432→                                        <span>P: {item.nutrition.proteinG}g</span>
   433→                                        <span>C: {item.nutrition.carbsG}g</span>
   434→                                        <span>F: {item.nutrition.fatG}g</span>
   435→                                      </div>
   436→                                    )}
   437→                                  </div>
