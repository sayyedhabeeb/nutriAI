  1114→function FoodLogView() {

  1117→  const [loading, setLoading] = useState(true);

  1119→  const [relogSlot, setRelogSlot] = useState('lunch');
  1120→
  1121→  const dates = Array.from({ length: 7 }, (_, i) => format(subDays(new Date(), 6 - i), 'yyyy-MM-dd'));
  1122→
  1123→  const fetchLog = useCallback(async () => {
  1124→    setLoading(true);
  1125→    try {
  1126→      const data = await apiFetch(`/api/food-logs?date=${selectedDate}`);
  1127→      setFoodLog(data);
  1128→    } catch { toast.error('Failed to load food log'); }
  1129→    finally { setLoading(false); }

  1135→    try {
  1136→      await apiFetch(`/api/food-logs?id=${itemId}`, { method: 'DELETE' });

  1138→      fetchLog();
  1139→    } catch (err) { toast.error((err as Error).message); }
