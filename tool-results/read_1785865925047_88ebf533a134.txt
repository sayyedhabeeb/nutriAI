   182→   182→        consumedCal,

   222→   222→          onClick={() => setExportOpen(true)}
   223→   223→          className="border border-gray-200 dark:border-gray-700 bg-white dark:bg-gray-800 shadow-sm hover:shadow-md rounded-xl px-3.5 py-2 text-xs font-medium text-gray-600 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-700 transition-all flex items-center gap-2"

   288→   288→              <BarChart3 className="h-12 w-12 mb-3 text-gray-200" />

   293→   293→              <BarChart data={chartData} barCategoryGap="20%">
   294→   294→                <XAxis dataKey="date" tick={{ fontSize: 11, fill: '#6b7280' }} axisLine={false} tickLine={false} />

   316→   316→      {/* Macro Pie + Weight Trend */}
   317→   317→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
