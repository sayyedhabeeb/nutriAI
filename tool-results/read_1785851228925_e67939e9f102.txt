   318→function AuthView({ onAuth }: { onAuth: (v: ViewType) => void }) {

   572→// CALORIE RING SVG (with gradient stroke)
   573→// ═══════════════════════════════════════════════════════════
   574→function CalorieRing({ consumed, target }: { consumed: number; target: number }) {
   575→  const pct = target > 0 ? Math.min(consumed / target, 1) : 0;
   576→  const radius = 70;
   577→  const circumference = 2 * Math.PI * radius;
   578→  const strokeDashoffset = circumference * (1 - pct);
   579→  const remaining = Math.max(0, target - consumed);
   580→  const gradId = 'calRingGrad';
   581→
   582→  return (
   583→    <div className="relative inline-flex items-center justify-center">
   584→      <svg width="180" height="180" viewBox="0 0 180 180" className="-rotate-90">
   585→        <defs>
   586→          <linearGradient id={gradId} x1="0%" y1="0%" x2="100%" y2="0%">
   587→            <stop offset="0%" stopColor="#10b981" />
   588→            <stop offset="100%" stopColor="#34d399" />
   589→          </linearGradient>
   590→        </defs>
   591→        <circle cx="90" cy="90" r={radius} fill="none" stroke="#e5e7eb" strokeWidth="12" />
   592→        <motion.circle
   593→          cx="90" cy="90" r={radius} fill="none"
   594→          stroke={pct > 1 ? '#f43f5e' : `url(#${gradId})`}
   595→          strokeWidth="12" strokeLinecap="round"
   596→          strokeDasharray={circumference}
   597→          initial={{ strokeDashoffset: circumference }}
   598→          animate={{ strokeDashoffset }}
   599→          transition={{ duration: 1, ease: 'easeOut' }}
   600→        />
   601→      </svg>
   602→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   603→        <Flame className="h-5 w-5 text-orange-500 mb-1" />
   604→        <span className="text-2xl font-bold text-gray-900">{consumed}</span>
   605→        <span className="text-xs text-gray-500">of {target} kcal</span>
   606→        <span className="text-xs text-emerald-600 font-semibold mt-1">{remaining} left</span>

  1301→// UPLOAD VIEW
  1302→// ═══════════════════════════════════════════════════════════
  1303→function UploadView() {
  1304→  const [imagePreview, setImagePreview] = useState<string | null>(null);
  1305→  const [imageFile, setImageFile] = useState<File | null>(null);
  1306→  const [recognizing, setRecognizing] = useState(false);
  1307→  const [results, setResults] = useState<RecognizedFood[]>([]);
  1308→  const [unknownForms, setUnknownForms] = useState<Record<number, {
  1309→    confirmedName: string; confirmedPortion: number; caloriesPer100g: number;
  1310→    proteinPer100g: number; carbsPer100g: number; fatPer100g: number;
  1311→    mealType: string; cuisine: string;
  1312→  }>>({});
  1313→  const fileInputRef = useRef<HTMLInputElement>(null);
  1314→
  1315→  const handleFileSelect = (file: File) => {
  1316→    setImageFile(file);
  1317→    const reader = new FileReader();
  1318→    reader.onload = (e) => setImagePreview(e.target?.result as string);
  1319→    reader.readAsDataURL(file);
  1320→  };
  1321→
  1322→  const handleDrop = (e: React.DragEvent) => {
  1323→    e.preventDefault();
  1324→    const file = e.dataTransfer.files[0];
  1325→    if (file && file.type.startsWith('image/')) handleFileSelect(file);
  1326→  };
  1327→
  1328→  const handleRecognize = async () => {
  1329→    if (!imageFile) return;
  1330→    setRecognizing(true);
  1331→    try {
  1332→      const formData = new FormData();
  1333→      formData.append('image', imageFile);
  1334→      const data = await apiFetch('/api/food-recognize', { method: 'POST', body: formData });
  1335→      setResults(data.foods || []);
  1336→      const forms: typeof unknownForms = {};
  1337→      (data.foods || []).forEach((f: RecognizedFood, idx: number) => {
  1338→        if (f.unknown_food) {
  1339→          forms[idx] = { confirmedName: f.name, confirmedPortion: f.servingWeightGrams || 200, caloriesPer100g: 0, proteinPer100g: 0, carbsPer100g: 0, fatPer100g: 0, mealType: 'lunch', cuisine: 'Mixed' };
  1340→        }
  1341→      });
  1342→      setUnknownForms(forms);
  1343→    } catch (err) { toast.error((err as Error).message || 'Recognition failed'); }
  1344→    finally { setRecognizing(false); }
  1345→  };
  1346→
  1347→  const confidenceBadge = (conf: number) => {
  1348→    const pct = Math.round(conf * 100);
  1349→    const color = conf >= 0.9 ? 'bg-green-100 text-green-700 border-green-200' : conf >= 0.7 ? 'bg-yellow-100 text-yellow-700 border-yellow-200' : conf >= 0.5 ? 'bg-orange-100 text-orange-700 border-orange-200' : 'bg-red-100 text-red-700 border-red-200';
  1350→    return <Badge className={`${color} text-xs`}>{pct}%</Badge>;
  1351→  };
  1352→
  1353→  const handleLogRecognized = async (mealId: string, servingWeightGrams: number) => {
  1354→    try {
  1355→      await apiFetch('/api/food-logs', { method: 'POST', body: JSON.stringify({ mealId, servingGms: servingWeightGrams, mealSlot: 'lunch' }) });
  1356→      toast.success('Food logged!');
  1357→    } catch (err) { toast.error((err as Error).message); }
  1358→  };
  1359→
  1360→  const handleSubmitUnknown = async (idx: number, aiName: string) => {
  1361→    const form = unknownForms[idx];
  1362→    if (!form || !form.confirmedName || !form.caloriesPer100g) { toast.error('Please fill in all required fields'); return; }
  1363→    try {
  1364→      const data = await apiFetch('/api/unknown-food/submit', { method: 'POST', body: JSON.stringify({ aiDetectedName: aiName, ...form }) });
  1365→      toast.success('Food submitted and logged!');
  1366→      if (data?.meal?.id) await handleLogRecognized(data.meal.id, form.confirmedPortion);
  1367→    } catch (err) { toast.error((err as Error).message); }
  1368→  };
  1369→
  1370→  return (
  1371→    <motion.div {...fadeIn} className="p-4 max-w-lg mx-auto space-y-4">
  1372→      <div className="flex items-center gap-2">
  1373→        <Camera className="h-5 w-5 text-emerald-600" />
  1374→        <h1 className="text-xl font-bold text-gray-900">Scan Food</h1>
  1375→      </div>
  1376→
  1377→      {/* Dropzone */}
  1378→      <div
  1379→        onDragOver={(e) => e.preventDefault()}
  1380→        onDrop={handleDrop}
  1381→        onClick={() => fileInputRef.current?.click()}
  1382→        className="border-2 border-dashed border-gray-300 rounded-2xl p-8 text-center cursor-pointer hover:border-emerald-400 hover:bg-emerald-50/30 transition-colors"
  1383→      >
  1384→        <input ref={fileInputRef} type="file" accept="image/*" className="hidden" onChange={(e) => e.target.files?.[0] && handleFileSelect(e.target.files[0])} />
  1385→        {imagePreview ? (
  1386→          <div className="space-y-3">
  1387→            <img src={imagePreview} alt="Preview" className="max-h-64 mx-auto rounded-xl object-contain" />
  1388→            <p className="text-sm text-gray-500">Tap to change image</p>
  1389→          </div>
  1390→        ) : (
  1391→          <div className="space-y-3">
  1392→            <div className="w-16 h-16 bg-emerald-50 rounded-2xl flex items-center justify-center mx-auto">
  1393→              <Camera className="h-8 w-8 text-emerald-400" />
  1394→            </div>
  1395→            <p className="text-gray-600 font-medium">Drag & drop or tap to upload</p>
  1396→            <p className="text-sm text-gray-400">Supports JPG, PNG, WebP</p>
  1397→          </div>
  1398→        )}
  1399→      </div>
  1400→
  1401→      {imageFile && !results.length && (
  1402→        <Button className="w-full bg-emerald-600 hover:bg-emerald-700 text-white min-h-[44px] rounded-xl font-semibold" onClick={handleRecognize} disabled={recognizing}>
  1403→          {recognizing ? <><Loader2 className="mr-2 h-4 w-4 animate-spin" />Analyzing...</> : 'Recognize Food'}
  1404→        </Button>
  1405→      )}
  1406→
  1407→      {results.length > 0 && (
  1408→        <div className="space-y-4">
  1409→          <h2 className="font-semibold text-gray-900">Recognized Foods</h2>

  1412→              <div className="flex items-start justify-between gap-2 mb-2">
  1413→                <div>
  1414→                  <p className="font-medium text-gray-900">{food.name}</p>
  1415→                  <p className="text-xs text-gray-500">{food.servingDescription}</p>
