   162→

   167→        method: 'POST',
   168→        body: JSON.stringify({ mealId: logDialog.meal.meal.id, servingGms, mealSlot: logDialog.slot }),
   169→      });

   171→      setLogDialog({ open: false });

   183→      try {
   184→        const res = await apiFetch(`/api/meals/search?q=${encodeURIComponent(query)}`);
   185→        setSearchResults((res.meals || []) as SearchMeal[]);
   186→      } catch { toast.error('Search failed'); }
   187→      finally { setSearchLoading(false); }

   189→    setSearchDebounce(timer);
   190→  };
   191→
   192→  const handleLogFromSearch = (meal: SearchMeal) => {
   193→    const slot = slotSearchDialog.open ? slotSearchDialog.slot : 'lunch';
   194→    setServingGms(meal.baseServingGms || 100);
   195→    setLogDialog({
   196→      open: true,
   197→      meal: {
   198→        meal: { id: meal.id, name: meal.name, mealType: meal.mealType, cuisine: meal.cuisine, imageUrl: null, prepTimeMin: meal.prepTimeMin, isVeg: meal.isVeg, isVegan: meal.isVegan },
   199→        score: 0, recommendedServingGms: meal.baseServingGms || 100, estimatedNutrition: meal.nutrition ? { calories: meal.nutrition.calories, proteinG: meal.nutrition.proteinG, carbsG: meal.nutrition.carbsG, fatG: meal.nutrition.fatG } : null,
   200→        baseNutritionPer100g: meal.nutrition ? { calories: meal.nutrition.calories, proteinG: meal.nutrition.proteinG, carbsG: meal.nutrition.carbsG, fatG: meal.nutrition.fatG } : null,
   201→      },
   202→      slot,
   203→      baseNutritionPer100g: meal.nutrition || null,
   204→      mealName: meal.name,
   205→      cuisine: meal.cuisine,
   206→      mealType: meal.mealType,
   207→      isVeg: meal.isVeg,
   208→      isVegan: meal.isVegan,
   209→    });
   210→    setSlotSearchDialog({ open: false });
   211→    setSearchQuery('');
   212→    setSearchResults([]);
   213→  };
   214→
   215→  const handleWaterAdd = async () => {
   216→    try {
   217→      const result = await apiFetch('/api/water-log', { method: 'POST', body: JSON.stringify({ glasses: 1 }) });
   218→      setWaterCount(result.glassesConsumed || waterCount + 1);
   219→      toast.success('\uD83D\uDCA7 +1 glass of water');
   220→    } catch { toast.error('Failed to log water'); }
   221→  };
   222→
   223→  // Generate meal plan (Feature 2)
   224→  const handleGeneratePlan = async () => {
   225→    setPlanLoading(true);
   226→    try {
   227→      const result = await apiFetch('/api/meal-plan/generate');
   228→      toast.success(result.status === 'existing' ? 'Plan already exists for today!' : "Today's meal plan generated!");

   230→      const plan = await apiFetch('/api/meal-plan');
   231→      setMealPlan({ exists: plan.exists, items: plan.items || [] });
   232→    } catch (err) { toast.error((err as Error).message); }
   233→    finally { setPlanLoading(false); }
   234→  };
   235→
   236→  const firstName = (user?.profile as Record<string, unknown> | null)?.firstName || (user?.name as string) || '';
   237→  const initial = firstName ? firstName.charAt(0).toUpperCase() : '';
   238→  const displayName = firstName || 'there';
   239→  const todayStr = format(new Date(), 'EEEE, MMMM d');
   240→
   241→  // Insights logic (Feature 3)
   242→  const targetCal = nutrition?.targets?.calories || 0;
   243→  const consumedCal = nutrition?.consumed?.calories || 0;
   244→  const pctConsumed = targetCal > 0 ? consumedCal / targetCal : 0;
   245→  const targetProtein = nutrition?.targets?.proteinG || 0;
   246→  const consumedProtein = nutrition?.consumed?.proteinG || 0;
   247→  const pctProtein = targetProtein > 0 ? consumedProtein / targetProtein : 0;
   248→  const hour = new Date().getHours();
   249→  const timeOfDayTip = hour < 12
   250→    ? 'Start your day with a protein-rich breakfast!'
   251→    : hour < 17
   252→      ? 'Keep up the good work! Stay consistent with your meals.'
   253→      : 'Plan a balanced dinner to finish strong today.';
   254→

   256→    return (
   257→      <div className="p-4 max-w-lg mx-auto space-y-4">
   258→        <Skeleton className="h-8 w-56 rounded-xl" />
   259→        <div className="flex justify-center"><Skeleton className="h-[180px] w-[180px] rounded-full" /></div>

   271→        <div className="absolute -bottom-6 -left-6 w-20 h-20 bg-teal-200/20 rounded-full blur-2xl" />
   272→        <div className="absolute top-1/2 right-1/4 w-12 h-12 bg-green-200/10 rounded-full blur-xl" />

   286→            )}
   287→            {initial ? (

   289→                <span className="text-white font-bold text-sm">{initial}</span>
   290→              </div>
   291→            ) : (
   292→              <div className="w-10 h-10 bg-emerald-100 dark:bg-emerald-900/40 rounded-xl flex items-center justify-center shadow-md">
   293→                <User className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   294→              </div>
   295→            )}
   296→          </div>
   297→        </div>
   298→      </div>
   299→

   303→          <p className="text-xl font-bold text-gray-900 dark:text-gray-100 tabular-nums">{mealsLoggedToday}</p>
   304→          <p className="text-[11px] text-gray-600 dark:text-gray-400 font-semibold">Meals Logged</p>

   313→        </div>
   314→      </div>
   315→
   316→      {/* ═══ Calorie Ring Card ═══ */}

   318→        <div className="flex justify-center">
   319→          <CalorieRing consumed={consumedCal} target={targetCal} />
   320→        </div>
   321→        {pctConsumed >= 1 && (
   322→          <motion.p
   323→            initial={{ scale: 0.8, opacity: 0 }}
   324→            animate={{ scale: 1, opacity: 1 }}
   325→            transition={{ type: 'spring', stiffness: 200 }}
   326→            className="text-center text-sm text-emerald-600 dark:text-emerald-400 font-bold mt-3"
   327→          >
   328→            You&apos;ve reached your calorie goal! Great job! 🎉
   329→          </motion.p>
   330→        )}
   331→      </Card>
   332→
   333→      {/* ═══ Macro Progress Bars ═══ */}
   334→      <Card className="p-4 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 space-y-3">
   335→        <h3 className="text-sm font-bold text-gray-800 dark:text-gray-200">Today&apos;s Macros</h3>
   336→        {[
   337→          { label: 'Protein', val: consumedProtein, target: targetProtein, from: '#3b82f6', to: '#60a5fa', unit: 'g' },
   338→          { label: 'Carbs', val: nutrition?.consumed?.carbsG || 0, target: nutrition?.targets?.carbsG || 0, from: '#f59e0b', to: '#fbbf24', unit: 'g' },
   339→          { label: 'Fat', val: nutrition?.consumed?.fatG || 0, target: nutrition?.targets?.fatG || 0, from: '#f43f5e', to: '#fb7185', unit: 'g' },
   340→        ].map((m) => {
   341→          const pct = Math.min(Math.round((m.val / (m.target || 1)) * 100), 100);
   342→          return (
   343→            <div key={m.label}>
   344→              <div className="flex justify-between mb-1.5">
   345→                <span className="text-sm text-gray-600 dark:text-gray-400 font-medium">{m.label}</span>
   346→                <span className="text-xs text-gray-600 dark:text-gray-400 font-medium text-right tabular-nums">{m.val}{m.unit} / {m.target}{m.unit} &middot; {pct}%</span>
   347→              </div>
   348→              <div className="h-3 bg-gray-100 dark:bg-gray-800 rounded-full overflow-hidden">
   349→                <motion.div
   350→                  className="h-full rounded-full"
   351→                  style={{ background: `linear-gradient(90deg, ${m.from}, ${m.to})` }}
   352→                  initial={{ width: 0 }}
   353→                  animate={{ width: `${pct}%` }}
   354→                  transition={{ duration: 0.8 }}
   355→                />
   356→              </div>
   357→            </div>
   358→          );
   359→        })}
   360→      </Card>
   361→
   362→      {/* ═══ Hydration Widget ═══ */}
   363→      <Card className="p-4 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60">
   364→        <div className="flex items-center justify-between mb-3">
   365→          <div className="flex items-center gap-2">
   366→            <Droplets className="h-4 w-4 text-cyan-500" />
   367→            <h3 className="text-sm font-bold text-gray-800 dark:text-gray-200">Hydration</h3>
   368→          </div>
   369→          <span className="text-xs text-gray-500 dark:text-gray-400">Goal: 8 glasses</span>
   370→        </div>
   371→        <div className="flex items-center gap-3">
   372→          <div className="flex-1 flex gap-1.5">
   373→            {[1, 2, 3, 4, 5, 6, 7, 8].map((g) => (
   374→              <div
   375→                key={g}
   376→                className={`h-6 flex-1 rounded-lg transition-all duration-500 ${g <= waterCount ? 'bg-gradient-to-t from-cyan-500 to-cyan-300 dark:from-cyan-600 dark:to-cyan-400' : 'bg-gray-100 dark:bg-gray-800'}`}
   377→              />
   378→            ))}
   379→          </div>
   380→          <Button
   381→            size="icon"
   382→            className="w-9 h-9 rounded-xl bg-cyan-500 hover:bg-cyan-600 text-white shadow-sm"
   383→            onClick={handleWaterAdd}

   385→            <Plus className="h-4 w-4" />
   386→          </Button>
   387→        </div>
   388→        <p className="text-xs text-gray-500 dark:text-gray-400 mt-2">{waterCount} of 8 glasses today</p>
   389→      </Card>
   390→
   391→      {/* ═══ Feature 3: Today's Insights Card ═══ */}
   392→      <motion.div
   393→        initial={{ opacity: 0, y: 8 }}
   394→        animate={{ opacity: 1, y: 0 }}
   395→        transition={{ delay: 0.3 }}
   396→      >
   397→        <Card className={`p-4 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 ${pctConsumed > 1 ? 'bg-amber-50/50 dark:bg-amber-900/10' : pctConsumed >= 0.75 ? 'bg-emerald-50/50 dark:bg-emerald-900/10' : 'bg-blue-50/50 dark:bg-blue-900/10'}`}>
   398→          <div className="flex items-start gap-3">
   399→            <div className={`w-9 h-9 rounded-xl flex items-center justify-center shrink-0 ${pctConsumed > 1 ? 'bg-amber-100/80 dark:bg-amber-900/30' : pctConsumed >= 0.75 ? 'bg-emerald-100/80 dark:bg-emerald-900/30' : 'bg-blue-100/80 dark:bg-blue-900/30'}`}>
   400→              <Lightbulb className={`h-4.5 w-4.5 ${pctConsumed > 1 ? 'text-amber-500' : pctConsumed >= 0.75 ? 'text-emerald-500' : 'text-blue-500'}`} />
   401→            </div>
   402→            <div className="flex-1 min-w-0 space-y-2">
   403→              <h3 className="text-sm font-semibold text-gray-900 dark:text-gray-100">Today&apos;s Insights</h3>

   405→              {/* Calorie status message */}
   406→              {pctConsumed > 1 && (

   416→              {pctConsumed < 0.75 && pctConsumed > 0 && (
   417→                <p className="text-xs text-blue-700 dark:text-blue-400 leading-relaxed">
   418→                  You still have <span className="font-bold">{Math.round(targetCal - consumedCal)} kcal</span> remaining. Time to fuel up!

   421→              {consumedCal === 0 && (
   422→                <p className="text-xs text-gray-500 dark:text-gray-400 leading-relaxed">
   423→                  No meals logged yet today. Start with a balanced breakfast!

   787→                      </div>
   788→                      <div className="flex items-center gap-2 text-xs text-gray-500 dark:text-gray-400 mt-0.5">
   789→                        <span>{meal.nutrition?.calories || 0} kcal/100g</span>
   790→                        <span>&middot;</span>
   791→                        <span className="text-blue-600 font-medium">P: {meal.nutrition?.proteinG || 0}g</span>
   792→                      </div>
   793→                    </div>
   794→                    <ChevronRight className="h-4 w-4 text-gray-300 dark:text-gray-600 shrink-0" />
   795→                  </div>
   796→                ))}
   797→              </div>
   798→            </ScrollArea>
   799→          </div>
   800→        </DialogContent>
   801→      </Dialog>
   802→
   803→      {/* ═══ Meal Detail Sheet ═══ */}
   804→      <Sheet open={detailSheet.open} onOpenChange={(open) => { if (!open) setDetailSheet({ open: false }); }}>
   805→        <SheetContent side="bottom" className="max-w-lg mx-auto rounded-t-2xl max-h-[85vh]">
   806→          {detailSheet.open && (
   807→            <>
   808→              <SheetHeader className="pt-2">
   809→                <div className="flex items-center gap-2 pr-8">
   810→                  {detailSheet.rec.meal.isVeg && <span className="text-green-600 text-xs font-bold border border-green-300 rounded px-1.5 py-0.5">Veg</span>}
   811→                  {detailSheet.rec.meal.isVegan && <span className="text-green-700 text-xs font-bold border border-green-400 rounded px-1.5 py-0.5">Vegan</span>}
   812→                </div>
   813→                <SheetTitle className="text-lg">{detailSheet.rec.meal.name}</SheetTitle>
   814→                <SheetDescription className="flex items-center gap-2 flex-wrap">
   815→                  <Badge variant="outline" className="text-xs">{detailSheet.rec.meal.cuisine}</Badge>
   816→                  <Badge variant="outline" className={`text-xs ${SLOT_BADGE_COLORS[detailSheet.rec.meal.mealType] || ''}`}>{detailSheet.rec.meal.mealType}</Badge>
   817→                  {detailSheet.rec.meal.prepTimeMin && (
   818→                    <span className="text-xs text-gray-500 dark:text-gray-400 flex items-center gap-1"><Clock className="h-3 w-3" />{detailSheet.rec.meal.prepTimeMin} min</span>
   819→                  )}
   820→                </SheetDescription>
   821→              </SheetHeader>
   822→              <ScrollArea className="flex-1 px-4 pb-4">
   823→                <div className="space-y-4">
   824→                  {detailSheet.rec.meal.description && (
   825→                    <p className="text-sm text-gray-600 dark:text-gray-400 leading-relaxed">{detailSheet.rec.meal.description}</p>
   826→                  )}
   827→                  {detailSheet.rec.baseNutritionPer100g && (
   828→                    <div>
   829→                      <h4 className="text-xs font-semibold text-gray-500 dark:text-gray-400 uppercase tracking-wide mb-2">Nutrition per 100g</h4>
   830→                      <NutritionFactsLabel nutrition={detailSheet.rec.baseNutritionPer100g} servingGms={100} label="Per 100g" />
   831→                    </div>
   832→                  )}
   833→                  {detailSheet.rec.estimatedNutrition && (
   834→                    <div className="bg-emerald-50 dark:bg-emerald-900/20 rounded-xl p-3 border border-emerald-100 dark:border-emerald-800">
   835→                      <h4 className="text-xs font-semibold text-emerald-700 dark:text-emerald-400 uppercase tracking-wide mb-2">Recommended: {detailSheet.rec.recommendedServingGms}g serving</h4>
   836→                      <div className="grid grid-cols-2 gap-1.5 text-xs">
   837→                        <span className="text-gray-600 dark:text-gray-400">Calories:</span><span className="font-bold">{detailSheet.rec.estimatedNutrition.calories} kcal</span>
   838→                        <span className="text-gray-600 dark:text-gray-400">Protein:</span><span className="font-bold text-blue-600">{detailSheet.rec.estimatedNutrition.proteinG}g</span>
   839→                        <span className="text-gray-600 dark:text-gray-400">Carbs:</span><span className="font-bold text-amber-600">{detailSheet.rec.estimatedNutrition.carbsG}g</span>
   840→                        <span className="text-gray-600 dark:text-gray-400">Fat:</span><span className="font-bold text-rose-600">{detailSheet.rec.estimatedNutrition.fatG}g</span>
   841→                      </div>
   842→                    </div>
   843→                  )}
   844→                </div>
   845→              </ScrollArea>
   846→              <SheetFooter className="border-t pt-3">
   847→                <Button
   848→                  className="w-full bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl font-semibold min-h-[44px]"
   849→                  onClick={() => {
   850→                    const r = detailSheet.rec;
   851→                    setServingGms(r.recommendedServingGms);
   852→                    setLogDialog({
   853→                      open: true, meal: r, slot: r.meal.mealType,
   854→                      baseNutritionPer100g: r.baseNutritionPer100g,
   855→                      mealName: r.meal.name, cuisine: r.meal.cuisine,
   856→                      mealType: r.meal.mealType, description: r.meal.description,
   857→                      isVeg: r.meal.isVeg, isVegan: r.meal.isVegan,
   858→                    });
   859→                    setDetailSheet({ open: false });
   860→                  }}
   861→                >
   862→                  Log This Meal
   863→                </Button>
   864→              </SheetFooter>
   865→            </>
   866→          )}
   867→        </SheetContent>
   868→      </Sheet>
   869→    </motion.div>
   870→  );
   871→}
