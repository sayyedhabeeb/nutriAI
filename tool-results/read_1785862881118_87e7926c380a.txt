     1→'use client';
     2→
     3→import React, { useState, useEffect, useCallback } from 'react';
     4→import { format, parseISO } from 'date-fns';
     5→import { motion, AnimatePresence } from 'framer-motion';
     6→import { toast } from 'sonner';
     7→import {
     8→  BarChart, Bar, AreaChart, Area, PieChart, Pie, Cell,
     9→  XAxis, YAxis, Tooltip, ResponsiveContainer, ReferenceLine,
    10→} from 'recharts';
    11→import { Button } from '@/components/ui/button';
    12→import { Card } from '@/components/ui/card';
    13→import { Input } from '@/components/ui/input';
    14→import { Skeleton } from '@/components/ui/skeleton';
    15→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
    16→import {
    17→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogClose,
    18→} from '@/components/ui/dialog';
    19→import { Label } from '@/components/ui/label';
    20→import {
    21→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    22→} from '@/components/ui/select';
    23→import {
    24→  TrendingUp, Flame, Dumbbell, Scale, Droplets, Plus, Minus,
    25→  BarChart3, Download, Lightbulb, Trophy, Lock, CalendarDays,
    26→} from 'lucide-react';
    27→import { apiFetch } from './api';
    28→import { PIE_COLORS, fadeIn } from './constants';
    29→
    30→export function ProgressView() {
    31→  const [tab, setTab] = useState('weekly');
    32→  const [weeklyData, setWeeklyData] = useState<Record<string, unknown>[]>([]);
    33→  const [summary, setSummary] = useState<Record<string, unknown> | null>(null);
    34→  const [weightLogs, setWeightLogs] = useState<Record<string, unknown>[]>([]);
    35→  const [waterGlasses, setWaterGlasses] = useState(0);
    36→  const [loading, setLoading] = useState(true);
    37→  const [weightInput, setWeightInput] = useState('');
    38→  const [weightNotes, setWeightNotes] = useState('');
    39→  const [justFilledIndex, setJustFilledIndex] = useState<number | null>(null);
    40→
    41→  // Achievements state
    42→  const [achievements, setAchievements] = useState<{ id: string; name: string; description: string; icon: string; earned: boolean; earnedDate?: string }[]>([]);
    43→
    44→  // Export dialog state
    45→  const [exportOpen, setExportOpen] = useState(false);
    46→  const [exportFormat, setExportFormat] = useState('csv');
    47→  const [exportDays, setExportDays] = useState('7');
    48→  const [exporting, setExporting] = useState(false);
    49→
    50→  const fetchProgress = useCallback(async () => {
    51→    setLoading(true);
    52→    try {
    53→      const [weekData, sumData, wLogs, waterData] = await Promise.all([
    54→        apiFetch('/api/progress/weekly'),
    55→        apiFetch(`/api/progress/summary?period=${tab === 'monthly' ? 'month' : 'week'}`),
    56→        apiFetch('/api/weight-log?limit=30'),
    57→        apiFetch('/api/water-log'),
    58→      ]);
    59→      setWeeklyData(weekData || []);
    60→      setSummary(sumData?.summary || null);
    61→      setWeightLogs(wLogs || []);
    62→      setWaterGlasses(waterData?.glassesConsumed || 0);
    63→
    64→      // Fetch achievements
    65→      try {
    66→        const achData = await apiFetch('/api/achievements');
    67→        setAchievements(achData.achievements || []);
    68→      } catch {
    69→        // Silent fail
    70→      }
    71→    } catch { toast.error('Failed to load progress'); }
    72→    finally { setLoading(false); }
    73→  }, [tab]);
    74→
    75→  useEffect(() => { fetchProgress(); }, [fetchProgress]);
    76→
    77→  const handleAddWater = async (delta: number) => {
    78→    const newCount = waterGlasses + delta;
    79→    if (newCount < 0) return;
    80→    if (delta > 0) {
    81→      setJustFilledIndex(waterGlasses);
    82→      setTimeout(() => setJustFilledIndex(null), 600);
    83→      try {
    84→        await apiFetch('/api/water-log', { method: 'POST', body: JSON.stringify({ glasses: delta }) });
    85→        setWaterGlasses(newCount);
    86→        toast.success(`\uD83D\uDCA7 +${delta} glass${delta > 1 ? 'es' : ''} of water`);
    87→      } catch { toast.error('Failed to log water'); }
    88→    } else {
    89→      setWaterGlasses(Math.max(0, newCount));
    90→    }
    91→  };
    92→
    93→  const handleWaterAdd = async () => {
    94→    setJustFilledIndex(waterGlasses);
    95→    setTimeout(() => setJustFilledIndex(null), 600);
    96→    try {
    97→      await apiFetch('/api/water-log', { method: 'POST', body: JSON.stringify({ glasses: 1 }) });
    98→      setWaterGlasses(waterGlasses + 1);
    99→      toast.success('\uD83D\uDCA7 +1 glass of water');
   100→    } catch { toast.error('Failed to log water'); }
   101→  };
   102→
   103→  const handleLogWeight = async () => {
   104→    if (!weightInput || Number(weightInput) <= 0) return;
   105→    try {
   106→      await apiFetch('/api/weight-log', { method: 'POST', body: JSON.stringify({ weightKg: Number(weightInput), notes: weightNotes || null }) });
   107→      toast.success('Weight logged!');
   108→      setWeightInput('');
   109→      setWeightNotes('');
   110→      fetchProgress();
   111→    } catch (err) { toast.error((err as Error).message); }
   112→  };
   113→
   114→  const handleExport = async () => {
   115→    setExporting(true);
   116→    try {
   117→      const token = localStorage.getItem('nutriai_session');
   118→      const url = `/api/export?format=${exportFormat}&days=${exportDays}`;
   119→      const res = await fetch(url, {
   120→        headers: { Authorization: `Bearer ${token}` },
   121→      });
   122→      if (!res.ok) throw new Error('Export failed');
   123→      const blob = await res.blob();
   124→      const ext = exportFormat === 'csv' ? 'csv' : 'json';
   125→      const contentType = exportFormat === 'csv' ? 'text/csv' : 'application/json';
   126→      const blobUrl = URL.createObjectURL(new Blob([blob], { type: contentType }));
   127→      const a = document.createElement('a');
   128→      a.href = blobUrl;
   129→      a.download = `nutriai-export-${exportDays}days.${ext}`;
   130→      document.body.appendChild(a);
   131→      a.click();
   132→      document.body.removeChild(a);
   133→      URL.revokeObjectURL(blobUrl);
   134→      toast.success('Data exported successfully!');
   135→      setExportOpen(false);
   136→    } catch {
   137→      toast.error('Failed to export data');
   138→    } finally {
   139→      setExporting(false);
   140→    }
   141→  };
   142→
   143→  const chartData = weeklyData.map((d) => {
   144→    const consumed = d.consumed as Record<string, number>;
   145→    const targets = d.targets as Record<string, number> | null;
   146→    return {
   147→      date: format(parseISO(d.date as string), 'EEE'),
   148→      calories: consumed?.calories || 0,
   149→      target: targets?.calories || 0,
   150→    };
   151→  });
   152→
   153→  const allCaloriesZero = chartData.every((d) => d.calories === 0 && d.target === 0);
   154→
   155→  const nonZeroDays = weeklyData.filter((d) => (d.consumed as Record<string, number>)?.calories > 0).length;
   156→  const isSparse = weeklyData.length > 0 && nonZeroDays <= 1;
   157→
   158→  const macroData = [
   159→    { name: 'Protein', value: summary?.avgProtein || 0, color: PIE_COLORS[0] },
   160→    { name: 'Carbs', value: summary?.avgCarbs || 0, color: PIE_COLORS[1] },
   161→    { name: 'Fat', value: summary?.avgFat || 0, color: PIE_COLORS[2] },
   162→  ];
   163→
   164→  const weightChartData = weightLogs.slice().reverse().slice(-7).map((w) => ({
   165→    date: format(parseISO(w.logDate as string), 'MMM d'),
   166→    weight: w.weightKg as number,
   167→  }));
   168→
   169→  const noWeightData = weightChartData.length < 2;
   170→
   171→  // Calorie breakdown table data - filter out zero-eaten days
   172→  const breakdownData = weeklyData
   173→    .map((d) => {
   174→      const consumed = d.consumed as Record<string, number>;
   175→      const targets = d.targets as Record<string, number> | null;
   176→      const consumedCal = consumed?.calories || 0;
   177→      const targetCal = targets?.calories || 0;
   178→      const diff = consumedCal - targetCal;
   179→      const pct = targetCal > 0 ? Math.round((consumedCal / targetCal) * 100) : 0;
   180→      return {
   181→        date: d.date as string,
   182→        consumedCal,
   183→        targetCal,
   184→        diff,
   185→        pct: Math.min(pct, 150),
   186→      };
   187→    })
   188→    .filter((d) => d.consumedCal > 0);
   189→
   190→  const weeklyTotalConsumed = breakdownData.reduce((sum, d) => sum + d.consumedCal, 0);
   191→  const weeklyTotalTarget = breakdownData.reduce((sum, d) => sum + d.targetCal, 0);
   192→  const weeklyTotalDiff = weeklyTotalConsumed - weeklyTotalTarget;
   193→
   194→  const statCards = [
   195→    { label: 'Avg Daily Calories', value: `${Math.round(summary?.avgCalories || 0)}`, icon: Flame, iconBg: 'bg-orange-100 dark:bg-orange-900/30', iconColor: 'text-orange-600 dark:text-orange-400' },
   196→    { label: 'Avg Protein', value: `${Math.round(summary?.avgProtein || 0)}g`, icon: Dumbbell, iconBg: 'bg-blue-100 dark:bg-blue-900/30', iconColor: 'text-blue-600 dark:text-blue-400' },
   197→    { label: 'Total Days', value: `${summary?.totalDays || 0}`, icon: CalendarDays, iconBg: 'bg-emerald-100 dark:bg-emerald-900/30', iconColor: 'text-emerald-600 dark:text-emerald-400' },
   198→    { label: 'Current Weight', value: summary?.currentWeight ? `${summary.currentWeight}kg` : 'N/A', icon: Scale, iconBg: 'bg-purple-100 dark:bg-purple-900/30', iconColor: 'text-purple-600 dark:text-purple-400' },
   199→  ];
   200→
   201→  if (loading) {
   202→    return (
   203→      <div className="p-4 max-w-lg mx-auto space-y-4">
   204→        <Skeleton className="h-8 w-32 rounded-xl" />
   205→        <Skeleton className="h-64 w-full rounded-2xl" />
   206→        <div className="grid grid-cols-2 gap-4"><Skeleton className="h-28 w-full rounded-2xl" /><Skeleton className="h-28 w-full rounded-2xl" /></div>
   207→      </div>
   208→    );
   209→  }
   210→
   211→  return (
   212→    <motion.div {...fadeIn} className="p-4 max-w-lg mx-auto space-y-5">
   213→      {/* Header */}
   214→      <div className="flex items-center justify-between">
   215→        <div className="flex items-center gap-2">
   216→          <div className="w-9 h-9 rounded-xl bg-emerald-100 flex items-center justify-center">
   217→            <TrendingUp className="h-5 w-5 text-emerald-600" />
   218→          </div>
   219→          <h1 className="text-xl font-bold text-gray-900">Progress</h1>
   220→        </div>
   221→        <button
   222→          onClick={() => setExportOpen(true)}
   223→          className="border border-gray-200 dark:border-gray-700 bg-white dark:bg-gray-800 shadow-sm hover:shadow-md rounded-xl px-3.5 py-2 text-xs font-medium text-gray-600 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-700 transition-all flex items-center gap-2"
   224→        >
   225→          <Download className="h-4 w-4" />
   226→          <span className="hidden sm:inline">Export</span>
   227→        </button>
   228→      </div>
   229→
   230→      {/* Export Dialog */}
   231→      <Dialog open={exportOpen} onOpenChange={setExportOpen}>
   232→        <DialogContent className="sm:max-w-sm rounded-2xl">
   233→          <DialogHeader>
   234→            <DialogTitle>Export Data</DialogTitle>
   235→          </DialogHeader>
   236→          <div className="space-y-4 py-2">
   237→            <div className="space-y-1.5">
   238→              <Label className="text-sm font-medium">Format</Label>
   239→              <Select value={exportFormat} onValueChange={setExportFormat}>
   240→                <SelectTrigger className="h-11 rounded-xl"><SelectValue /></SelectTrigger>
   241→                <SelectContent>
   242→                  <SelectItem value="csv">CSV Spreadsheet</SelectItem>
   243→                  <SelectItem value="json">JSON</SelectItem>
   244→                </SelectContent>
   245→              </Select>
   246→            </div>
   247→            <div className="space-y-1.5">
   248→              <Label className="text-sm font-medium">Period</Label>
   249→              <Select value={exportDays} onValueChange={setExportDays}>
   250→                <SelectTrigger className="h-11 rounded-xl"><SelectValue /></SelectTrigger>
   251→                <SelectContent>
   252→                  <SelectItem value="7">Last 7 days</SelectItem>
   253→                  <SelectItem value="14">Last 14 days</SelectItem>
   254→                  <SelectItem value="30">Last 30 days</SelectItem>
   255→                </SelectContent>
   256→              </Select>
   257→            </div>
   258→          </div>
   259→          <DialogFooter className="gap-2 sm:gap-0">
   260→            <DialogClose asChild>
   261→              <Button variant="outline" className="rounded-xl">Cancel</Button>
   262→            </DialogClose>
   263→            <Button className="bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl" onClick={handleExport} disabled={exporting}>
   264→              {exporting ? 'Exporting...' : 'Download'}
   265→            </Button>
   266→          </DialogFooter>
   267→        </DialogContent>
   268→      </Dialog>
   269→
   270→      <Tabs value={tab} onValueChange={setTab}>
   271→        <TabsList className="w-full rounded-xl">
   272→          <TabsTrigger value="weekly" className="flex-1 rounded-lg data-[state=active]:shadow-sm">Weekly</TabsTrigger>
   273→          <TabsTrigger value="monthly" className="flex-1 rounded-lg text-gray-500 dark:text-gray-400 data-[state=active]:text-foreground data-[state=active]:shadow-sm">Monthly</TabsTrigger>
   274→        </TabsList>
   275→      </Tabs>
   276→
   277→      {/* Calorie Chart */}
   278→      <Card className="p-5 rounded-2xl shadow-lg shadow-gray-200/50 dark:shadow-black/20 border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   279→        <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100 mb-3">Calorie Intake</h3>
   280→        <div className="h-48 relative">
   281→          {isSparse && !allCaloriesZero && (
   282→            <div className="absolute inset-0 flex items-center justify-center z-10 bg-white/70 dark:bg-gray-900/70 rounded-lg">
   283→              <p className="text-xs text-gray-400 dark:text-gray-500 font-medium">Log meals to see your weekly trends</p>
   284→            </div>
   285→          )}
   286→          {allCaloriesZero ? (
   287→            <div className="h-full flex flex-col items-center justify-center py-8 text-gray-400">
   288→              <BarChart3 className="h-12 w-12 mb-3 text-gray-200" />
   289→              <p className="text-sm font-medium">Start logging meals to see your calorie trends!</p>
   290→            </div>
   291→          ) : (
   292→            <ResponsiveContainer width="100%" height="100%">
   293→              <BarChart data={chartData} barCategoryGap="20%">
   294→                <XAxis dataKey="date" tick={{ fontSize: 11, fill: '#6b7280' }} axisLine={false} tickLine={false} />
   295→                <YAxis tick={{ fontSize: 11, fill: '#6b7280' }} axisLine={false} tickLine={false} />
   296→                <Tooltip
   297→                  contentStyle={{ borderRadius: '12px', border: '1px solid #e5e7eb', boxShadow: '0 4px 6px -1px rgba(0,0,0,0.1)' }}
   298→                />
   299→                <defs>
   300→                  <linearGradient id="barGrad" x1="0" y1="0" x2="0" y2="1">
   301→                    <stop offset="0%" stopColor="#10b981" stopOpacity={1} />
   302→                    <stop offset="100%" stopColor="#34d399" stopOpacity={0.8} />
   303→                  </linearGradient>
   304→                </defs>
   305→                <Bar dataKey="calories" fill="url(#barGrad)" radius={[4, 4, 0, 0]} name="Consumed" />
   306→                <Bar dataKey="target" fill="#e5e7eb" radius={[4, 4, 0, 0]} name="Target" />
   307→                {!allCaloriesZero && chartData[0]?.target > 0 && (
   308→                  <ReferenceLine y={chartData[0].target} stroke="#10b981" strokeDasharray="6 3" strokeOpacity={0.6} />
   309→                )}
   310→              </BarChart>
   311→            </ResponsiveContainer>
   312→          )}
   313→        </div>
   314→      </Card>
   315→
   316→      {/* Macro Pie + Weight Trend */}
   317→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   318→        <Card className="p-5 rounded-2xl shadow-lg shadow-gray-200/50 dark:shadow-black/20 border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   319→          <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100 mb-2">Macro Breakdown</h3>
   320→          <div className="h-44 relative">
   321→            <ResponsiveContainer width="100%" height="100%">
   322→              <PieChart>
   323→                <Pie data={macroData} cx="50%" cy="50%" innerRadius={38} outerRadius={60} dataKey="value" strokeWidth={2}>
   324→                  {macroData.map((entry, i) => <Cell key={i} fill={entry.color} />)}
   325→                </Pie>
   326→                <Tooltip contentStyle={{ borderRadius: '12px', border: '1px solid #e5e7eb' }} />
   327→              </PieChart>
   328→            </ResponsiveContainer>
   329→            <div className="absolute inset-0 flex flex-col items-center justify-center pointer-events-none">
   330→              <span className="text-lg font-bold text-gray-800 dark:text-gray-100 tabular-nums">{Math.round(summary?.avgCalories || 0)}</span>
   331→              <span className="text-[10px] text-gray-400 dark:text-gray-500 font-medium">avg kcal</span>
   332→            </div>
   333→          </div>
   334→          <div className="flex justify-center gap-4 mt-1">
   335→            {macroData.map((m) => (
   336→              <div key={m.name} className="flex items-center gap-1.5">
   337→                <div className="w-2.5 h-2.5 rounded-full" style={{ backgroundColor: m.color }} />
   338→                <span className="text-xs text-gray-600 dark:text-gray-300 font-medium">{m.name} {Math.round(m.value)}g</span>
   339→              </div>
   340→            ))}
   341→          </div>
   342→        </Card>
   343→        <Card className="p-5 rounded-2xl shadow-lg shadow-gray-200/50 dark:shadow-black/20 border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   344→          <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100 mb-2">Weight Trend</h3>
   345→          {noWeightData ? (
   346→            <div className="h-44 flex flex-col items-center justify-center py-8 text-gray-400">
   347→              <Scale className="h-10 w-10 mb-2 text-gray-200" />
   348→              <p className="text-sm font-medium">Log your weight daily to track your progress!</p>
   349→            </div>
   350→          ) : (
   351→            <div className="h-44">
   352→              <ResponsiveContainer width="100%" height="100%">
   353→                <AreaChart data={weightChartData}>
   354→                  <defs>
   355→                    <linearGradient id="weightGrad" x1="0" y1="0" x2="0" y2="1">
   356→                      <stop offset="0%" stopColor="#10b981" stopOpacity={0.3} />
   357→                      <stop offset="100%" stopColor="#10b981" stopOpacity={0.02} />
   358→                    </linearGradient>
   359→                  </defs>
   360→                  <XAxis dataKey="date" tick={{ fontSize: 10, fill: '#6b7280' }} axisLine={false} tickLine={false} />
   361→                  <YAxis domain={['dataMin - 1', 'dataMax + 1']} tick={{ fontSize: 10, fill: '#6b7280' }} axisLine={false} tickLine={false} />
   362→                  <Tooltip contentStyle={{ borderRadius: '12px', border: '1px solid #e5e7eb' }} />
   363→                  <Area type="monotone" dataKey="weight" stroke="#10b981" strokeWidth={2.5} fill="url(#weightGrad)" dot={{ r: 4, fill: '#10b981', strokeWidth: 2, stroke: '#fff' }} activeDot={{ r: 5 }} />
   364→                </AreaChart>
   365→              </ResponsiveContainer>
   366→            </div>
   367→          )}
   368→        </Card>
   369→      </div>
   370→
   371→      {/* Weekly Summary Insight */}
   372→      {summary && !allCaloriesZero && (
   373→        <Card className="p-4 rounded-2xl shadow-lg shadow-gray-200/50 dark:shadow-black/20 border border-gray-100/60 dark:border-gray-800/60 bg-gradient-to-r from-emerald-50/50 to-teal-50/50 dark:from-emerald-900/10 dark:to-teal-900/10">
   374→          <div className="flex items-start gap-3">
   375→            <div className="w-9 h-9 rounded-xl bg-emerald-100/80 dark:bg-emerald-900/30 flex items-center justify-center shrink-0">
   376→              <Lightbulb className="h-[18px] w-[18px] text-emerald-500" />
   377→            </div>
   378→            <div>
   379→              <h3 className="text-sm font-bold text-gray-800 dark:text-gray-200">Weekly Insight</h3>
   380→              <p className="text-xs text-gray-600 dark:text-gray-300 mt-1 leading-relaxed">
   381→                {(() => {
   382→                  const avgCal = Math.round(summary?.avgCalories || 0);
   383→                  const targetCal = chartData[0]?.target || 0;
   384→                  const avgProtein = Math.round(summary?.avgProtein || 0);
   385→                  const targetProtein = summary?.targetProtein ? Math.round(summary.targetProtein as number) : 0;
   386→                  const proteinPct = targetProtein > 0 ? Math.round((avgProtein / targetProtein) * 100) : 0;
   387→                  const bestDay = weeklyData.reduce((best: Record<string, unknown> | null, d) => {
   388→                    const cal = (d.consumed as Record<string, number>)?.calories || 0;
   389→                    const bestCal = best ? (best.consumed as Record<string, number>)?.calories || 0 : 0;
   390→                    return cal > bestCal ? d : best;
   391→                  }, null);
   392→                  const bestDayName = bestDay ? format(parseISO(bestDay.date as string), 'EEEE') : '';
   393→                  const parts = [`You averaged ${avgCal} kcal/day this week.`];
   394→                  if (proteinPct > 0) parts.push(`Protein intake was ${proteinPct}% of target.`);
   395→                  if (targetCal > 0 && avgCal < targetCal) parts.push(`You're ${targetCal - avgCal} kcal under daily target.`);
   396→                  else if (targetCal > 0 && avgCal > targetCal) parts.push(`You're ${avgCal - targetCal} kcal over daily target.`);
   397→                  if (bestDayName) parts.push(`Best day: ${bestDayName}.`);
   398→                  parts.push('Keep it up!');
   399→                  return parts.join(' ');
   400→                })()}
   401→              </p>
   402→            </div>
   403→          </div>
   404→        </Card>
   405→      )}
   406→
   407→      {/* Stats Cards */}
   408→      <div className="grid grid-cols-2 gap-3">
   409→        {statCards.map((s) => (
   410→          <div key={s.label} className="flex items-center gap-2.5 p-3 rounded-xl bg-gray-50/80 dark:bg-gray-800/30">
   411→            <div className={`w-9 h-9 rounded-lg ${s.iconBg} flex items-center justify-center shrink-0`}>
   412→              <s.icon className={`h-4 w-4 ${s.iconColor}`} />
   413→            </div>
   414→            <div>
   415→              <p className="text-[10px] text-gray-500 dark:text-gray-300 font-medium uppercase">{s.label}</p>
   416→              <p className="text-base font-bold text-gray-900 dark:text-gray-100 tabular-nums">{s.value}</p>
   417→            </div>
   418→          </div>
   419→        ))}
   420→      </div>
   421→
   422→      {/* Achievements Grid */}
