     1→'use client';
     2→
     3→import React, { useState, useEffect, useCallback } from 'react';
     4→import { format, subDays, parseISO } from 'date-fns';
     5→import { motion, AnimatePresence } from 'framer-motion';
     6→import { toast } from 'sonner';
     7→import { Button } from '@/components/ui/button';
     8→import { Card } from '@/components/ui/card';
     9→import { Input } from '@/components/ui/input';
    10→import { Label } from '@/components/ui/label';
    11→import { Badge } from '@/components/ui/badge';
    12→import { Skeleton } from '@/components/ui/skeleton';
    13→import { ScrollArea } from '@/components/ui/scroll-area';
    14→import {
    15→  Collapsible, CollapsibleContent, CollapsibleTrigger,
    16→} from '@/components/ui/collapsible';
    17→import {
    18→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    19→} from '@/components/ui/select';
    20→import {
    21→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter,
    22→} from '@/components/ui/dialog';
    23→import { Camera, RotateCcw, Trash2, Plus, Search, ChevronDown, ChevronRight, Flame, Dumbbell, Wheat, Droplets } from 'lucide-react';
    24→import { apiFetch } from './api';
    25→import { SLOTS, SLOT_LABELS, SLOT_ICONS, SLOT_BORDER_COLORS, fadeIn } from './constants';
    26→import type { FoodLogItem, SearchMeal } from './types';
    27→
    28→export function FoodLogView() {
    29→  const [selectedDate, setSelectedDate] = useState(format(new Date(), 'yyyy-MM-dd'));
    30→  const [foodLog, setFoodLog] = useState<Record<string, unknown> | null>(null);
    31→  const [loading, setLoading] = useState(true);
    32→  const [relogDialog, setRelogDialog] = useState<{ open: boolean; item: FoodLogItem } | { open: false }>({ open: false });
    33→  const [relogSlot, setRelogSlot] = useState('lunch');
    34→
    35→  // Feature 1: Quick Add Custom Food
    36→  const [quickAddDialog, setQuickAddDialog] = useState(false);
    37→  const [quickName, setQuickName] = useState('');
    38→  const [quickSlot, setQuickSlot] = useState('lunch');
    39→  const [quickServing, setQuickServing] = useState(100);
    40→  const [quickCalories, setQuickCalories] = useState('');
    41→  const [quickProtein, setQuickProtein] = useState('');
    42→  const [quickCarbs, setQuickCarbs] = useState('');
    43→  const [quickFat, setQuickFat] = useState('');
    44→  const [showMacros, setShowMacros] = useState(false);
    45→  const [quickAdding, setQuickAdding] = useState(false);
    46→
    47→  // Feature 4: Search Meals
    48→  const [searchDialog, setSearchDialog] = useState(false);
    49→  const [searchQuery, setSearchQuery] = useState('');
    50→  const [searchResults, setSearchResults] = useState<SearchMeal[]>([]);
    51→  const [searchLoading, setSearchLoading] = useState(false);
    52→  const [searchDebounce, setSearchDebounce] = useState<ReturnType<typeof setTimeout> | null>(null);
    53→
    54→  // Log meal dialog from search (Feature 4)
    55→  const [logFromSearchDialog, setLogFromSearchDialog] = useState<{
    56→    open: boolean;
    57→    meal: SearchMeal;
    58→  } | { open: false }>({ open: false });
    59→  const [searchLogServing, setSearchLogServing] = useState(100);
    60→
    61→  const dates = Array.from({ length: 7 }, (_, i) => format(subDays(new Date(), 6 - i), 'yyyy-MM-dd'));
    62→
    63→  const fetchLog = useCallback(async () => {
    64→    setLoading(true);
    65→    try {
    66→      const data = await apiFetch(`/api/food-logs?date=${selectedDate}`);
    67→      setFoodLog(data);
    68→    } catch { toast.error('Failed to load food log'); }
    69→    finally { setLoading(false); }
    70→  }, [selectedDate]);
    71→
    72→  useEffect(() => { fetchLog(); }, [fetchLog]);
    73→
    74→  const handleDelete = async (itemId: string) => {
    75→    try {
    76→      await apiFetch(`/api/food-logs?id=${itemId}`, { method: 'DELETE' });
    77→      toast.success('Item removed');
    78→      fetchLog();
    79→    } catch (err) { toast.error((err as Error).message); }
    80→  };
    81→
    82→  const handleRelog = async () => {
    83→    if (!relogDialog.open) return;
    84→    const item = relogDialog.item;
    85→    try {
    86→      if (item.mealId) {
    87→        await apiFetch('/api/food-logs', {
    88→          method: 'POST',
    89→          body: JSON.stringify({ mealId: item.mealId, servingGms: item.servingGms, mealSlot: relogSlot }),
    90→        });
    91→      } else {
    92→        await apiFetch('/api/food-logs', {
    93→          method: 'POST',
    94→          body: JSON.stringify({ customName: item.meal?.name || 'Food', servingGms: item.servingGms, mealSlot: relogSlot, caloriesPer100g: Math.round(item.calories / item.servingGms * 100), proteinPer100g: Math.round(item.proteinG / item.servingGms * 100) }),
    95→        });
    96→      }
    97→      toast.success(`Re-logged ${item.meal?.name || 'meal'}!`);
    98→      setRelogDialog({ open: false });
    99→      fetchLog();
   100→    } catch (err) { toast.error((err as Error).message); }
   101→  };
   102→
   103→  // Feature 1: Handle quick add
   104→  const handleQuickAdd = async () => {
   105→    if (!quickName.trim()) { toast.error('Please enter a food name'); return; }
   106→    if (!quickCalories || Number(quickCalories) <= 0) { toast.error('Please enter calories'); return; }
   107→    setQuickAdding(true);
   108→    try {
   109→      await apiFetch('/api/food-logs/quick', {
   110→        method: 'POST',
   111→        body: JSON.stringify({
   112→          name: quickName.trim(),
   113→          calories: Number(quickCalories),
   114→          proteinG: Number(quickProtein) || 0,
   115→          carbsG: Number(quickCarbs) || 0,
   116→          fatG: Number(quickFat) || 0,
   117→          mealSlot: quickSlot,
   118→          servingGms: quickServing,
   119→        }),
   120→      });
   121→      toast.success(`Logged ${quickName.trim()}!`);
   122→      // Reset form
   123→      setQuickAddDialog(false);
   124→      setQuickName('');
   125→      setQuickCalories('');
   126→      setQuickProtein('');
   127→      setQuickCarbs('');
   128→      setQuickFat('');
   129→      setQuickServing(100);
   130→      setShowMacros(false);
   131→      fetchLog();
   132→    } catch (err) { toast.error((err as Error).message); }
   133→    finally { setQuickAdding(false); }
   134→  };
   135→
   136→  // Feature 4: Search handlers
   137→  const handleSearch = (query: string) => {
   138→    setSearchQuery(query);
   139→    if (searchDebounce) clearTimeout(searchDebounce);
   140→    if (!query.trim()) { setSearchResults([]); return; }
   141→    const timer = setTimeout(async () => {
   142→      setSearchLoading(true);
   143→      try {
   144→        const res = await apiFetch(`/api/meals/search?q=${encodeURIComponent(query)}`);
   145→        setSearchResults((res.meals || []) as SearchMeal[]);
   146→      } catch { toast.error('Search failed'); }
   147→      finally { setSearchLoading(false); }
   148→    }, 350);
   149→    setSearchDebounce(timer);
   150→  };
   151→
   152→  const handleLogFromSearch = async () => {
   153→    if (!logFromSearchDialog.open) return;
   154→    const meal = logFromSearchDialog.meal;
   155→    setQuickSlot(quickSlot);
   156→    try {
   157→      await apiFetch('/api/food-logs', {
   158→        method: 'POST',
   159→        body: JSON.stringify({ mealId: meal.id, servingGms: searchLogServing, mealSlot: quickSlot }),
   160→      });
   161→      toast.success(`Logged ${meal.name}!`);
   162→      setLogFromSearchDialog({ open: false });
   163→      setSearchDialog(false);
   164→      setSearchQuery('');
   165→      setSearchResults([]);
   166→      fetchLog();
   167→    } catch (err) { toast.error((err as Error).message); }
   168→  };
   169→
   170→  const itemsBySlot = (foodLog?.itemsBySlot || {}) as Record<string, FoodLogItem[]>;
   171→  const hasItems = Object.values(itemsBySlot).some((items: FoodLogItem[]) => items.length > 0);
   172→
   173→  return (
   174→    <motion.div {...fadeIn} className="p-4 max-w-lg mx-auto space-y-4">
   175→      <h1 className="text-xl font-bold text-gray-900">Food Log</h1>
   176→
   177→      {/* Feature 4: Search bar */}
   178→      <motion.div
   179→        initial={{ opacity: 0, y: -4 }}
   180→        animate={{ opacity: 1, y: 0 }}
   181→        transition={{ delay: 0.1 }}
   182→      >
   183→        <div className="relative">
   184→          <Search className="absolute left-3.5 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400 shrink-0 pointer-events-none" />
   185→          <button
   186→            onClick={() => { setSearchQuery(''); setSearchResults([]); setSearchDialog(true); }}
   187→            className="w-full flex items-center gap-2 pl-10 pr-4 h-10 rounded-full bg-gray-50 border border-gray-200 hover:bg-gray-100 transition-colors text-left"
   188→          >
   189→            <span className="text-sm text-gray-400">Search meals to log...</span>
   190→          </button>
   191→        </div>
   192→      </motion.div>
   193→
   194→      {/* Date strip */}
   195→      <div className="flex gap-2 overflow-x-auto pb-2 scrollbar-hide">
   196→        {dates.map((date) => {
   197→          const isSelected = date === selectedDate;
   198→          return (
   199→            <button
   200→              key={date}
   201→              onClick={() => setSelectedDate(date)}
   202→              className={`flex flex-col items-center min-w-[52px] py-2 px-2 rounded-xl transition-all shrink-0 min-h-[44px] ${
   203→                isSelected
   204→                  ? 'bg-emerald-600 text-white shadow-sm'
   205→                  : 'text-gray-400 hover:text-gray-600 hover:bg-gray-50'
   206→              }`}
   207→            >
   208→              <span className="text-[10px] font-medium uppercase">{format(parseISO(date), 'EEE')}</span>
   209→              <span className={`text-sm font-bold ${isSelected ? '' : ''}`}>{format(parseISO(date), 'd')}</span>
   210→            </button>
   211→          );
   212→        })}
   213→      </div>
   214→      <div className="h-px bg-gray-200/60 -mt-1" />
   215→
   216→      {/* Summary */}
   217→      <Card className="p-4 rounded-2xl shadow-sm border border-gray-100/80 bg-white">
   218→        <div className="grid grid-cols-4 gap-2 text-center">
   219→          {[
   220→            { label: 'Calories', val: foodLog?.totalCalories || 0, unit: 'kcal', color: 'text-orange-600', iconBg: 'bg-orange-100', Icon: Flame },
   221→            { label: 'Protein', val: foodLog?.totalProtein || 0, unit: 'g', color: 'text-blue-600', iconBg: 'bg-blue-100', Icon: Dumbbell },
   222→            { label: 'Carbs', val: foodLog?.totalCarbs || 0, unit: 'g', color: 'text-amber-600', iconBg: 'bg-amber-100', Icon: Wheat },
   223→            { label: 'Fat', val: foodLog?.totalFat || 0, unit: 'g', color: 'text-rose-600', iconBg: 'bg-rose-100', Icon: Droplets },
   224→          ].map((s) => (
   225→            <div key={s.label} className="flex flex-col items-center gap-1.5">
   226→              <div className={`w-8 h-8 rounded-full ${s.iconBg} flex items-center justify-center`}>
   227→                <s.Icon className={`h-4 w-4 ${s.color}`} />
   228→              </div>
   229→              <div>
   230→                <p className={`text-lg font-bold ${s.color}`}>{Math.round(s.val)}</p>
   231→                <p className="text-xs text-gray-500 font-medium">{s.label}</p>
   232→              </div>
   233→            </div>
   234→          ))}
   235→        </div>
   236→      </Card>
   237→
   238→      {/* Loading */}
   239→      {loading && (
   240→        <div className="space-y-3">
   241→          <Skeleton className="h-20 w-full rounded-xl" />
   242→          <Skeleton className="h-20 w-full rounded-xl" />
   243→        </div>
   244→      )}
   245→
   246→      {/* Empty state */}
   247→      {!loading && !hasItems && (
   248→        <Card className="p-8 text-center rounded-2xl shadow-sm border border-gray-100/80 bg-white">
   249→          <div className="w-16 h-16 bg-gray-100 rounded-2xl flex items-center justify-center mx-auto mb-4">
   250→            <Camera className="h-8 w-8 text-gray-300" />
   251→          </div>
   252→          <p className="text-gray-700 font-semibold text-base">Start logging your meals!</p>
   253→          <p className="text-sm text-gray-400 mt-1">Search for meals or add a custom food</p>
   254→        </Card>
   255→      )}
   256→
   257→      {/* Food items by slot */}
   258→      {!loading && SLOTS.map((slot) => {
   259→        const items = itemsBySlot[slot] || [];
   260→        if (items.length === 0) return null;
   261→        return (
   262→          <Card key={slot} className={`p-0 rounded-2xl shadow-sm border border-gray-100/80 border-l-4 ${SLOT_BORDER_COLORS[slot]} overflow-hidden bg-white`}>
   263→            <div className="p-4 pb-2">
   264→              <div className="flex items-center gap-2">
   265→                <span className="text-base">{SLOT_ICONS[slot]}</span>
   266→                <h3 className="font-semibold text-sm text-gray-900">{SLOT_LABELS[slot]}</h3>
   267→                <Badge variant="secondary" className="text-xs ml-auto bg-gray-100 text-gray-600">{items.length} item{items.length > 1 ? 's' : ''}</Badge>
   268→              </div>
   269→            </div>
   270→            <div className="px-4 pb-3 space-y-3">
   271→              {items.map((item) => (
   272→                <div key={item.id} className="flex items-center justify-between p-3 rounded-xl bg-gray-50/80 border border-gray-100 group hover:shadow-sm hover:bg-gray-50/100 hover:border-gray-200/80 transition-all">
   273→                  <div className="flex-1 min-w-0">
   274→                    <p className="text-sm font-medium text-gray-800 truncate">{item.meal?.name || 'Unknown'}</p>
   275→                    <p className="text-xs text-gray-500 mt-0.5">{item.servingGms}g &middot; {Math.round(item.calories)} kcal &middot; P: {Math.round(item.proteinG * 10) / 10}g</p>
   276→                  </div>
   277→                  <div className="flex items-center gap-0.5 shrink-0 ml-2">
   278→                    <Button
   279→                      variant="ghost" size="icon"
   280→                      className="h-7 w-7 text-gray-400 hover:text-emerald-600 hover:bg-emerald-50 opacity-0 group-hover:opacity-100 transition-opacity"
   281→                      onClick={() => { setRelogDialog({ open: true, item }); setRelogSlot(item.mealSlot); }}
   282→                      title="Log Again"
   283→                    >
   284→                      <RotateCcw className="h-3 w-3" />
   285→                    </Button>
   286→                    <Button
   287→                      variant="ghost" size="icon"
   288→                      className="h-7 w-7 text-gray-300 hover:text-red-500 hover:bg-red-50"
   289→                      onClick={() => handleDelete(item.id)}
   290→                    >
   291→                      <Trash2 className="h-3 w-3" />
   292→                    </Button>
   293→                  </div>
   294→                </div>
   295→              ))}
   296→            </div>
   297→          </Card>
   298→        );
   299→      })}
   300→
   301→      {/* Feature 1: Quick Add FAB (above nav) */}
   302→      <motion.div
   303→        className="fixed bottom-20 right-4 z-40"
   304→        initial={{ scale: 0 }}
   305→        animate={{ scale: 1 }}
   306→        transition={{ type: 'spring', stiffness: 260, damping: 20, delay: 0.2 }}
   307→      >
   308→        <Button
   309→          size="icon"
   310→          className="w-14 h-14 rounded-full bg-gradient-to-br from-emerald-500 to-emerald-600 hover:from-emerald-600 hover:to-emerald-700 text-white shadow-xl hover:shadow-2xl transition-shadow"
   311→          onClick={() => setQuickAddDialog(true)}
   312→        >
   313→          <Plus className="h-6 w-6" />
   314→        </Button>
   315→      </motion.div>
   316→
   317→      {/* ═══ Feature 1: Quick Add Custom Food Dialog ═══ */}
   318→      <Dialog open={quickAddDialog} onOpenChange={setQuickAddDialog}>
   319→        <DialogContent className="max-w-md">
   320→          <DialogHeader>
   321→            <DialogTitle>Quick Add Food</DialogTitle>
   322→            <DialogDescription>Log a custom food with its nutrition info</DialogDescription>
   323→          </DialogHeader>
   324→          <div className="space-y-3">
   325→            <div className="space-y-2">
   326→              <Label className="text-sm font-medium">Food Name *</Label>
   327→              <Input
   328→                placeholder="e.g. Chicken Sandwich"
   329→                value={quickName}
   330→                onChange={(e) => setQuickName(e.target.value)}
   331→                className="h-11 rounded-xl"
   332→                autoFocus
   333→              />
   334→            </div>
   335→            <div className="space-y-2">
   336→              <Label className="text-sm font-medium">Meal Slot</Label>
   337→              <Select value={quickSlot} onValueChange={setQuickSlot}>
   338→                <SelectTrigger className="h-11 rounded-xl"><SelectValue /></SelectTrigger>
   339→                <SelectContent>{SLOTS.map((s) => <SelectItem key={s} value={s}>{SLOT_LABELS[s]}</SelectItem>)}</SelectContent>
   340→              </Select>
   341→            </div>
   342→            <div className="grid grid-cols-2 gap-3">
   343→              <div className="space-y-2">
   344→                <Label className="text-sm font-medium">Serving (g)</Label>
   345→                <Input
   346→                  type="number"
   347→                  value={quickServing}
   348→                  onChange={(e) => setQuickServing(Number(e.target.value) || 100)}
   349→                  min={1} max={5000}
   350→                  className="h-11 rounded-xl"
   351→                />
   352→              </div>
   353→              <div className="space-y-2">
   354→                <Label className="text-sm font-medium">Calories * (kcal)</Label>
   355→                <Input
   356→                  type="number"
   357→                  placeholder="e.g. 350"
   358→                  value={quickCalories}
   359→                  onChange={(e) => setQuickCalories(e.target.value)}
   360→                  min={1}
   361→                  className="h-11 rounded-xl"
   362→                />
   363→              </div>
   364→            </div>
   365→
   366→            {/* Collapsible macros */}
   367→            <Collapsible open={showMacros} onOpenChange={setShowMacros}>
   368→              <CollapsibleTrigger asChild>
   369→                <button className="flex items-center gap-1.5 text-xs text-emerald-600 font-medium hover:text-emerald-700 transition-colors">
   370→                  <ChevronDown className={`h-3.5 w-3.5 transition-transform ${showMacros ? 'rotate-180' : ''}`} />
   371→                  {showMacros ? 'Hide' : 'Add'} protein, carbs & fat
   372→                </button>
   373→              </CollapsibleTrigger>
   374→              <AnimatePresence>
   375→                {showMacros && (
   376→                  <CollapsibleContent forceMount>
   377→                    <motion.div
   378→                      initial={{ height: 0, opacity: 0 }}
   379→                      animate={{ height: 'auto', opacity: 1 }}
   380→                      exit={{ height: 0, opacity: 0 }}
   381→                      transition={{ duration: 0.2 }}
   382→                      className="overflow-hidden"
   383→                    >
   384→                      <div className="grid grid-cols-3 gap-2 pt-2">
   385→                        <div className="space-y-1.5">
   386→                          <Label className="text-xs text-gray-500">Protein (g)</Label>
   387→                          <Input type="number" placeholder="0" value={quickProtein} onChange={(e) => setQuickProtein(e.target.value)} min={0} className="h-10 rounded-lg text-sm" />
   388→                        </div>
   389→                        <div className="space-y-1.5">
   390→                          <Label className="text-xs text-gray-500">Carbs (g)</Label>
   391→                          <Input type="number" placeholder="0" value={quickCarbs} onChange={(e) => setQuickCarbs(e.target.value)} min={0} className="h-10 rounded-lg text-sm" />
   392→                        </div>
   393→                        <div className="space-y-1.5">
   394→                          <Label className="text-xs text-gray-500">Fat (g)</Label>
   395→                          <Input type="number" placeholder="0" value={quickFat} onChange={(e) => setQuickFat(e.target.value)} min={0} className="h-10 rounded-lg text-sm" />
   396→                        </div>
   397→                      </div>
   398→                    </motion.div>
   399→                  </CollapsibleContent>
   400→                )}
   401→              </AnimatePresence>
   402→            </Collapsible>
   403→          </div>
   404→          <DialogFooter>
   405→            <Button variant="outline" onClick={() => setQuickAddDialog(false)} className="rounded-xl">Cancel</Button>
   406→            <Button
   407→              className="bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl font-semibold"
   408→              onClick={handleQuickAdd}
   409→              disabled={quickAdding || !quickName.trim() || !quickCalories}
   410→            >
   411→              {quickAdding ? 'Logging...' : 'Log It'}
   412→            </Button>
   413→          </DialogFooter>
   414→        </DialogContent>
   415→      </Dialog>
   416→
   417→      {/* ═══ Feature 4: Search Meals Dialog ═══ */}
   418→      <Dialog open={searchDialog} onOpenChange={(open) => { if (!open) { setSearchDialog(false); setSearchQuery(''); setSearchResults([]); } }}>
   419→        <DialogContent className="max-w-md">
   420→          <DialogHeader>
   421→            <DialogTitle>Search Meals</DialogTitle>
   422→            <DialogDescription>Find and log a meal</DialogDescription>
   423→          </DialogHeader>
   424→          <div className="space-y-3">
   425→            <div className="relative">
   426→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400" />
   427→              <Input
   428→                placeholder="Search meals..."
   429→                value={searchQuery}
   430→                onChange={(e) => handleSearch(e.target.value)}
   431→                className="h-11 rounded-xl pl-9"
   432→                autoFocus
   433→              />
   434→            </div>
   435→            <ScrollArea className="max-h-72">
   436→              <div className="space-y-2">
   437→                {searchLoading && <div className="flex justify-center py-6"><SpinnerIcon /></div>}
   438→                {!searchLoading && searchQuery && searchResults.length === 0 && (
   439→                  <p className="text-sm text-gray-400 text-center py-6">No results found</p>
   440→                )}
   441→                {!searchLoading && searchResults.map((meal) => (
   442→                  <div
   443→                    key={meal.id}
   444→                    className="flex items-center justify-between p-3 rounded-xl border border-gray-100 hover:bg-emerald-50/50 hover:border-emerald-100 transition-colors cursor-pointer"
   445→                    onClick={() => {
   446→                      setSearchLogServing(meal.baseServingGms || 100);
   447→                      setLogFromSearchDialog({ open: true, meal });
   448→                    }}
   449→                  >
   450→                    <div className="flex-1 min-w-0">
   451→                      <div className="flex items-center gap-2">
   452→                        <p className="text-sm font-medium text-gray-800 truncate">{meal.name}</p>
   453→                        <Badge variant="outline" className="text-[10px] px-1.5 py-0 h-4 font-medium border-gray-200 text-gray-500 shrink-0">{meal.cuisine}</Badge>
   454→                      </div>
   455→                      <div className="flex items-center gap-2 text-xs text-gray-500 mt-0.5">
   456→                        <span>{meal.nutrition?.calories || 0} kcal/100g</span>
   457→                        <span>&middot;</span>
   458→                        <span className="text-blue-600 font-medium">P: {meal.nutrition?.proteinG || 0}g</span>
   459→                      </div>
   460→                    </div>
   461→                    <ChevronRight className="h-4 w-4 text-gray-300 shrink-0" />
   462→                  </div>
   463→                ))}
   464→              </div>
   465→            </ScrollArea>
   466→          </div>
   467→        </DialogContent>
   468→      </Dialog>
   469→
   470→      {/* ═══ Feature 4: Log Meal from Search Dialog ═══ */}
   471→      <Dialog open={logFromSearchDialog.open} onOpenChange={(open) => { if (!open) setLogFromSearchDialog({ open: false }); }}>
   472→        <DialogContent className="max-w-md">
   473→          <DialogHeader>
   474→            <DialogTitle>Log Meal</DialogTitle>
   475→            <DialogDescription className="flex items-center gap-2">
   476→              {logFromSearchDialog.open && <span className="font-medium text-gray-700">{logFromSearchDialog.meal.name}</span>}
   477→              {logFromSearchDialog.open && <Badge variant="outline" className="text-xs">{logFromSearchDialog.meal.cuisine}</Badge>}
   478→            </DialogDescription>
   479→          </DialogHeader>
   480→          <div className="space-y-4">
   481→            <div className="space-y-2">
   482→              <Label className="text-sm font-medium">Serving Size (grams)</Label>
   483→              <Input type="number" value={searchLogServing} onChange={(e) => setSearchLogServing(Number(e.target.value))} min={10} max={1000} className="h-11 rounded-xl" />
   484→            </div>
   485→            {logFromSearchDialog.open && logFromSearchDialog.meal.nutrition && (
   486→              <div className="bg-emerald-50 rounded-xl p-3 text-sm space-y-1 border border-emerald-100">
   487→                <p className="text-xs font-semibold text-emerald-700 uppercase tracking-wide mb-1.5">Estimated for {searchLogServing}g serving</p>
   488→                <div className="grid grid-cols-2 gap-1 text-xs">
   489→                  <span className="text-gray-600">Calories:</span><span className="font-bold text-gray-900">{Math.round((logFromSearchDialog.meal.nutrition!.calories / 100) * searchLogServing)} kcal</span>
   490→                  <span className="text-gray-600">Protein:</span><span className="font-bold text-blue-600">{Math.round((logFromSearchDialog.meal.nutrition!.proteinG / 100) * searchLogServing * 10) / 10}g</span>
   491→                  <span className="text-gray-600">Carbs:</span><span className="font-bold text-amber-600">{Math.round((logFromSearchDialog.meal.nutrition!.carbsG / 100) * searchLogServing * 10) / 10}g</span>
   492→                  <span className="text-gray-600">Fat:</span><span className="font-bold text-rose-600">{Math.round((logFromSearchDialog.meal.nutrition!.fatG / 100) * searchLogServing * 10) / 10}g</span>
   493→                </div>
   494→              </div>
   495→            )}
   496→            <div className="space-y-2">
   497→              <Label className="text-sm font-medium">Meal Slot</Label>
   498→              <Select value={quickSlot} onValueChange={setQuickSlot}>
   499→                <SelectTrigger className="h-11 rounded-xl"><SelectValue /></SelectTrigger>
   500→                <SelectContent>{SLOTS.map((s) => <SelectItem key={s} value={s}>{SLOT_LABELS[s]}</SelectItem>)}</SelectContent>
   501→              </Select>
   502→            </div>
   503→          </div>
   504→          <DialogFooter>
   505→            <Button variant="outline" onClick={() => setLogFromSearchDialog({ open: false })} className="rounded-xl">Cancel</Button>
   506→            <Button className="bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl font-semibold" onClick={handleLogFromSearch}>Log It</Button>
   507→          </DialogFooter>
   508→        </DialogContent>
   509→      </Dialog>
   510→
   511→      {/* Quick Re-log Dialog */}
   512→      <Dialog open={relogDialog.open} onOpenChange={(open) => setRelogDialog(open ? relogDialog : { open: false })}>
   513→        <DialogContent className="max-w-sm">
   514→          <DialogHeader>
   515→            <DialogTitle>Log Again</DialogTitle>
   516→            <DialogDescription>{relogDialog.open ? relogDialog.item.meal?.name || 'This meal' : ''} ({relogDialog.open ? relogDialog.item.servingGms : 0}g)</DialogDescription>
   517→          </DialogHeader>
   518→          <div className="space-y-3">
   519→            <div className="space-y-2">
   520→              <Label className="text-sm font-medium">Log as</Label>
   521→              <Select value={relogSlot} onValueChange={setRelogSlot}>
   522→                <SelectTrigger className="h-11 rounded-xl"><SelectValue /></SelectTrigger>
   523→                <SelectContent>{SLOTS.map((s) => <SelectItem key={s} value={s}>{SLOT_LABELS[s]}</SelectItem>)}</SelectContent>
   524→              </Select>
   525→            </div>
   526→          </div>
   527→          <DialogFooter>
   528→            <Button variant="outline" onClick={() => setRelogDialog({ open: false })} className="rounded-xl">Cancel</Button>
   529→            <Button className="bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl font-semibold" onClick={handleRelog}>Log Again</Button>
   530→          </DialogFooter>
   531→        </DialogContent>
   532→      </Dialog>
   533→    </motion.div>
   534→  );
   535→}
