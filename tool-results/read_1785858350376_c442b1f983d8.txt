     4→import { format } from 'date-fns';

     6→import { toast } from 'sonner';
     7→import { Button } from '@/components/ui/button';

    28→} from 'lucide-react';
    29→import { apiFetch } from './api';
    30→import {
    31→  SLOTS, SLOT_LABELS, SLOT_ICONS, SLOT_BORDER_COLORS, SLOT_BADGE_COLORS,
    32→  SLOT_GRADIENT_COLORS, fadeIn,
    33→} from './constants';
    34→import { CalorieRing, NutritionFactsLabel } from './shared';
    35→import type { ViewType, NutritionData, MealRecommendation, SearchMeal, MealPlanItemData } from './types';
    36→
    37→export function DashboardView({ onNavigate }: { onNavigate: (v: ViewType) => void }) {
    38→  const [user, setUser] = useState<Record<string, unknown> | null>(null);
    39→  const [nutrition, setNutrition] = useState<NutritionData | null>(null);
    40→  const [recommendations, setRecommendations] = useState<Record<string, MealRecommendation[]>>({});
    41→  const [loading, setLoading] = useState(true);
    42→  const [streak, setStreak] = useState(0);
    43→
    44→  // Log meal dialog state
    45→  const [logDialog, setLogDialog] = useState<{
    46→    open: boolean;
    47→    meal: MealRecommendation;
    48→    slot: string;
    49→    baseNutritionPer100g: { calories: number; proteinG: number; carbsG: number; fatG: number } | null;
    50→    mealName: string;
    51→    cuisine: string;
    52→    mealType: string;
    53→    description?: string | null;
    54→    isVeg?: boolean;
    55→    isVegan?: boolean;
    56→  } | { open: false }>({ open: false });
    57→  const [servingGms, setServingGms] = useState(100);
    58→
    59→  // Meal search dialog state (per-slot)
    60→  const [slotSearchDialog, setSlotSearchDialog] = useState<{ open: boolean; slot: string } | { open: false }>({ open: false });
    61→  const [searchQuery, setSearchQuery] = useState('');
    62→  const [searchResults, setSearchResults] = useState<SearchMeal[]>([]);
    63→  const [searchLoading, setSearchLoading] = useState(false);
    64→  const [searchDebounce, setSearchDebounce] = useState<ReturnType<typeof setTimeout> | null>(null);
    65→
    66→  // Meal detail sheet
    67→  const [detailSheet, setDetailSheet] = useState<{
    68→    open: boolean;
    69→    rec: MealRecommendation;
    70→  } | { open: false }>({ open: false });
    71→
    72→  // Achievements state
    73→  const [achievements, setAchievements] = useState<{ id: string; name: string; description: string; icon: string; earned: boolean; earnedDate?: string }[]>([]);
    74→
    75→  // Meal plan state (Feature 2)
    76→  const [mealPlan, setMealPlan] = useState<{
    77→    exists: boolean;
    78→    items: MealPlanItemData[];
    79→  } | null>(null);
    80→  const [planLoading, setPlanLoading] = useState(false);
    81→  const [planExpanded, setPlanExpanded] = useState<Record<string, boolean>>({});
    82→
    83→  const fetchData = useCallback(async () => {
    84→    try {
    85→      const [me, nut, summaryData] = await Promise.all([
    86→        apiFetch('/api/auth/me'),
    87→        apiFetch('/api/nutrition/daily'),
    88→        apiFetch('/api/progress/summary?period=week').catch(() => ({ summary: { totalDays: 0 } })),
    89→      ]);
    90→      setUser(me);
    91→      setNutrition(nut);
    92→
    93→      // Calculate streak from weekly data
    94→      const weeklyData = await apiFetch('/api/progress/weekly').catch(() => [] as Record<string, unknown>[]);
    95→      let streakCount = 0;
    96→      for (let i = (weeklyData as Record<string, unknown>[]).length - 1; i >= 0; i--) {
    97→        const day = weeklyData[i];
    98→        const consumed = (day.consumed as Record<string, number>)?.calories || 0;
    99→        if (consumed > 0) streakCount++;
   100→        else break;
   101→      }
   102→      setStreak(streakCount);
   103→
   104→      const recs = await Promise.all(
   105→        SLOTS.map(async (slot) => {
   106→          try {
   107→            const r = await apiFetch(`/api/recommendations?slot=${slot}`);
   108→            return { slot, recs: r.recommendations || [] };
   109→          } catch { return { slot, recs: [] }; }
   110→        })
   111→      );
   112→      const recMap: Record<string, MealRecommendation[]> = {};
   113→      for (const r of recs) recMap[r.slot] = r.recs;
   114→      setRecommendations(recMap);
   115→
   116→      // Fetch meal plan (Feature 2)
   117→      try {
   118→        const plan = await apiFetch('/api/meal-plan');
   119→        setMealPlan({ exists: plan.exists, items: plan.items || [] });
   120→      } catch {
   121→        setMealPlan(null);
   122→      }
   123→
   124→      // Fetch achievements
   125→      try {
   126→        const ach = await apiFetch('/api/achievements');
   127→        setAchievements(ach.achievements || []);
   128→      } catch {
   129→        // Silent fail
   130→      }
   131→    } catch {
   132→      toast.error('Failed to load dashboard');
   133→    } finally {
   134→      setLoading(false);
   135→    }
   136→  }, []);
   137→
   138→  useEffect(() => { fetchData(); }, [fetchData]);
   139→
   140→  const handleLogMeal = async () => {
   141→    if (!logDialog.open) return;
   142→    try {
   143→      await apiFetch('/api/food-logs', {
   144→        method: 'POST',
   145→        body: JSON.stringify({ mealId: logDialog.meal.meal.id, servingGms, mealSlot: logDialog.slot }),
   146→      });
   147→      toast.success(`Logged ${logDialog.mealName}!`);
   148→      setLogDialog({ open: false });
   149→      fetchData();
   150→    } catch (err) { toast.error((err as Error).message); }
   151→  };
   152→
   153→  // Per-slot search with debounce
   154→  const handleSlotSearch = (query: string) => {
   155→    setSearchQuery(query);
   156→    if (searchDebounce) clearTimeout(searchDebounce);
   157→    if (!query.trim()) { setSearchResults([]); return; }
   158→    const timer = setTimeout(async () => {
   159→      setSearchLoading(true);
   160→      try {
   161→        const res = await apiFetch(`/api/meals/search?q=${encodeURIComponent(query)}`);
   162→        setSearchResults((res.meals || []) as SearchMeal[]);
   163→      } catch { toast.error('Search failed'); }
   164→      finally { setSearchLoading(false); }
   165→    }, 350);
   166→    setSearchDebounce(timer);
   167→  };
   168→
   169→  const handleLogFromSearch = (meal: SearchMeal) => {
   170→    const slot = slotSearchDialog.open ? slotSearchDialog.slot : 'lunch';
   171→    setServingGms(meal.baseServingGms || 100);
   172→    setLogDialog({
   173→      open: true,
   174→      meal: {
   175→        meal: { id: meal.id, name: meal.name, mealType: meal.mealType, cuisine: meal.cuisine, imageUrl: null, prepTimeMin: meal.prepTimeMin, isVeg: meal.isVeg, isVegan: meal.isVegan },
   176→        score: 0, recommendedServingGms: meal.baseServingGms || 100, estimatedNutrition: meal.nutrition ? { calories: meal.nutrition.calories, proteinG: meal.nutrition.proteinG, carbsG: meal.nutrition.carbsG, fatG: meal.nutrition.fatG } : null,
   177→        baseNutritionPer100g: meal.nutrition ? { calories: meal.nutrition.calories, proteinG: meal.nutrition.proteinG, carbsG: meal.nutrition.carbsG, fatG: meal.nutrition.fatG } : null,
   178→      },
   179→      slot,
   180→      baseNutritionPer100g: meal.nutrition || null,
   181→      mealName: meal.name,
   182→      cuisine: meal.cuisine,
   183→      mealType: meal.mealType,
   184→      isVeg: meal.isVeg,
   185→      isVegan: meal.isVegan,
   186→    });
   187→    setSlotSearchDialog({ open: false });
   188→    setSearchQuery('');
   189→    setSearchResults([]);
   190→  };
   191→
   192→  const handleWaterAdd = async () => {
   193→    try {
   194→      await apiFetch('/api/water-log', { method: 'POST', body: JSON.stringify({ glasses: 1 }) });
   195→      toast.success('\uD83D\uDCA7 +1 glass of water');
   196→    } catch { toast.error('Failed to log water'); }
   197→  };
   198→
   199→  // Generate meal plan (Feature 2)
   200→  const handleGeneratePlan = async () => {
   201→    setPlanLoading(true);
   202→    try {
   203→      const result = await apiFetch('/api/meal-plan/generate');
   204→      toast.success(result.status === 'existing' ? 'Plan already exists for today!' : "Today's meal plan generated!");
   205→      // Fetch the plan
   206→      const plan = await apiFetch('/api/meal-plan');
   207→      setMealPlan({ exists: plan.exists, items: plan.items || [] });
   208→    } catch (err) { toast.error((err as Error).message); }
   209→    finally { setPlanLoading(false); }
   210→  };
   211→
   212→  const firstName = (user?.profile as Record<string, unknown> | null)?.firstName || (user?.name as string) || 'there';
   213→  const todayStr = format(new Date(), 'EEEE, MMMM d');
   214→
   215→  // Insights logic (Feature 3)
   216→  const targetCal = nutrition?.targets?.calories || 0;
   217→  const consumedCal = nutrition?.consumed?.calories || 0;
   218→  const pctConsumed = targetCal > 0 ? consumedCal / targetCal : 0;
   219→  const targetProtein = nutrition?.targets?.proteinG || 0;
   220→  const consumedProtein = nutrition?.consumed?.proteinG || 0;
   221→  const pctProtein = targetProtein > 0 ? consumedProtein / targetProtein : 0;
   222→  const hour = new Date().getHours();
   223→  const timeOfDayTip = hour < 12
   224→    ? 'Start your day with a protein-rich breakfast!'
   225→    : hour < 17
   226→      ? 'Keep up the good work! Stay consistent with your meals.'
   227→      : 'Plan a balanced dinner to finish strong today.';
   228→
   229→  if (loading) {
   230→    return (
   231→      <div className="p-4 max-w-lg mx-auto space-y-4">
   232→        <Skeleton className="h-8 w-56 rounded-xl" />
   233→        <div className="flex justify-center"><Skeleton className="h-[180px] w-[180px] rounded-full" /></div>
   234→        <div className="space-y-3"><Skeleton className="h-10 w-full rounded-xl" /><Skeleton className="h-10 w-full rounded-xl" /><Skeleton className="h-10 w-full rounded-xl" /></div>
   235→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4"><Skeleton className="h-56 w-full rounded-xl" /><Skeleton className="h-56 w-full rounded-xl" /></div>
   236→      </div>
   237→    );
   238→  }
   239→
   240→  return (
   241→    <motion.div {...fadeIn} className="p-4 max-w-lg mx-auto space-y-4">
   242→      {/* ═══ Hero Section with gradient bg & decorative blobs ═══ */}
   243→      <div className="relative overflow-hidden rounded-2xl bg-gradient-to-r from-emerald-500/5 to-teal-500/5 p-5 border border-emerald-100/50 backdrop-blur-sm">
   244→        <div className="absolute -top-8 -right-8 w-24 h-24 bg-emerald-200/20 rounded-full blur-2xl" />
   245→        <div className="absolute -bottom-6 -left-6 w-20 h-20 bg-teal-200/20 rounded-full blur-2xl" />
   246→        <div className="absolute top-1/2 right-1/4 w-12 h-12 bg-green-200/10 rounded-full blur-xl" />
   247→        <div className="relative flex items-start justify-between">
   248→          <div>
   249→            <h1 className="text-2xl font-bold text-gray-900">Hi, {firstName}! {'\uD83D\uDC4B'}</h1>
   250→            <div className="flex items-center gap-1.5 mt-1 text-sm text-gray-500">
   251→              <CalendarIcon className="h-3.5 w-3.5" />
   252→              <span>{todayStr}</span>
   253→            </div>
   254→          </div>
   255→          <div className="flex items-center gap-2">
   256→            {streak > 0 && (
   257→              <Badge className="bg-amber-50 text-amber-700 border border-amber-200 px-2.5 py-1 gap-1 shadow-sm">
   258→                <Zap className="h-3 w-3" />{streak} day{streak > 1 ? 's' : ''}
   259→              </Badge>
   260→            )}
   261→            <div className="w-10 h-10 bg-emerald-100 rounded-xl flex items-center justify-center shadow-sm">
   262→              <User className="h-5 w-5 text-emerald-600" />
   263→            </div>
   264→          </div>
   265→        </div>
   266→      </div>
   267→
   268→      {/* ═══ Calorie Ring Card ═══ */}
   269→      <Card className="p-6 rounded-2xl shadow-sm border border-gray-100/80">
   270→        <div className="flex justify-center">
   271→          <CalorieRing consumed={consumedCal} target={targetCal} />
   272→        </div>
   273→        {pctConsumed >= 1 && (
   274→          <p className="text-center text-sm text-rose-600 font-medium mt-2">
   275→            You&apos;ve reached your calorie goal! Great job! {'\uD83C\uDF89'}
   276→          </p>
   277→        )}
   278→      </Card>
   279→
   280→      {/* ═══ Macro Progress Bars ═══ */}
   281→      <Card className="p-4 rounded-2xl shadow-sm border border-gray-100/80 space-y-3">
   282→        <h3 className="text-sm font-semibold text-gray-700">Today&apos;s Macros</h3>
   283→        {[
   284→          { label: 'Protein', val: consumedProtein, target: targetProtein, from: '#3b82f6', to: '#60a5fa', unit: 'g' },
   285→          { label: 'Carbs', val: nutrition?.consumed?.carbsG || 0, target: nutrition?.targets?.carbsG || 0, from: '#f59e0b', to: '#fbbf24', unit: 'g' },
   286→          { label: 'Fat', val: nutrition?.consumed?.fatG || 0, target: nutrition?.targets?.fatG || 0, from: '#f43f5e', to: '#fb7185', unit: 'g' },
   287→        ].map((m) => {
   288→          const pct = Math.min(Math.round((m.val / (m.target || 1)) * 100), 100);
   289→          return (
   290→            <div key={m.label}>
   291→              <div className="flex justify-between mb-1.5">
   292→                <span className="text-sm text-gray-600 font-medium">{m.label}</span>
   293→                <span className="text-xs text-gray-600 font-medium">{m.val}{m.unit} / {m.target}{m.unit} &middot; {pct}%</span>
   294→              </div>
   295→              <div className="h-3 bg-gray-100 rounded-full overflow-hidden">
   296→                <motion.div
   297→                  className="h-full rounded-full"
   298→                  style={{ background: `linear-gradient(90deg, ${m.from}, ${m.to})` }}
   299→                  initial={{ width: 0 }}
   300→                  animate={{ width: `${pct}%` }}
   301→                  transition={{ duration: 0.8 }}
   302→                />
   303→              </div>
   304→            </div>
   305→          );
   306→        })}
   307→      </Card>
   308→
   309→      {/* ═══ Feature 3: Today's Insights Card ═══ */}
   310→      <motion.div
   311→        initial={{ opacity: 0, y: 8 }}
   312→        animate={{ opacity: 1, y: 0 }}
   313→        transition={{ delay: 0.3 }}
   314→      >
   315→        <Card className={`p-4 rounded-2xl shadow-sm border border-gray-100/80 ${pctConsumed > 1 ? 'bg-amber-50/50' : pctConsumed >= 0.75 ? 'bg-emerald-50/50' : 'bg-blue-50/50'}`}>
   316→          <div className="flex items-start gap-3">
   317→            <div className={`w-9 h-9 rounded-xl flex items-center justify-center shrink-0 ${pctConsumed > 1 ? 'bg-amber-100/80' : pctConsumed >= 0.75 ? 'bg-emerald-100/80' : 'bg-blue-100/80'}`}>
   318→              <Lightbulb className={`h-4.5 w-4.5 ${pctConsumed > 1 ? 'text-amber-500' : pctConsumed >= 0.75 ? 'text-emerald-500' : 'text-blue-500'}`} />
   319→            </div>
   320→            <div className="flex-1 min-w-0 space-y-2">
   321→              <h3 className="text-sm font-semibold text-gray-900">Today&apos;s Insights</h3>
   322→
   323→              {/* Calorie status message */}
   324→              {pctConsumed > 1 && (
   325→                <p className="text-xs text-amber-700 leading-relaxed">
   326→                  You&apos;ve exceeded your calorie goal by <span className="font-bold">{Math.round(consumedCal - targetCal)} kcal</span>. Consider lighter meals for the rest of the day.
   327→                </p>
   328→              )}
   329→              {pctConsumed >= 0.75 && pctConsumed <= 1 && (
   330→                <p className="text-xs text-emerald-700 leading-relaxed">
   331→                  Great progress! You&apos;re <span className="font-bold">{Math.round(targetCal - consumedCal)} kcal</span> away from your daily goal.
   332→                </p>
   333→              )}
   334→              {pctConsumed < 0.75 && pctConsumed > 0 && (
   335→                <p className="text-xs text-blue-700 leading-relaxed">
   336→                  You still have <span className="font-bold">{Math.round(targetCal - consumedCal)} kcal</span> remaining. Time to fuel up!
   337→                </p>
   338→              )}
   339→              {consumedCal === 0 && (
   340→                <p className="text-xs text-gray-500 leading-relaxed">
   341→                  No meals logged yet today. Start with a balanced breakfast!
   342→                </p>
   343→              )}
   344→
   345→              {/* Protein status */}
   346→              {consumedProtein > 0 && pctProtein >= 0.8 && (
   347→                <p className="text-xs text-emerald-600 font-medium">
   348→                  Protein goal almost reached! {'\uD83C\uDF89'} ({Math.round(consumedProtein)}/{Math.round(targetProtein)}g)
   349→                </p>
   350→              )}
   351→
   352→              {/* Time-of-day tip */}
   353→              <p className="text-xs text-gray-500 italic">{'\uD83D\uDCAC'} {timeOfDayTip}</p>
   354→            </div>
   355→          </div>
   356→        </Card>
   357→      </motion.div>
   358→
   359→      {/* ═══ Feature 2: Meal Plan Generator Banner ═══ */}
   360→      <motion.div
   361→        initial={{ opacity: 0, y: 8 }}
   362→        animate={{ opacity: 1, y: 0 }}
   363→        transition={{ delay: 0.4 }}
   364→      >
   365→        <Card className="p-4 rounded-2xl shadow-sm border border-gray-100/80 bg-gradient-to-r from-emerald-500/5 to-teal-500/5">
   366→          {!mealPlan?.exists ? (
   367→            <div className="flex items-center justify-between">
   368→              <div className="flex items-center gap-3">
   369→                <div className="w-9 h-9 bg-emerald-100 rounded-xl flex items-center justify-center">
   370→                  <Sparkles className="h-4.5 w-4.5 text-emerald-600" />
   371→                </div>
   372→                <div>
   373→                  <h3 className="text-sm font-semibold text-gray-900">Today&apos;s Plan</h3>
   374→                  <p className="text-xs text-gray-500">Generate a personalized meal plan</p>
   375→                </div>
   376→              </div>
   377→              <Button
   378→                size="sm"
   379→                className="bg-emerald-600 hover:bg-emerald-700 text-white rounded-lg font-medium min-h-[36px]"
   380→                onClick={handleGeneratePlan}
   381→                disabled={planLoading}
   382→              >
   383→                {planLoading ? <Loader2 className="h-4 w-4 animate-spin mr-1.5" /> : <Sparkles className="h-4 w-4 mr-1.5" />}
   384→                Generate
   385→              </Button>
   386→            </div>
   387→          ) : (
   388→            <div className="space-y-2">
   389→              <div className="flex items-center justify-between">
   390→                <div className="flex items-center gap-3">
   391→                  <div className="w-9 h-9 bg-emerald-100 rounded-xl flex items-center justify-center">
   392→                    <Sparkles className="h-4.5 w-4.5 text-emerald-600" />
   393→                  </div>
   394→                  <div>
   395→                    <h3 className="text-sm font-semibold text-gray-900">Today&apos;s Meal Plan</h3>
   396→                    <p className="text-xs text-gray-500">AI-generated for your goals</p>
   397→                  </div>
   398→                </div>
   399→                <Badge className="bg-emerald-50 text-emerald-700 border-emerald-200 text-xs">Active</Badge>
   400→              </div>
   401→
   402→              {/* Plan items by slot */}
   403→              {SLOTS.map((slot) => {
   404→                const slotItems = (mealPlan?.items || []).filter((i) => i.mealSlot === slot);
   405→                if (slotItems.length === 0) return null;
   406→                const isExpanded = planExpanded[slot] || false;
   407→                return (
   408→                  <Collapsible
   409→                    key={slot}
   410→                    open={isExpanded}
   411→                    onOpenChange={(open) => setPlanExpanded((prev) => ({ ...prev, [slot]: open }))}
   412→                  >
   413→                    <CollapsibleTrigger asChild>
   414→                      <button className={`w-full flex items-center gap-2 p-2.5 rounded-xl border border-gray-100 hover:bg-white/80 transition-colors text-left ${SLOT_GRADIENT_COLORS[slot].from} ${SLOT_GRADIENT_COLORS[slot].to}`}>
   415→                        <span>{SLOT_ICONS[slot]}</span>
   416→                        <span className="text-xs font-semibold text-gray-700">{SLOT_LABELS[slot]}</span>
   417→                        <span className="text-xs text-gray-500 truncate flex-1">{slotItems[0]?.meal.name}</span>
   418→                        <ChevronDown className={`h-3.5 w-3.5 text-gray-400 transition-transform ${isExpanded ? 'rotate-180' : ''}`} />
   419→                      </button>
   420→                    </CollapsibleTrigger>
   421→                    <AnimatePresence>
   422→                      {isExpanded && (
   423→                        <CollapsibleContent forceMount>
   424→                          <motion.div
   425→                            initial={{ height: 0, opacity: 0 }}
   426→                            animate={{ height: 'auto', opacity: 1 }}
   427→                            exit={{ height: 0, opacity: 0 }}
   428→                            transition={{ duration: 0.2 }}
   429→                            className="overflow-hidden"
   430→                          >
   431→                            <div className="pl-4 pr-1 pt-1 space-y-1.5">
   432→                              {slotItems.map((item) => (
   433→                                <div key={item.id} className="flex items-center justify-between p-2.5 rounded-lg bg-white border border-gray-100">
   434→                                  <div className="flex-1 min-w-0">
   435→                                    <p className="text-xs font-medium text-gray-800 truncate">{item.meal.name}</p>
   436→                                    <div className="flex items-center gap-1.5 mt-0.5">
   437→                                      <Badge variant="outline" className="text-xs px-1.5 py-0 h-4 font-medium border-gray-200 text-gray-500">{item.meal.cuisine}</Badge>
