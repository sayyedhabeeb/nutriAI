     1→'use client';
     2→
     3→import React, { useState, useEffect, useCallback } from 'react';
     4→import { format, parseISO } from 'date-fns';
     5→import { motion, AnimatePresence } from 'framer-motion';
     6→import { toast } from 'sonner';
     7→import {
     8→  BarChart, Bar, AreaChart, Area, PieChart, Pie, Cell,
     9→  XAxis, YAxis, Tooltip, ResponsiveContainer, ReferenceLine,
    10→} from 'recharts';
    11→import { Button } from '@/components/ui/button';
    12→import { Card } from '@/components/ui/card';
    13→import { Input } from '@/components/ui/input';
    14→import { Skeleton } from '@/components/ui/skeleton';
    15→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
    16→import {
    17→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogClose,
    18→} from '@/components/ui/dialog';
    19→import { Label } from '@/components/ui/label';
    20→import {
    21→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    22→} from '@/components/ui/select';
    23→import {
    24→  TrendingUp, Flame, Dumbbell, Scale, Droplets, Plus, Minus,
    25→  BarChart3, Download, Lightbulb, Trophy, Lock, CalendarDays,
    26→} from 'lucide-react';
    27→import { apiFetch } from './api';
    28→import { PIE_COLORS, fadeIn } from './constants';
    29→
    30→export function ProgressView() {
    31→  const [tab, setTab] = useState('weekly');
    32→  const [weeklyData, setWeeklyData] = useState<Record<string, unknown>[]>([]);
    33→  const [summary, setSummary] = useState<Record<string, unknown> | null>(null);
    34→  const [weightLogs, setWeightLogs] = useState<Record<string, unknown>[]>([]);
    35→  const [waterGlasses, setWaterGlasses] = useState(0);
    36→  const [loading, setLoading] = useState(true);
    37→  const [weightInput, setWeightInput] = useState('');
    38→  const [weightNotes, setWeightNotes] = useState('');
    39→  const [justFilledIndex, setJustFilledIndex] = useState<number | null>(null);
    40→
    41→  // Achievements state
    42→  const [achievements, setAchievements] = useState<{ id: string; name: string; description: string; icon: string; earned: boolean; earnedDate?: string }[]>([]);
    43→
    44→  // Export dialog state
    45→  const [exportOpen, setExportOpen] = useState(false);
    46→  const [exportFormat, setExportFormat] = useState('csv');
    47→  const [exportDays, setExportDays] = useState('7');
    48→  const [exporting, setExporting] = useState(false);
    49→
    50→  const fetchProgress = useCallback(async () => {
    51→    setLoading(true);
    52→    try {
    53→      const [weekData, sumData, wLogs, waterData] = await Promise.all([
    54→        apiFetch('/api/progress/weekly'),
    55→        apiFetch(`/api/progress/summary?period=${tab === 'monthly' ? 'month' : 'week'}`),
    56→        apiFetch('/api/weight-log?limit=30'),
    57→        apiFetch('/api/water-log'),
    58→      ]);
    59→      setWeeklyData(weekData || []);
    60→      setSummary(sumData?.summary || null);
    61→      setWeightLogs(wLogs || []);
    62→      setWaterGlasses(waterData?.glassesConsumed || 0);
    63→
    64→      // Fetch achievements
    65→      try {
    66→        const achData = await apiFetch('/api/achievements');
    67→        setAchievements(achData.achievements || []);
    68→      } catch {
    69→        // Silent fail
    70→      }
    71→    } catch { toast.error('Failed to load progress'); }
    72→    finally { setLoading(false); }
    73→  }, [tab]);
    74→
    75→  useEffect(() => { fetchProgress(); }, [fetchProgress]);
    76→
    77→  const handleAddWater = async (delta: number) => {
    78→    const newCount = waterGlasses + delta;
    79→    if (newCount < 0) return;
    80→    if (delta > 0) {
    81→      setJustFilledIndex(waterGlasses);
    82→      setTimeout(() => setJustFilledIndex(null), 600);
    83→      try {
    84→        await apiFetch('/api/water-log', { method: 'POST', body: JSON.stringify({ glasses: delta }) });
    85→        setWaterGlasses(newCount);
    86→        toast.success(`\uD83D\uDCA7 +${delta} glass${delta > 1 ? 'es' : ''} of water`);
    87→      } catch { toast.error('Failed to log water'); }
    88→    } else {
    89→      setWaterGlasses(Math.max(0, newCount));
    90→    }
    91→  };
    92→
    93→  const handleWaterAdd = async () => {
    94→    setJustFilledIndex(waterGlasses);
    95→    setTimeout(() => setJustFilledIndex(null), 600);
    96→    try {
    97→      await apiFetch('/api/water-log', { method: 'POST', body: JSON.stringify({ glasses: 1 }) });
    98→      setWaterGlasses(waterGlasses + 1);
    99→      toast.success('\uD83D\uDCA7 +1 glass of water');
   100→    } catch { toast.error('Failed to log water'); }
   101→  };
   102→
   103→  const handleLogWeight = async () => {
   104→    if (!weightInput || Number(weightInput) <= 0) return;
   105→    try {
   106→      await apiFetch('/api/weight-log', { method: 'POST', body: JSON.stringify({ weightKg: Number(weightInput), notes: weightNotes || null }) });
   107→      toast.success('Weight logged!');
   108→      setWeightInput('');
   109→      setWeightNotes('');
   110→      fetchProgress();
   111→    } catch (err) { toast.error((err as Error).message); }
   112→  };
   113→
   114→  const handleExportCsv = () => {
   115→    if (weeklyData.length === 0) {
   116→      toast.error('No data to export');
   117→      return;
   118→    }
   119→    const header = 'Date,Calories,Protein,Carbs,Fat';
   120→    const rows = weeklyData.map((d) => {
   121→      const consumed = d.consumed as Record<string, number>;
   122→      return `${d.date},${consumed?.calories || 0},${consumed?.proteinG || 0},${consumed?.carbsG || 0},${consumed?.fatG || 0}`;
   123→    });
   124→    const csvContent = [header, ...rows].join('\n');
   125→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
   126→    const url = URL.createObjectURL(blob);
   127→    const a = document.createElement('a');
   128→    a.href = url;
   129→    a.download = `nutriai-${tab}-report.csv`;
   130→    document.body.appendChild(a);
   131→    a.click();
   132→    document.body.removeChild(a);
   133→    URL.revokeObjectURL(url);
   134→    toast.success('Report exported!');
   135→    setExportOpen(false);
   136→  };
   137→
   138→  const handleExport = async () => {
   139→    if (exportFormat === 'csv') {
   140→      handleExportCsv();
   141→      return;
   142→    }
   143→    setExporting(true);
   144→    try {
   145→      const token = localStorage.getItem('nutriai_session');
   146→      const url = `/api/export?format=${exportFormat}&days=${exportDays}`;
   147→      const res = await fetch(url, {
   148→        headers: { Authorization: `Bearer ${token}` },
   149→      });
   150→      if (!res.ok) throw new Error('Export failed');
   151→      const blob = await res.blob();
   152→      const ext = exportFormat === 'csv' ? 'csv' : 'json';
   153→      const contentType = exportFormat === 'csv' ? 'text/csv' : 'application/json';
   154→      const blobUrl = URL.createObjectURL(new Blob([blob], { type: contentType }));
   155→      const a = document.createElement('a');
   156→      a.href = blobUrl;
   157→      a.download = `nutriai-export-${exportDays}days.${ext}`;
   158→      document.body.appendChild(a);
   159→      a.click();
   160→      document.body.removeChild(a);
   161→      URL.revokeObjectURL(blobUrl);
   162→      toast.success('Data exported successfully!');
   163→      setExportOpen(false);
   164→    } catch {
   165→      toast.error('Failed to export data');
   166→    } finally {
   167→      setExporting(false);
   168→    }
   169→  };
   170→
   171→  const chartData = weeklyData.map((d) => {
   172→    const consumed = d.consumed as Record<string, number>;
   173→    const targets = d.targets as Record<string, number> | null;
   174→    return {
   175→      date: format(parseISO(d.date as string), 'EEE'),
   176→      calories: consumed?.calories || 0,
   177→      target: targets?.calories || 0,
   178→    };
   179→  });
   180→
   181→  const allCaloriesZero = chartData.every((d) => d.calories === 0 && d.target === 0);
   182→
   183→  const nonZeroDays = weeklyData.filter((d) => (d.consumed as Record<string, number>)?.calories > 0).length;
   184→  const isSparse = weeklyData.length > 0 && nonZeroDays <= 1;
   185→
   186→  const macroData = [
   187→    { name: 'Protein', value: summary?.avgProtein || 0, color: PIE_COLORS[0] },
   188→    { name: 'Carbs', value: summary?.avgCarbs || 0, color: PIE_COLORS[1] },
   189→    { name: 'Fat', value: summary?.avgFat || 0, color: PIE_COLORS[2] },
   190→  ];
   191→  const allMacrosZero = macroData.every((m) => m.value === 0);
   192→
   193→  const weightChartData = weightLogs.slice().reverse().slice(-7).map((w) => ({
   194→    date: format(parseISO(w.logDate as string), 'MMM d'),
   195→    weight: w.weightKg as number,
   196→  }));
   197→
   198→  const noWeightData = weightChartData.length < 2;
   199→
   200→  // Calorie breakdown table data - filter out zero-eaten days
   201→  const breakdownData = weeklyData
   202→    .map((d) => {
   203→      const consumed = d.consumed as Record<string, number>;
   204→      const targets = d.targets as Record<string, number> | null;
   205→      const consumedCal = consumed?.calories || 0;
   206→      const targetCal = targets?.calories || 0;
   207→      const diff = consumedCal - targetCal;
   208→      const pct = targetCal > 0 ? Math.round((consumedCal / targetCal) * 100) : 0;
   209→      return {
   210→        date: d.date as string,
   211→        consumedCal,
   212→        targetCal,
   213→        diff,
   214→        pct: Math.min(pct, 150),
   215→      };
   216→    })
   217→    .filter((d) => d.consumedCal > 0);
   218→
   219→  const weeklyTotalConsumed = breakdownData.reduce((sum, d) => sum + d.consumedCal, 0);
   220→  const weeklyTotalTarget = breakdownData.reduce((sum, d) => sum + d.targetCal, 0);
   221→  const weeklyTotalDiff = weeklyTotalConsumed - weeklyTotalTarget;
   222→
   223→  // Yearly summary data
   224→  const yearlyDaysLogged = summary?.totalDays || 0;
   225→  const yearlyAvgCalories = Math.round(summary?.avgCalories || 0);
   226→  const yearlyHasData = yearlyDaysLogged > 0 || yearlyAvgCalories > 0;
   227→
   228→  const yearlyBestDay = weeklyData.reduce((best: Record<string, unknown> | null, d) => {
   229→    const cal = (d.consumed as Record<string, number>)?.calories || 0;
   230→    const bestCal = best ? (best.consumed as Record<string, number>)?.calories || 0 : 0;
   231→    return cal > bestCal ? d : best;
   232→  }, null);
   233→  const yearlyWorstDay = weeklyData.reduce((worst: Record<string, unknown> | null, d) => {
   234→    const cal = (d.consumed as Record<string, number>)?.calories || 0;
   235→    const worstCal = worst ? (worst.consumed as Record<string, number>)?.calories || 0 : Infinity;
   236→    return (cal < worstCal && cal > 0) ? d : worst;
   237→  }, null);
   238→
   239→  const currentWeight = summary?.currentWeight as number | undefined;
   240→  const firstWeightLog = weightLogs.length > 0 ? weightLogs[0] : null;
   241→  const startWeight = firstWeightLog ? (firstWeightLog.weightKg as number) : null;
   242→  const weightDiff = currentWeight && startWeight ? (currentWeight - startWeight).toFixed(1) : null;
   243→
   244→  const statCards = [
   245→    { label: 'Avg Daily Calories', value: `${Math.round(summary?.avgCalories || 0)}`, icon: Flame, iconBg: 'bg-orange-100 dark:bg-orange-900/30', iconColor: 'text-orange-600 dark:text-orange-400' },
   246→    { label: 'Avg Protein', value: `${Math.round(summary?.avgProtein || 0)}g`, icon: Dumbbell, iconBg: 'bg-blue-100 dark:bg-blue-900/30', iconColor: 'text-blue-600 dark:text-blue-400' },
   247→    { label: 'Total Days', value: `${summary?.totalDays || 0}`, icon: CalendarDays, iconBg: 'bg-emerald-100 dark:bg-emerald-900/30', iconColor: 'text-emerald-600 dark:text-emerald-400' },
   248→    { label: 'Current Weight', value: currentWeight ? `${currentWeight}kg` : 'N/A', icon: Scale, iconBg: 'bg-purple-100 dark:bg-purple-900/30', iconColor: 'text-purple-600 dark:text-purple-400' },
   249→  ];
   250→
   251→  if (loading) {
   252→    return (
   253→      <div className="p-4 max-w-lg mx-auto space-y-4">
   254→        <Skeleton className="h-8 w-32 rounded-xl" />
   255→        <Skeleton className="h-64 w-full rounded-2xl" />
   256→        <div className="grid grid-cols-2 gap-4"><Skeleton className="h-28 w-full rounded-2xl" /><Skeleton className="h-28 w-full rounded-2xl" /></div>
   257→      </div>
   258→    );
   259→  }
   260→
   261→  return (
   262→    <motion.div {...fadeIn} className="p-4 max-w-lg mx-auto space-y-5 pb-28">
   263→      {/* Header */}
   264→      <div className="flex items-center justify-between">
   265→        <div className="flex items-center gap-2">
   266→          <div className="w-9 h-9 rounded-xl bg-emerald-100 flex items-center justify-center">
   267→            <TrendingUp className="h-5 w-5 text-emerald-600" />
   268→          </div>
   269→          <h1 className="text-xl font-bold text-gray-900">Progress</h1>
   270→        </div>
   271→        <button
   272→          onClick={() => setExportOpen(true)}
   273→          className="border border-gray-200 dark:border-gray-700 bg-white dark:bg-gray-800 shadow-sm hover:shadow-md rounded-xl px-3.5 py-2 text-xs font-medium text-gray-600 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-700 transition-all flex items-center gap-2"
   274→        >
   275→          <Download className="h-4 w-4" />
   276→          <span className="hidden sm:inline">Export</span>
   277→        </button>
   278→      </div>
   279→
   280→      {/* Export Dialog */}
   281→      <Dialog open={exportOpen} onOpenChange={setExportOpen}>
   282→        <DialogContent className="sm:max-w-sm rounded-2xl">
   283→          <DialogHeader>
   284→            <DialogTitle>Export Data</DialogTitle>
   285→          </DialogHeader>
   286→          <div className="space-y-4 py-2">
   287→            <div className="space-y-1.5">
   288→              <Label className="text-sm font-medium">Format</Label>
   289→              <Select value={exportFormat} onValueChange={setExportFormat}>
   290→                <SelectTrigger className="h-11 rounded-xl focus-visible:ring-2 focus-visible:ring-emerald-500 focus-visible:ring-offset-2 focus-visible:ring-offset-background"><SelectValue /></SelectTrigger>
   291→                <SelectContent>
   292→                  <SelectItem value="csv">CSV Spreadsheet</SelectItem>
   293→                  <SelectItem value="json">JSON</SelectItem>
   294→                </SelectContent>
   295→              </Select>
   296→            </div>
   297→            <div className="space-y-1.5">
   298→              <Label className="text-sm font-medium">Period</Label>
   299→              <Select value={exportDays} onValueChange={setExportDays}>
   300→                <SelectTrigger className="h-11 rounded-xl focus-visible:ring-2 focus-visible:ring-emerald-500 focus-visible:ring-offset-2 focus-visible:ring-offset-background"><SelectValue /></SelectTrigger>
   301→                <SelectContent>
   302→                  <SelectItem value="7">Last 7 days</SelectItem>
   303→                  <SelectItem value="14">Last 14 days</SelectItem>
   304→                  <SelectItem value="30">Last 30 days</SelectItem>
   305→                </SelectContent>
   306→              </Select>
   307→            </div>
   308→          </div>
   309→          <DialogFooter className="gap-2 sm:gap-0">
   310→            <DialogClose asChild>
   311→              <Button variant="outline" className="rounded-xl">Cancel</Button>
   312→            </DialogClose>
   313→            <Button className="bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl" onClick={handleExport} disabled={exporting}>
   314→              {exporting ? 'Exporting...' : 'Download'}
   315→            </Button>
   316→          </DialogFooter>
   317→        </DialogContent>
   318→      </Dialog>
   319→
   320→      <Tabs value={tab} onValueChange={setTab}>
   321→        <TabsList className="w-full rounded-xl">
   322→          <TabsTrigger value="weekly" className="flex-1 rounded-lg data-[state=active]:shadow-sm">Weekly</TabsTrigger>
   323→          <TabsTrigger value="monthly" className="flex-1 rounded-lg text-gray-500 dark:text-gray-400 data-[state=active]:text-foreground data-[state=active]:shadow-sm">Monthly</TabsTrigger>
   324→          <TabsTrigger value="yearly" className="flex-1 rounded-lg text-gray-500 dark:text-gray-400 data-[state=active]:text-foreground data-[state=active]:shadow-sm">Yearly</TabsTrigger>
   325→        </TabsList>
   326→
   327→        {/* Yearly Tab Content */}
   328→        <TabsContent value="yearly">
   329→          {yearlyHasData ? (
   330→            <Card className="p-5 rounded-2xl shadow-lg shadow-gray-200/50 dark:shadow-black/20 border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   331→              <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100 mb-4">Yearly Overview</h3>
   332→              <div className="space-y-4">
   333→                <div className="grid grid-cols-2 gap-3">
   334→                  <div className="p-3 rounded-xl bg-emerald-50 dark:bg-emerald-900/20">
   335→                    <p className="text-xs text-gray-500 dark:text-gray-400 font-medium">Total Days Logged</p>
   336→                    <p className="text-2xl font-bold text-emerald-700 dark:text-emerald-400 mt-1 tabular-nums">{yearlyDaysLogged}</p>
   337→                  </div>
   338→                  <div className="p-3 rounded-xl bg-orange-50 dark:bg-orange-900/20">
   339→                    <p className="text-xs text-gray-500 dark:text-gray-400 font-medium">Avg Daily Calories</p>
   340→                    <p className="text-2xl font-bold text-orange-700 dark:text-orange-400 mt-1 tabular-nums">{yearlyAvgCalories}</p>
   341→                  </div>
   342→                </div>
   343→                {yearlyBestDay && (
   344→                  <div className="p-3 rounded-xl bg-gray-50 dark:bg-gray-800/50">
   345→                    <p className="text-xs text-gray-500 dark:text-gray-400 font-medium">Best Day</p>
   346→                    <p className="text-sm font-semibold text-gray-800 dark:text-gray-200 mt-1">
   347→                      {format(parseISO(yearlyBestDay.date as string), 'EEEE, MMM d')} — {((yearlyBestDay.consumed as Record<string, number>)?.calories || 0).toLocaleString()} kcal
   348→                    </p>
   349→                  </div>
   350→                )}
   351→                {yearlyWorstDay && (
   352→                  <div className="p-3 rounded-xl bg-gray-50 dark:bg-gray-800/50">
   353→                    <p className="text-xs text-gray-500 dark:text-gray-400 font-medium">Lowest Day</p>
   354→                    <p className="text-sm font-semibold text-gray-800 dark:text-gray-200 mt-1">
   355→                      {format(parseISO(yearlyWorstDay.date as string), 'EEEE, MMM d')} — {((yearlyWorstDay.consumed as Record<string, number>)?.calories || 0).toLocaleString()} kcal
   356→                    </p>
   357→                  </div>
   358→                )}
   359→                {weightDiff !== null && (
   360→                  <div className="p-3 rounded-xl bg-purple-50 dark:bg-purple-900/20">
   361→                    <p className="text-xs text-gray-500 dark:text-gray-400 font-medium">Weight Change</p>
   362→                    <p className="text-sm font-semibold text-gray-800 dark:text-gray-200 mt-1">
   363→                      {startWeight}kg → {currentWeight}kg ({Number(weightDiff) > 0 ? '+' : ''}{weightDiff}kg)
   364→                    </p>
   365→                  </div>
   366→                )}
   367→              </div>
   368→            </Card>
   369→          ) : (
   370→            <Card className="p-8 rounded-2xl shadow-lg shadow-gray-200/50 dark:shadow-black/20 border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   371→              <div className="flex flex-col items-center justify-center py-8">
   372→                <CalendarDays className="h-12 w-12 text-gray-400 dark:text-gray-500 mb-3" />
   373→                <p className="text-sm font-medium text-gray-400 dark:text-gray-500">Start tracking to see your yearly overview</p>
   374→              </div>
   375→            </Card>
   376→          )}
   377→        </TabsContent>
   378→      </Tabs>
   379→
   380→      {/* Only show charts for weekly/monthly */
   381→      tab !== 'yearly' && (
   382→        <>
   383→          {/* Calorie Chart */}
   384→          <Card className="p-5 rounded-2xl shadow-lg shadow-gray-200/50 dark:shadow-black/20 border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   385→            <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100 mb-3">Calorie Intake</h3>
   386→            <div className="h-48 relative">
   387→              {isSparse && !allCaloriesZero && (
   388→                <div className="absolute inset-0 flex items-center justify-center z-10 bg-white/70 dark:bg-gray-900/70 rounded-lg">
   389→                  <p className="text-xs text-gray-400 dark:text-gray-500 font-medium">Log meals to see your weekly trends</p>
   390→                </div>
   391→              )}
   392→              {allCaloriesZero ? (
   393→                <div className="h-full flex flex-col items-center justify-center py-8">
   394→                  <Flame className="h-12 w-12 mb-3 text-gray-400 dark:text-gray-500" />
   395→                  <p className="text-sm font-medium text-gray-400 dark:text-gray-500">No calorie data yet</p>
   396→                </div>
   397→              ) : (
   398→                <ResponsiveContainer width="100%" height="100%">
   399→                  <BarChart data={chartData} barCategoryGap="20%">
   400→                    <XAxis dataKey="date" tick={{ fontSize: 11, fill: '#6b7280' }} axisLine={false} tickLine={false} />
   401→                    <YAxis tick={{ fontSize: 11, fill: '#6b7280' }} axisLine={false} tickLine={false} />
   402→                    <Tooltip
   403→                      contentStyle={{ borderRadius: '12px', border: '1px solid #e5e7eb', boxShadow: '0 4px 6px -1px rgba(0,0,0,0.1)' }}
   404→                    />
   405→                    <defs>
   406→                      <linearGradient id="barGrad" x1="0" y1="0" x2="0" y2="1">
   407→                        <stop offset="0%" stopColor="#10b981" stopOpacity={1} />
   408→                        <stop offset="100%" stopColor="#34d399" stopOpacity={0.8} />
   409→                      </linearGradient>
   410→                    </defs>
   411→                    <Bar dataKey="calories" fill="url(#barGrad)" radius={[4, 4, 0, 0]} name="Consumed" />
   412→                    <Bar dataKey="target" fill="#e5e7eb" radius={[4, 4, 0, 0]} name="Target" />
   413→                    {!allCaloriesZero && chartData[0]?.target > 0 && (
   414→                      <ReferenceLine y={chartData[0].target} stroke="#10b981" strokeDasharray="6 3" strokeOpacity={0.6} />
   415→                    )}
   416→                  </BarChart>
   417→                </ResponsiveContainer>
   418→              )}
   419→            </div>
   420→          </Card>
   421→
   422→          {/* Macro Pie + Weight Trend */}
   423→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   424→            <Card className="p-5 rounded-2xl shadow-lg shadow-gray-200/50 dark:shadow-black/20 border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   425→              <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100 mb-2">Macro Breakdown</h3>
