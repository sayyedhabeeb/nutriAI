    44→  const [mealsLoggedToday, setMealsLoggedToday] = useState(0);
    45→

    49→    meal: MealRecommendation;

    52→    mealName: string;

    55→    description?: string | null;
    56→    isVeg?: boolean;

    60→

    63→  const [searchQuery, setSearchQuery] = useState('');
    64→  const [searchResults, setSearchResults] = useState<SearchMeal[]>([]);
    65→  const [searchLoading, setSearchLoading] = useState(false);

    67→

    78→  const [mealPlan, setMealPlan] = useState<{

    80→    items: MealPlanItemData[];

    83→  const [calPulse, setCalPulse] = useState(false);

    90→        apiFetch('/api/auth/me'),
    91→        apiFetch('/api/nutrition/daily'),

   113→        // Silent fail
   114→      }

   141→      // Fetch meal plan (Feature 2)

   143→        const plan = await apiFetch('/api/meal-plan');
   144→        setMealPlan({ exists: plan.exists, items: plan.items || [] });

   147→      }

   149→      // Fetch achievements

   151→        const ach = await apiFetch('/api/achievements');
   152→        setAchievements(ach.achievements || []);

   168→      await apiFetch('/api/food-logs', {
   169→        method: 'POST',
   170→        body: JSON.stringify({ mealId: logDialog.meal.meal.id, servingGms, mealSlot: logDialog.slot }),
   171→      });

   173→      setLogDialog({ open: false });

   175→    } catch (err) { toast.error((err as Error).message); }

   182→    if (!query.trim()) { setSearchResults([]); return; }

   184→      setSearchLoading(true);
   185→      try {
   186→        const res = await apiFetch(`/api/meals/search?q=${encodeURIComponent(query)}`);
   187→        setSearchResults((res.meals || []) as SearchMeal[]);
   188→      } catch { toast.error('Search failed'); }
   189→      finally { setSearchLoading(false); }

   191→    setSearchDebounce(timer);
   192→  };
   193→
   194→  const handleLogFromSearch = (meal: SearchMeal) => {
   195→    const slot = slotSearchDialog.open ? slotSearchDialog.slot : 'lunch';
   196→    setServingGms(meal.baseServingGms || 100);
   197→    setLogDialog({
   198→      open: true,
   199→      meal: {
   200→        meal: { id: meal.id, name: meal.name, mealType: meal.mealType, cuisine: meal.cuisine, imageUrl: null, prepTimeMin: meal.prepTimeMin, isVeg: meal.isVeg, isVegan: meal.isVegan },
   201→        score: 0, recommendedServingGms: meal.baseServingGms || 100, estimatedNutrition: meal.nutrition ? { calories: meal.nutrition.calories, proteinG: meal.nutrition.proteinG, carbsG: meal.nutrition.carbsG, fatG: meal.nutrition.fatG } : null,
   202→        baseNutritionPer100g: meal.nutrition ? { calories: meal.nutrition.calories, proteinG: meal.nutrition.proteinG, carbsG: meal.nutrition.carbsG, fatG: meal.nutrition.fatG } : null,

   204→      slot,
   205→      baseNutritionPer100g: meal.nutrition || null,
   206→      mealName: meal.name,
   207→      cuisine: meal.cuisine,
   208→      mealType: meal.mealType,
   209→      isVeg: meal.isVeg,
   210→      isVegan: meal.isVegan,
   211→    });
   212→    setSlotSearchDialog({ open: false });
   213→    setSearchQuery('');
   214→    setSearchResults([]);
   215→  };
   216→

   218→    try {
   219→      const result = await apiFetch('/api/water-log', { method: 'POST', body: JSON.stringify({ glasses: 1 }) });
   220→      setWaterCount(result.glassesConsumed || waterCount + 1);
   221→      toast.success('\uD83D\uDCA7 +1 glass of water');
   222→    } catch { toast.error('Failed to log water'); }
   223→  };
   224→
   225→  // Generate meal plan (Feature 2)
   226→  const handleGeneratePlan = async () => {
   227→    setPlanLoading(true);
   228→    try {
   229→      const result = await apiFetch('/api/meal-plan/generate');
   230→      toast.success(result.status === 'existing' ? 'Plan already exists for today!' : "Today's meal plan generated!");
   231→      // Fetch the plan
   232→      const plan = await apiFetch('/api/meal-plan');
   233→      setMealPlan({ exists: plan.exists, items: plan.items || [] });
   234→    } catch (err) { toast.error((err as Error).message); }
   235→    finally { setPlanLoading(false); }
   236→  };
   237→
   238→  const firstName = (user?.profile as Record<string, unknown> | null)?.firstName || (user?.name as string) || '';
   239→  const initial = firstName ? firstName.charAt(0).toUpperCase() : '';
   240→  const displayName = firstName || 'there';
   241→  const todayStr = format(new Date(), 'EEEE, MMMM d');
   242→
   243→  // Insights logic (Feature 3)
   244→  const targetCal = nutrition?.targets?.calories || 0;
   245→  const consumedCal = nutrition?.consumed?.calories || 0;
   246→
   247→  // Shimmer pulse when calorie value changes
   248→  useEffect(() => {

   256→  const pctConsumed = targetCal > 0 ? consumedCal / targetCal : 0;
   257→  const targetProtein = nutrition?.targets?.proteinG || 0;

   259→  const pctProtein = targetProtein > 0 ? consumedProtein / targetProtein : 0;
   260→  const hour = new Date().getHours();

   278→  return (
   279→    <motion.div {...fadeIn} className="p-4 max-w-lg mx-auto space-y-4">
   280→      {/* ═══ Hero Section with gradient bg & decorative blobs ═══ */}
   281→      <div className="relative overflow-hidden rounded-2xl bg-gradient-to-r from-emerald-500/5 to-teal-500/5 p-5 border border-emerald-100/50 dark:border-emerald-800/30 backdrop-blur-sm">
   282→        <div className="absolute -top-8 -right-8 w-24 h-24 bg-emerald-200/20 rounded-full blur-2xl" />
   283→        <div className="absolute -bottom-6 -left-6 w-20 h-20 bg-teal-200/20 rounded-full blur-2xl" />

   292→          </div>

   296→                <Zap className="h-3 w-3" />{streak} day{streak > 1 ? 's' : ''}
   297→              </Badge>
   298→            )}
   299→            {initial ? (
   300→              <div className="w-10 h-10 bg-gradient-to-br from-emerald-400 to-emerald-600 rounded-xl flex items-center justify-center shadow-md">
   301→                <span className="text-white font-bold text-sm">{initial}</span>
   302→              </div>

   304→              <div className="w-10 h-10 bg-emerald-100 dark:bg-emerald-900/40 rounded-xl flex items-center justify-center shadow-md">
   305→                <User className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   306→              </div>
   307→            )}
   308→          </div>
   309→        </div>
   310→      </div>

   316→          <p className="text-[11px] text-gray-600 dark:text-gray-400 font-semibold">Meals Logged</p>

   320→          <p className="text-[11px] text-gray-600 dark:text-gray-400 font-semibold">Water</p>
   321→        </div>
   322→        <div className="bg-white dark:bg-gray-900 rounded-2xl p-3 ring-1 ring-inset ring-gray-200/60 dark:ring-gray-700/40 shadow-sm text-center">
   323→          <p className="text-2xl font-extrabold tracking-tight text-emerald-600 dark:text-emerald-400 tabular-nums mb-1">{streak}</p>
   324→          <p className="text-[11px] text-gray-600 dark:text-gray-400 font-semibold">Day Streak</p>
   325→        </div>
   326→      </div>
   327→
   328→      {/* ═══ Calorie Ring Card ═══ */}
   329→      <Card className="p-6 rounded-2xl shadow-lg shadow-gray-200/50 dark:shadow-black/20 border border-gray-100/60 dark:border-gray-800/60">
   330→        <div className="flex justify-center">
   331→          <CalorieRing consumed={consumedCal} target={targetCal} pulse={calPulse} />
   332→        </div>
   333→        {pctConsumed >= 1 && (
   334→          <div className="text-center rounded-xl bg-gradient-to-r from-emerald-50 to-teal-50 dark:from-emerald-900/20 dark:to-teal-900/20 px-3 py-2">
   335→            <motion.p
   336→              initial={{ scale: 0.8, opacity: 0 }}
   337→              animate={{ scale: 1, opacity: 1 }}
   338→              transition={{ type: 'spring', stiffness: 200 }}
   339→              className="text-sm text-emerald-600 dark:text-emerald-400 font-bold"
   340→            >
   341→              You&apos;ve reached your calorie goal! Great job! 🎉
   342→            </motion.p>
   343→          </div>
   344→        )}
   345→      </Card>
   346→
   347→      {/* ═══ Macro Progress Bars ═══ */}
   348→      <Card className="p-4 rounded-2xl shadow-lg shadow-gray-200/50 dark:shadow-black/20 border border-gray-100/60 dark:border-gray-800/60 bg-white/70 dark:bg-gray-900/70 backdrop-blur-sm space-y-3 mb-5">
   349→        <h3 className="text-sm font-bold text-gray-800 dark:text-gray-200">Today&apos;s Macros</h3>
   350→        {[
   351→          { label: 'Protein', val: consumedProtein, target: targetProtein, from: '#3b82f6', to: '#60a5fa', unit: 'g' },
   352→          { label: 'Carbs', val: nutrition?.consumed?.carbsG || 0, target: nutrition?.targets?.carbsG || 0, from: '#f59e0b', to: '#fbbf24', unit: 'g' },

   355→          const pct = Math.min(Math.round((m.val / (m.target || 1)) * 100), 100);
   356→          return (
   357→            <div key={m.label}>
   358→              <div className="flex justify-between mb-1.5">
   359→                <span className="text-sm text-gray-600 dark:text-gray-400 font-medium">{m.label}</span>
   360→                <span className="text-xs text-gray-600 dark:text-gray-400 font-medium text-right tabular-nums">{m.val}{m.unit} / {m.target}{m.unit} &middot; {pct}%</span>
   361→              </div>
   362→              <div className="h-3 bg-gray-100 dark:bg-gray-800 rounded-full overflow-hidden">
   363→                <motion.div
   364→                  className="h-full rounded-full"
   365→                  style={{ background: `linear-gradient(90deg, ${m.from}, ${m.to})` }}
   366→                  initial={{ width: 0 }}
   367→                  animate={{ width: `${pct}%` }}
   368→                  transition={{ duration: 0.8 }}
   369→                />
   370→              </div>
   371→            </div>
   372→          );
   373→        })}
   374→      </Card>
   375→
   376→      {/* ═══ Hydration Widget ═══ */}
   377→      <Card className="p-4 rounded-2xl shadow-lg shadow-gray-200/50 dark:shadow-black/20 border border-gray-100/60 dark:border-gray-800/60">
   378→        <div className="flex items-center justify-between mb-3">
   379→          <div className="flex items-center gap-2">
   380→            <Droplets className="h-4 w-4 text-cyan-500" />
   381→            <h3 className="text-sm font-bold text-gray-800 dark:text-gray-200">Hydration</h3>
   382→          </div>
   383→          <span className="text-xs text-gray-500 dark:text-gray-400">Goal: 8 glasses</span>
   384→        </div>
   385→        <div className="flex items-center gap-3">
   386→          <div className="flex-1 flex gap-1.5">
   387→            {[1, 2, 3, 4, 5, 6, 7, 8].map((g) => (

   391→              />
   392→            ))}
   393→          </div>
   394→          <Button
   395→            size="icon"
   396→            className="w-9 h-9 rounded-xl bg-cyan-500 hover:bg-cyan-600 text-white shadow-sm"
   397→            onClick={handleWaterAdd}
   398→          >

   400→          </Button>

   402→        <p className="text-xs text-gray-500 dark:text-gray-400 mt-2">{waterCount} of 8 glasses today</p>

   404→
   405→      {/* ═══ Feature 3: Today's Insights Card ═══ */}
   406→      <motion.div
   407→        initial={{ opacity: 0, y: 8 }}
   408→        animate={{ opacity: 1, y: 0 }}
   409→        transition={{ delay: 0.3 }}
   410→      >
   411→        <Card className={`p-4 rounded-2xl shadow-lg shadow-gray-200/50 dark:shadow-black/20 border border-gray-100/60 dark:border-gray-800/60 backdrop-blur-sm ${pctConsumed > 1 ? 'bg-amber-50/50 dark:bg-amber-900/10' : pctConsumed >= 0.75 ? 'bg-emerald-50/50 dark:bg-emerald-900/10' : 'bg-blue-50/50 dark:bg-blue-900/10'}`}>

   414→              <Lightbulb className={`h-4.5 w-4.5 ${pctConsumed > 1 ? 'text-amber-500' : pctConsumed >= 0.75 ? 'text-emerald-500' : 'text-blue-500'}`} />
   415→            </div>
   416→            <div className="flex-1 min-w-0 space-y-2">
   417→              <h3 className="text-sm font-semibold text-gray-900 dark:text-gray-100">Today&apos;s Insights</h3>
   418→
   419→              {/* Calorie status message */}

   780→                <SelectContent>{SLOTS.map((s) => <SelectItem key={s} value={s}>{SLOT_LABELS[s]}</SelectItem>)}</SelectContent>
   781→              </Select>
   782→            </div>
   783→          </div>
   784→          <DialogFooter>
   785→            <Button variant="outline" onClick={() => setLogDialog({ open: false })} className="rounded-xl">Cancel</Button>
   786→            <Button className="bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl font-semibold" onClick={handleLogMeal}>Log It</Button>
   787→          </DialogFooter>
   788→        </DialogContent>
   789→      </Dialog>
   790→
   791→      {/* ═══ Per-Slot Meal Search Dialog ═══ */}
   792→      <Dialog open={slotSearchDialog.open} onOpenChange={(open) => { if (!open) { setSlotSearchDialog({ open: false }); setSearchQuery(''); setSearchResults([]); } }}>
   793→        <DialogContent className="max-w-md">
   794→          <DialogHeader>
   795→            <DialogTitle>Search Meals</DialogTitle>
   796→            <DialogDescription>Find and log a meal for {slotSearchDialog.open ? SLOT_LABELS[slotSearchDialog.slot] : ''}</DialogDescription>
   797→          </DialogHeader>
   798→          <div className="space-y-3">
   799→            <div className="relative">
   800→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400" />
   801→              <Input
   802→                placeholder="Search meals..."
   803→                value={searchQuery}
   804→                onChange={(e) => handleSlotSearch(e.target.value)}
   805→                className="h-11 rounded-xl pl-9"
   806→                autoFocus
   807→              />
   808→            </div>
   809→            <ScrollArea className="max-h-72">
   810→              <div className="space-y-2">
   811→                {searchLoading && <div className="flex justify-center py-6"><LoaderIcon className="h-5 w-5 animate-spin text-emerald-600" /></div>}
   812→                {!searchLoading && searchQuery && searchResults.length === 0 && (
   813→                  <p className="text-sm text-gray-400 dark:text-gray-500 text-center py-6">No results found</p>
   814→                )}
   815→                {!searchLoading && searchResults.map((meal) => (
   816→                  <div
   817→                    key={meal.id}
   818→                    className="flex items-center justify-between p-3 rounded-xl border border-gray-100 dark:border-gray-800 hover:bg-emerald-50/50 dark:hover:bg-emerald-900/10 hover:border-emerald-100 dark:hover:border-emerald-800 transition-colors cursor-pointer"
   819→                    onClick={() => handleLogFromSearch(meal)}
   820→                  >
   821→                    <div className="flex-1 min-w-0">
   822→                      <div className="flex items-center gap-2">
   823→                        <p className="text-sm font-medium text-gray-800 dark:text-gray-200 truncate">{meal.name}</p>
   824→                        <Badge variant="outline" className="text-xs px-1.5 py-0 h-4 font-medium border-gray-200 dark:border-gray-700 text-gray-500 dark:text-gray-400 shrink-0">{meal.cuisine}</Badge>
   825→                      </div>
   826→                      <div className="flex items-center gap-2 text-xs text-gray-500 dark:text-gray-400 mt-0.5">
   827→                        <span>{meal.nutrition?.calories || 0} kcal/100g</span>
   828→                        <span>&middot;</span>
   829→                        <span className="text-blue-600 font-medium">P: {meal.nutrition?.proteinG || 0}g</span>
   830→                      </div>
   831→                    </div>
   832→                    <ChevronRight className="h-4 w-4 text-gray-300 dark:text-gray-600 shrink-0" />
   833→                  </div>
   834→                ))}
   835→              </div>
   836→            </ScrollArea>
   837→          </div>
   838→        </DialogContent>
   839→      </Dialog>
   840→
   841→      {/* ═══ Meal Detail Sheet ═══ */}
   842→      <Sheet open={detailSheet.open} onOpenChange={(open) => { if (!open) setDetailSheet({ open: false }); }}>
   843→        <SheetContent side="bottom" className="max-w-lg mx-auto rounded-t-2xl max-h-[85vh]">
   844→          {detailSheet.open && (
   845→            <>
   846→              <SheetHeader className="pt-2">
   847→                <div className="flex items-center gap-2 pr-8">
   848→                  {detailSheet.rec.meal.isVeg && <span className="text-green-600 text-xs font-bold border border-green-300 rounded px-1.5 py-0.5">Veg</span>}
   849→                  {detailSheet.rec.meal.isVegan && <span className="text-green-700 text-xs font-bold border border-green-400 rounded px-1.5 py-0.5">Vegan</span>}
   850→                </div>
   851→                <SheetTitle className="text-lg">{detailSheet.rec.meal.name}</SheetTitle>
   852→                <SheetDescription className="flex items-center gap-2 flex-wrap">
   853→                  <Badge variant="outline" className="text-xs">{detailSheet.rec.meal.cuisine}</Badge>
   854→                  <Badge variant="outline" className={`text-xs ${SLOT_BADGE_COLORS[detailSheet.rec.meal.mealType] || ''}`}>{detailSheet.rec.meal.mealType}</Badge>
   855→                  {detailSheet.rec.meal.prepTimeMin && (
   856→                    <span className="text-xs text-gray-500 dark:text-gray-400 flex items-center gap-1"><Clock className="h-3 w-3" />{detailSheet.rec.meal.prepTimeMin} min</span>
   857→                  )}
   858→                </SheetDescription>
   859→              </SheetHeader>
   860→              <ScrollArea className="flex-1 px-4 pb-4">
   861→                <div className="space-y-4">
   862→                  {detailSheet.rec.meal.description && (
   863→                    <p className="text-sm text-gray-600 dark:text-gray-400 leading-relaxed">{detailSheet.rec.meal.description}</p>
   864→                  )}
   865→                  {detailSheet.rec.baseNutritionPer100g && (
   866→                    <div>
   867→                      <h4 className="text-xs font-semibold text-gray-500 dark:text-gray-400 uppercase tracking-wide mb-2">Nutrition per 100g</h4>
   868→                      <NutritionFactsLabel nutrition={detailSheet.rec.baseNutritionPer100g} servingGms={100} label="Per 100g" />
   869→                    </div>
   870→                  )}
   871→                  {detailSheet.rec.estimatedNutrition && (
   872→                    <div className="bg-emerald-50 dark:bg-emerald-900/20 rounded-xl p-3 border border-emerald-100 dark:border-emerald-800">
   873→                      <h4 className="text-xs font-semibold text-emerald-700 dark:text-emerald-400 uppercase tracking-wide mb-2">Recommended: {detailSheet.rec.recommendedServingGms}g serving</h4>
   874→                      <div className="grid grid-cols-2 gap-1.5 text-xs">
   875→                        <span className="text-gray-600 dark:text-gray-400">Calories:</span><span className="font-bold">{detailSheet.rec.estimatedNutrition.calories} kcal</span>
   876→                        <span className="text-gray-600 dark:text-gray-400">Protein:</span><span className="font-bold text-blue-600">{detailSheet.rec.estimatedNutrition.proteinG}g</span>
   877→                        <span className="text-gray-600 dark:text-gray-400">Carbs:</span><span className="font-bold text-amber-600">{detailSheet.rec.estimatedNutrition.carbsG}g</span>
   878→                        <span className="text-gray-600 dark:text-gray-400">Fat:</span><span className="font-bold text-rose-600">{detailSheet.rec.estimatedNutrition.fatG}g</span>
   879→                      </div>
   880→                    </div>
   881→                  )}
   882→                </div>
   883→              </ScrollArea>
   884→              <SheetFooter className="border-t pt-3">
   885→                <Button
   886→                  className="w-full bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl font-semibold min-h-[44px]"
   887→                  onClick={() => {
   888→                    const r = detailSheet.rec;
   889→                    setServingGms(r.recommendedServingGms);
   890→                    setLogDialog({
   891→                      open: true, meal: r, slot: r.meal.mealType,
   892→                      baseNutritionPer100g: r.baseNutritionPer100g,
   893→                      mealName: r.meal.name, cuisine: r.meal.cuisine,
   894→                      mealType: r.meal.mealType, description: r.meal.description,
   895→                      isVeg: r.meal.isVeg, isVegan: r.meal.isVegan,
   896→                    });
   897→                    setDetailSheet({ open: false });
   898→                  }}
   899→                >
   900→                  Log This Meal
   901→                </Button>
   902→              </SheetFooter>
   903→            </>
   904→          )}
   905→        </SheetContent>
   906→      </Sheet>
   907→    </motion.div>
   908→  );
   909→}
