   468→          ) : (
   469→            <>
   470→              {/* Column headers */}
   471→              <div className="grid grid-cols-12 gap-1 px-5 py-2.5 text-xs font-semibold text-gray-500 uppercase tracking-wider border-b border-gray-100">
   472→                <div className="col-span-3">Date</div>
   473→                <div className="col-span-3 text-right">Eaten</div>

   475→                <div className="col-span-3 text-right">Diff</div>

   477→
   478→              {/* Table rows */}
   479→              <div className="divide-y divide-gray-50">
   480→                {breakdownData.map((row, idx) => {
   481→                  const isOver = row.diff > 0;
   482→                  const isToday = row.date === format(new Date(), 'yyyy-MM-dd');

   485→                      key={row.date}
   486→                      className={`grid grid-cols-12 gap-1 py-3 px-4 items-center text-xs hover:bg-gray-50/50 transition-colors ${

   519→                            initial={{ width: 0 }}
   520→                            animate={{ width: `${Math.min(row.pct, 100)}%` }}

   530→              {/* Weekly total row */}
   531→              <div className="grid grid-cols-12 gap-1 px-5 py-3 bg-gray-50 border-t border-gray-200 text-xs font-bold">

   534→                <div className="col-span-3 text-right text-gray-500">{weeklyTotalTarget.toLocaleString()}</div>
   535→                <div className={`col-span-3 text-right ${weeklyTotalDiff > 0 ? 'text-red-500' : 'text-emerald-600'}`}>{
   536→                  weeklyTotalDiff > 0 ? '+' : ''}{weeklyTotalDiff}
   537→                </div>
   538→              </div>
   539→            </>
   540→          )}
   541→        </Card>
   542→      </motion.div>
   543→
   544→      {/* Water Tracking */}
   545→      <Card className="p-5 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   546→        <div className="flex items-center justify-between mb-4">
   547→          <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100 flex items-center gap-2">
   548→            <Droplets className="h-4 w-4 text-blue-500" /> Water Intake

   552→        <div className="flex items-center justify-center gap-4">
   553→          <Button
   554→            size="icon"
   555→            variant="outline"
   556→            className="h-10 w-10 rounded-xl"
   557→            onClick={() => handleAddWater(-1)}
   558→            disabled={waterGlasses <= 0}
   559→          >
   560→            <Minus className="h-4 w-4" />
   561→          </Button>
   562→          <div className="flex gap-2 flex-wrap justify-center">
   563→            {Array.from({ length: 8 }, (_, i) => {
   564→              const isFilled = i < waterGlasses;
   565→              const isJustFilled = justFilledIndex === i;
   566→              return (

   568→                  key={i}
   569→                  className="relative"
   570→                  animate={isJustFilled ? { scale: [1, 1.25, 1] } : {}}

   574→                    className={`w-7 h-9 rounded-md border-2 flex items-center justify-center transition-colors ${
   575→                      isFilled
   576→                        ? 'bg-blue-100 border-blue-300'
   577→                        : 'bg-gray-50 border-gray-200'
   578→                    }`}
   579→                    animate={{
   580→                      backgroundColor: isJustFilled ? ['#bfdbfe', '#dbeafe'] : undefined,
   581→                    }}
   582→                    transition={{ duration: 0.4 }}
   583→                  >
   584→                    <Droplets
   585→                      className={`h-4 w-4 transition-colors ${
   586→                        isFilled ? 'text-blue-500' : 'text-gray-200'

   594→                      animate={{ opacity: 0, scale: 1.4 }}
   595→                      transition={{ duration: 0.6 }}
   596→                    />
   597→                  )}
   598→                </motion.div>
   599→              );
   600→            })}
   601→          </div>
   602→          <Button
   603→            size="icon"
   604→            className="h-10 w-10 bg-blue-500 hover:bg-blue-600 rounded-xl"
   605→            onClick={() => handleAddWater(1)}
   606→            disabled={waterGlasses >= 16}
   607→          >
   608→            <Plus className="h-4 w-4" />
   609→          </Button>
   610→        </div>
   611→      </Card>
   612→
   613→      {/* Weight Logging */}
   614→      <Card className="p-5 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 bg-white dark:bg-gray-900">
   615→        <h3 className="text-base font-semibold text-gray-800 dark:text-gray-100 mb-3 flex items-center gap-2">
   616→          <Scale className="h-4 w-4 text-purple-500" /> Log Weight
   617→        </h3>
   618→        <div className="flex gap-2">
   619→          <Input
   620→            type="number"
   621→            placeholder="e.g., 74.5"
   622→            value={weightInput}
   623→            onChange={(e) => setWeightInput(e.target.value)}
   624→            className="flex-1 rounded-xl h-11"
   625→          />
   626→          <Input
   627→            placeholder="Optional notes..."
   628→            value={weightNotes}
   629→            onChange={(e) => setWeightNotes(e.target.value)}
   630→            className="flex-1 rounded-xl h-11"
   631→          />
   632→          <Button
   633→            className="bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl min-h-[44px] font-semibold px-5"
   634→            onClick={handleLogWeight}
   635→            disabled={!weightInput}
   636→          >
   637→            Log
   638→          </Button>
   639→        </div>
   640→        {weightLogs.length > 0 && (
   641→          <div className="mt-4 space-y-1">
   642→            <p className="text-xs text-gray-500 font-semibold uppercase tracking-wider mb-2">Recent Entries</p>
   643→            {weightLogs.slice(0, 5).map((w) => (
   644→              <div key={w.id as string} className="flex justify-between text-xs text-gray-600 py-2 px-3 border-b border-gray-50 last:border-0 hover:bg-gray-50/50 transition-colors rounded">
   645→                <span>{format(parseISO(w.logDate as string), 'MMM d, yyyy')}</span>
   646→                <span className="font-semibold">{w.weightKg as number} kg</span>
   647→              </div>
   648→            ))}
   649→          </div>
   650→        )}
   651→      </Card>
   652→    </motion.div>
   653→  );
