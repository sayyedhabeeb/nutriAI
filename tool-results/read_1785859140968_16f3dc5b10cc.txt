    88→        apiFetch('/api/auth/me'),

   149→        const ach = await apiFetch('/api/achievements');

   162→

   166→      await apiFetch('/api/food-logs', {
   167→        method: 'POST',
   168→        body: JSON.stringify({ mealId: logDialog.meal.meal.id, servingGms, mealSlot: logDialog.slot }),
   169→      });

   171→      setLogDialog({ open: false });

   180→    if (!query.trim()) { setSearchResults([]); return; }

   182→      setSearchLoading(true);
   183→      try {
   184→        const res = await apiFetch(`/api/meals/search?q=${encodeURIComponent(query)}`);
   185→        setSearchResults((res.meals || []) as SearchMeal[]);
   186→      } catch { toast.error('Search failed'); }
   187→      finally { setSearchLoading(false); }
   188→    }, 350);
   189→    setSearchDebounce(timer);
   190→  };
   191→
   192→  const handleLogFromSearch = (meal: SearchMeal) => {
   193→    const slot = slotSearchDialog.open ? slotSearchDialog.slot : 'lunch';
   194→    setServingGms(meal.baseServingGms || 100);
   195→    setLogDialog({
   196→      open: true,
   197→      meal: {
   198→        meal: { id: meal.id, name: meal.name, mealType: meal.mealType, cuisine: meal.cuisine, imageUrl: null, prepTimeMin: meal.prepTimeMin, isVeg: meal.isVeg, isVegan: meal.isVegan },
   199→        score: 0, recommendedServingGms: meal.baseServingGms || 100, estimatedNutrition: meal.nutrition ? { calories: meal.nutrition.calories, proteinG: meal.nutrition.proteinG, carbsG: meal.nutrition.carbsG, fatG: meal.nutrition.fatG } : null,
   200→        baseNutritionPer100g: meal.nutrition ? { calories: meal.nutrition.calories, proteinG: meal.nutrition.proteinG, carbsG: meal.nutrition.carbsG, fatG: meal.nutrition.fatG } : null,
   201→      },
   202→      slot,
   203→      baseNutritionPer100g: meal.nutrition || null,
   204→      mealName: meal.name,
   205→      cuisine: meal.cuisine,
   206→      mealType: meal.mealType,
   207→      isVeg: meal.isVeg,
   208→      isVegan: meal.isVegan,
   209→    });

   211→    setSearchQuery('');
   212→    setSearchResults([]);
   213→  };
   214→
   215→  const handleWaterAdd = async () => {
   216→    try {
   217→      const result = await apiFetch('/api/water-log', { method: 'POST', body: JSON.stringify({ glasses: 1 }) });
   218→      setWaterCount(result.glassesConsumed || waterCount + 1);
   219→      toast.success('\uD83D\uDCA7 +1 glass of water');
   220→    } catch { toast.error('Failed to log water'); }
   221→  };
   222→
   223→  // Generate meal plan (Feature 2)
   224→  const handleGeneratePlan = async () => {
   225→    setPlanLoading(true);
   226→    try {
   227→      const result = await apiFetch('/api/meal-plan/generate');
   228→      toast.success(result.status === 'existing' ? 'Plan already exists for today!' : "Today's meal plan generated!");
   229→      // Fetch the plan
   230→      const plan = await apiFetch('/api/meal-plan');
   231→      setMealPlan({ exists: plan.exists, items: plan.items || [] });
   232→    } catch (err) { toast.error((err as Error).message); }
   233→    finally { setPlanLoading(false); }
   234→  };
   235→
   236→  const firstName = (user?.profile as Record<string, unknown> | null)?.firstName || (user?.name as string) || '';
   237→  const initial = firstName ? firstName.charAt(0).toUpperCase() : '';
   238→  const displayName = firstName || 'there';
   239→  const todayStr = format(new Date(), 'EEEE, MMMM d');
   240→
   241→  // Insights logic (Feature 3)
   242→  const targetCal = nutrition?.targets?.calories || 0;
   243→  const consumedCal = nutrition?.consumed?.calories || 0;
   244→  const pctConsumed = targetCal > 0 ? consumedCal / targetCal : 0;
   245→  const targetProtein = nutrition?.targets?.proteinG || 0;
   246→  const consumedProtein = nutrition?.consumed?.proteinG || 0;
   247→  const pctProtein = targetProtein > 0 ? consumedProtein / targetProtein : 0;
   248→  const hour = new Date().getHours();
   249→  const timeOfDayTip = hour < 12
   250→    ? 'Start your day with a protein-rich breakfast!'
   251→    : hour < 17
   252→      ? 'Keep up the good work! Stay consistent with your meals.'
   253→      : 'Plan a balanced dinner to finish strong today.';
   254→
   255→  if (loading) {
   256→    return (
   257→      <div className="p-4 max-w-lg mx-auto space-y-4">
   258→        <Skeleton className="h-8 w-56 rounded-xl" />
   259→        <div className="flex justify-center"><Skeleton className="h-[180px] w-[180px] rounded-full" /></div>
   260→        <div className="space-y-3"><Skeleton className="h-10 w-full rounded-xl" /><Skeleton className="h-10 w-full rounded-xl" /><Skeleton className="h-10 w-full rounded-xl" /></div>

   270→        <div className="absolute -top-8 -right-8 w-24 h-24 bg-emerald-200/20 rounded-full blur-2xl" />
   271→        <div className="absolute -bottom-6 -left-6 w-20 h-20 bg-teal-200/20 rounded-full blur-2xl" />
   272→        <div className="absolute top-1/2 right-1/4 w-12 h-12 bg-green-200/10 rounded-full blur-xl" />

   275→            <h1 className="text-2xl font-bold text-gray-900 dark:text-gray-100">Hi, {displayName}! {'\uD83D\uDC4B'}</h1>

   280→          </div>

   289→                <span className="text-white font-bold text-sm">{initial}</span>
   290→              </div>
   291→            ) : (
   292→              <div className="w-10 h-10 bg-emerald-100 dark:bg-emerald-900/40 rounded-xl flex items-center justify-center shadow-md">
   293→                <User className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   294→              </div>
   295→            )}
   296→          </div>
   297→        </div>
   298→      </div>
   299→

   304→          <p className="text-[10px] text-gray-500 dark:text-gray-400 font-medium">Meals Logged</p>

   313→        </div>
   314→      </div>
   315→
   316→      {/* ═══ Calorie Ring Card ═══ */}

   318→        <div className="flex justify-center">
   319→          <CalorieRing consumed={consumedCal} target={targetCal} />
   320→        </div>
   321→        {pctConsumed >= 1 && (
   322→          <p className="text-center text-sm text-rose-600 font-medium mt-2">
   323→            You&apos;ve reached your calorie goal! Great job! {'\uD83C\uDF89'}
   324→          </p>
   325→        )}
   326→      </Card>
   327→
   328→      {/* ═══ Macro Progress Bars ═══ */}
   329→      <Card className="p-4 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 space-y-3">
   330→        <h3 className="text-sm font-bold text-gray-800 dark:text-gray-200">Today&apos;s Macros</h3>
   331→        {[
   332→          { label: 'Protein', val: consumedProtein, target: targetProtein, from: '#3b82f6', to: '#60a5fa', unit: 'g' },
   333→          { label: 'Carbs', val: nutrition?.consumed?.carbsG || 0, target: nutrition?.targets?.carbsG || 0, from: '#f59e0b', to: '#fbbf24', unit: 'g' },
   334→          { label: 'Fat', val: nutrition?.consumed?.fatG || 0, target: nutrition?.targets?.fatG || 0, from: '#f43f5e', to: '#fb7185', unit: 'g' },
   335→        ].map((m) => {
   336→          const pct = Math.min(Math.round((m.val / (m.target || 1)) * 100), 100);
   337→          return (
   338→            <div key={m.label}>
   339→              <div className="flex justify-between mb-1.5">
   340→                <span className="text-sm text-gray-600 dark:text-gray-400 font-medium">{m.label}</span>
   341→                <span className="text-xs text-gray-600 dark:text-gray-400 font-medium text-right tabular-nums">{m.val}{m.unit} / {m.target}{m.unit} &middot; {pct}%</span>
   342→              </div>
   343→              <div className="h-3 bg-gray-100 dark:bg-gray-800 rounded-full overflow-hidden">
   344→                <motion.div
   345→                  className="h-full rounded-full"
   346→                  style={{ background: `linear-gradient(90deg, ${m.from}, ${m.to})` }}
   347→                  initial={{ width: 0 }}
   348→                  animate={{ width: `${pct}%` }}
   349→                  transition={{ duration: 0.8 }}
   350→                />
   351→              </div>
   352→            </div>
   353→          );
   354→        })}
   355→      </Card>
   356→
   357→      {/* ═══ Hydration Widget ═══ */}
   358→      <Card className="p-4 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60">
   359→        <div className="flex items-center justify-between mb-3">
   360→          <div className="flex items-center gap-2">
   361→            <Droplets className="h-4 w-4 text-cyan-500" />
   362→            <h3 className="text-sm font-bold text-gray-800 dark:text-gray-200">Hydration</h3>
   363→          </div>
   364→          <span className="text-xs text-gray-500 dark:text-gray-400">Goal: 8 glasses</span>
   365→        </div>
   366→        <div className="flex items-center gap-3">
   367→          <div className="flex-1 flex gap-1.5">
   368→            {[1, 2, 3, 4, 5, 6, 7, 8].map((g) => (
   369→              <div
   370→                key={g}
   371→                className={`h-6 flex-1 rounded-md transition-colors ${g <= waterCount ? 'bg-cyan-400 dark:bg-cyan-500' : 'bg-gray-100 dark:bg-gray-800'}`}
   372→              />
   373→            ))}
   374→          </div>
   375→          <Button
   376→            size="icon"
   377→            className="w-9 h-9 rounded-xl bg-cyan-500 hover:bg-cyan-600 text-white shadow-sm"
   378→            onClick={handleWaterAdd}
   379→          >
   380→            <Plus className="h-4 w-4" />
   381→          </Button>
   382→        </div>
   383→        <p className="text-xs text-gray-500 dark:text-gray-400 mt-2">{waterCount} of 8 glasses today</p>
   384→      </Card>
   385→
   386→      {/* ═══ Feature 3: Today's Insights Card ═══ */}

   389→        animate={{ opacity: 1, y: 0 }}
   390→        transition={{ delay: 0.3 }}
   391→      >
   392→        <Card className={`p-4 rounded-2xl shadow-md border border-gray-100/60 dark:border-gray-800/60 ${pctConsumed > 1 ? 'bg-amber-50/50 dark:bg-amber-900/10' : pctConsumed >= 0.75 ? 'bg-emerald-50/50 dark:bg-emerald-900/10' : 'bg-blue-50/50 dark:bg-blue-900/10'}`}>
   393→          <div className="flex items-start gap-3">

   395→              <Lightbulb className={`h-4.5 w-4.5 ${pctConsumed > 1 ? 'text-amber-500' : pctConsumed >= 0.75 ? 'text-emerald-500' : 'text-blue-500'}`} />
   396→            </div>
   397→            <div className="flex-1 min-w-0 space-y-2">
   398→              <h3 className="text-sm font-semibold text-gray-900 dark:text-gray-100">Today&apos;s Insights</h3>
   399→
   400→              {/* Calorie status message */}
   401→              {pctConsumed > 1 && (
   402→                <p className="text-xs text-amber-700 dark:text-amber-400 leading-relaxed">

   413→                  You still have <span className="font-bold">{Math.round(targetCal - consumedCal)} kcal</span> remaining. Time to fuel up!

   416→              {consumedCal === 0 && (
   417→                <p className="text-xs text-gray-500 dark:text-gray-400 leading-relaxed">
   418→                  No meals logged yet today. Start with a balanced breakfast!
   419→                </p>

   421→
   422→              {/* Protein status */}
   423→              {consumedProtein > 0 && pctProtein >= 0.8 && (
   424→                <p className="text-xs text-emerald-600 dark:text-emerald-400 font-medium">
   425→                  Protein goal almost reached! {'\uD83C\uDF89'} ({Math.round(consumedProtein)}/{Math.round(targetProtein)}g)

   777→            <DialogDescription>Find and log a meal for {slotSearchDialog.open ? SLOT_LABELS[slotSearchDialog.slot] : ''}</DialogDescription>
   778→          </DialogHeader>
   779→          <div className="space-y-3">
   780→            <div className="relative">
   781→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400" />
   782→              <Input
   783→                placeholder="Search meals..."
   784→                value={searchQuery}
   785→                onChange={(e) => handleSlotSearch(e.target.value)}
   786→                className="h-11 rounded-xl pl-9"
   787→                autoFocus
   788→              />
   789→            </div>
   790→            <ScrollArea className="max-h-72">
   791→              <div className="space-y-2">
   792→                {searchLoading && <div className="flex justify-center py-6"><LoaderIcon className="h-5 w-5 animate-spin text-emerald-600" /></div>}
   793→                {!searchLoading && searchQuery && searchResults.length === 0 && (
   794→                  <p className="text-sm text-gray-400 dark:text-gray-500 text-center py-6">No results found</p>
   795→                )}
   796→                {!searchLoading && searchResults.map((meal) => (
   797→                  <div
   798→                    key={meal.id}
   799→                    className="flex items-center justify-between p-3 rounded-xl border border-gray-100 dark:border-gray-800 hover:bg-emerald-50/50 dark:hover:bg-emerald-900/10 hover:border-emerald-100 dark:hover:border-emerald-800 transition-colors cursor-pointer"
   800→                    onClick={() => handleLogFromSearch(meal)}
   801→                  >
   802→                    <div className="flex-1 min-w-0">
   803→                      <div className="flex items-center gap-2">
   804→                        <p className="text-sm font-medium text-gray-800 dark:text-gray-200 truncate">{meal.name}</p>
   805→                        <Badge variant="outline" className="text-xs px-1.5 py-0 h-4 font-medium border-gray-200 dark:border-gray-700 text-gray-500 dark:text-gray-400 shrink-0">{meal.cuisine}</Badge>
   806→                      </div>
   807→                      <div className="flex items-center gap-2 text-xs text-gray-500 dark:text-gray-400 mt-0.5">
   808→                        <span>{meal.nutrition?.calories || 0} kcal/100g</span>
   809→                        <span>&middot;</span>
   810→                        <span className="text-blue-600 font-medium">P: {meal.nutrition?.proteinG || 0}g</span>
   811→                      </div>
   812→                    </div>
   813→                    <ChevronRight className="h-4 w-4 text-gray-300 dark:text-gray-600 shrink-0" />
   814→                  </div>
   815→                ))}
   816→              </div>
   817→            </ScrollArea>
   818→          </div>
   819→        </DialogContent>
   820→      </Dialog>
   821→
   822→      {/* ═══ Meal Detail Sheet ═══ */}
   823→      <Sheet open={detailSheet.open} onOpenChange={(open) => { if (!open) setDetailSheet({ open: false }); }}>
   824→        <SheetContent side="bottom" className="max-w-lg mx-auto rounded-t-2xl max-h-[85vh]">
   825→          {detailSheet.open && (
   826→            <>
   827→              <SheetHeader className="pt-2">
   828→                <div className="flex items-center gap-2 pr-8">
   829→                  {detailSheet.rec.meal.isVeg && <span className="text-green-600 text-xs font-bold border border-green-300 rounded px-1.5 py-0.5">Veg</span>}
   830→                  {detailSheet.rec.meal.isVegan && <span className="text-green-700 text-xs font-bold border border-green-400 rounded px-1.5 py-0.5">Vegan</span>}
   831→                </div>
   832→                <SheetTitle className="text-lg">{detailSheet.rec.meal.name}</SheetTitle>
   833→                <SheetDescription className="flex items-center gap-2 flex-wrap">
   834→                  <Badge variant="outline" className="text-xs">{detailSheet.rec.meal.cuisine}</Badge>
   835→                  <Badge variant="outline" className={`text-xs ${SLOT_BADGE_COLORS[detailSheet.rec.meal.mealType] || ''}`}>{detailSheet.rec.meal.mealType}</Badge>
   836→                  {detailSheet.rec.meal.prepTimeMin && (
   837→                    <span className="text-xs text-gray-500 dark:text-gray-400 flex items-center gap-1"><Clock className="h-3 w-3" />{detailSheet.rec.meal.prepTimeMin} min</span>
   838→                  )}
   839→                </SheetDescription>
   840→              </SheetHeader>
   841→              <ScrollArea className="flex-1 px-4 pb-4">
   842→                <div className="space-y-4">
   843→                  {detailSheet.rec.meal.description && (
   844→                    <p className="text-sm text-gray-600 dark:text-gray-400 leading-relaxed">{detailSheet.rec.meal.description}</p>
   845→                  )}
   846→                  {detailSheet.rec.baseNutritionPer100g && (
   847→                    <div>
   848→                      <h4 className="text-xs font-semibold text-gray-500 dark:text-gray-400 uppercase tracking-wide mb-2">Nutrition per 100g</h4>
   849→                      <NutritionFactsLabel nutrition={detailSheet.rec.baseNutritionPer100g} servingGms={100} label="Per 100g" />
   850→                    </div>
   851→                  )}
   852→                  {detailSheet.rec.estimatedNutrition && (
   853→                    <div className="bg-emerald-50 dark:bg-emerald-900/20 rounded-xl p-3 border border-emerald-100 dark:border-emerald-800">
   854→                      <h4 className="text-xs font-semibold text-emerald-700 dark:text-emerald-400 uppercase tracking-wide mb-2">Recommended: {detailSheet.rec.recommendedServingGms}g serving</h4>
   855→                      <div className="grid grid-cols-2 gap-1.5 text-xs">
   856→                        <span className="text-gray-600 dark:text-gray-400">Calories:</span><span className="font-bold">{detailSheet.rec.estimatedNutrition.calories} kcal</span>
   857→                        <span className="text-gray-600 dark:text-gray-400">Protein:</span><span className="font-bold text-blue-600">{detailSheet.rec.estimatedNutrition.proteinG}g</span>
   858→                        <span className="text-gray-600 dark:text-gray-400">Carbs:</span><span className="font-bold text-amber-600">{detailSheet.rec.estimatedNutrition.carbsG}g</span>
   859→                        <span className="text-gray-600 dark:text-gray-400">Fat:</span><span className="font-bold text-rose-600">{detailSheet.rec.estimatedNutrition.fatG}g</span>
   860→                      </div>
   861→                    </div>
   862→                  )}
   863→                </div>
   864→              </ScrollArea>
   865→              <SheetFooter className="border-t pt-3">
   866→                <Button
   867→                  className="w-full bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl font-semibold min-h-[44px]"
   868→                  onClick={() => {
   869→                    const r = detailSheet.rec;
   870→                    setServingGms(r.recommendedServingGms);
   871→                    setLogDialog({
   872→                      open: true, meal: r, slot: r.meal.mealType,
   873→                      baseNutritionPer100g: r.baseNutritionPer100g,
   874→                      mealName: r.meal.name, cuisine: r.meal.cuisine,
   875→                      mealType: r.meal.mealType, description: r.meal.description,
   876→                      isVeg: r.meal.isVeg, isVegan: r.meal.isVegan,
   877→                    });
   878→                    setDetailSheet({ open: false });
   879→                  }}
   880→                >
   881→                  Log This Meal
   882→                </Button>
   883→              </SheetFooter>
   884→            </>
   885→          )}
   886→        </SheetContent>
   887→      </Sheet>
   888→    </motion.div>
   889→  );
   890→}
