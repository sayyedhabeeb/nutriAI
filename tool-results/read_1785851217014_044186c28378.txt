   101→   101→// ═══ API Helper ═══
   102→   102→function getToken(): string | null {
   103→   103→  if (typeof window === 'undefined') return null;
   104→   104→  return localStorage.getItem('nutriai_session');
   105→   105→}
   106→   106→function setToken(token: string) {
   107→   107→  localStorage.setItem('nutriai_session', token);
   108→   108→}
   109→   109→function clearToken() {
   110→   110→  localStorage.removeItem('nutriai_session');
   111→   111→}
   112→   112→async function apiFetch(url: string, options: RequestInit = {}) {
   113→   113→  const token = getToken();
   114→   114→  const headers: Record<string, string> = {
   115→   115→    ...(options.headers as Record<string, string> || {}),
   116→   116→  };
   117→   117→  if (token) headers['Authorization'] = `Bearer ${token}`;
   118→   118→  if (!(options.body instanceof FormData)) headers['Content-Type'] = 'application/json';
   119→   119→  const res = await fetch(url, { ...options, headers });
   120→   120→  const json = await res.json();
   121→   121→  if (!json.success) throw new Error(json.error || 'Request failed');
   122→   122→  return json.data;
   123→   123→}

   318→   318→function AuthView({ onAuth }: { onAuth: (v: ViewType) => void }) {
   319→   319→  const [isLogin, setIsLogin] = useState(true);
   320→   320→  const [email, setEmail] = useState('');
   321→   321→  const [password, setPassword] = useState('');
   322→   322→  const [name, setName] = useState('');
   323→   323→  const [loading, setLoading] = useState(false);
   324→   324→
   325→   325→  const handleSubmit = async (e: React.FormEvent) => {
   326→   326→    e.preventDefault();
   327→   327→    setLoading(true);
   328→   328→    try {
   329→   329→      const url = isLogin ? '/api/auth/login' : '/api/auth/register';
   330→   330→      const body: Record<string, string> = { email, password };
   331→   331→      if (!isLogin && name) body.name = name;
   332→   332→      const res = await fetch(url, {
   333→   333→        method: 'POST',
   334→   334→        headers: { 'Content-Type': 'application/json' },
   335→   335→        body: JSON.stringify(body),
   336→   336→      });
   337→   337→      const json = await res.json();
   338→   338→      if (!json.success) {
   339→   339→        toast.error(json.error || 'Authentication failed');
   340→   340→        return;
   341→   341→      }
   342→   342→      setToken(json.data.token);
   343→   343→      toast.success(isLogin ? 'Welcome back!' : 'Account created!');
   344→   344→      const me = await apiFetch('/api/auth/me');
   345→   345→      if (me.profile && me.goal && me.preference) {
   346→   346→        onAuth('dashboard');
   347→   347→      } else {
   348→   348→        onAuth('onboarding');
   349→   349→      }
   350→   350→    } catch {
   351→   351→      toast.error('Something went wrong. Please try again.');
   352→   352→    } finally {
   353→   353→      setLoading(false);
   354→   354→    }
   355→   355→  };
   356→   356→
   357→   357→  return (
   358→   358→    <motion.div {...fadeIn} className="min-h-screen flex items-center justify-center p-4 bg-gradient-to-br from-emerald-50 via-green-50/30 to-white">

   361→   361→          <div className="mx-auto w-16 h-16 bg-white/20 backdrop-blur-sm rounded-2xl flex items-center justify-center mb-3">
   362→   362→            <Leaf className="h-8 w-8 text-white" />
   363→   363→          </div>
   364→   364→          <h2 className="text-2xl font-bold text-white">NutriAI</h2>
   365→   365→          <p className="text-emerald-100 text-sm mt-1">AI-Powered Nutrition Tracker</p>
   366→   366→        </div>
   367→   367→        <CardContent className="pt-6 px-6 pb-6">
   368→   368→          <form onSubmit={handleSubmit} className="space-y-4">
   369→   369→            {!isLogin && (
   370→   370→              <div className="space-y-2">
   371→   371→                <Label htmlFor="name" className="text-sm font-medium">Full Name</Label>
   372→   372→                <Input id="name" placeholder="John Doe" value={name} onChange={(e) => setName(e.target.value)} className="h-11 rounded-xl" />
   373→   373→              </div>
   374→   374→            )}
   375→   375→            <div className="space-y-2">
   376→   376→              <Label htmlFor="email" className="text-sm font-medium">Email</Label>
   377→   377→              <Input id="email" type="email" placeholder="you@example.com" value={email} onChange={(e) => setEmail(e.target.value)} required className="h-11 rounded-xl" />

   382→   382→            </div>
   383→   383→            <Button type="submit" className="w-full bg-emerald-600 hover:bg-emerald-700 text-white min-h-[44px] rounded-xl font-semibold" disabled={loading}>

   389→   389→            {isLogin ? "Don't have an account?" : 'Already have an account?'}{' '}

   572→   572→// CALORIE RING SVG (with gradient stroke)
   573→   573→// ═══════════════════════════════════════════════════════════
   574→   574→function CalorieRing({ consumed, target }: { consumed: number; target: number }) {
   575→   575→  const pct = target > 0 ? Math.min(consumed / target, 1) : 0;
   576→   576→  const radius = 70;
   577→   577→  const circumference = 2 * Math.PI * radius;
   578→   578→  const strokeDashoffset = circumference * (1 - pct);
   579→   579→  const remaining = Math.max(0, target - consumed);
   580→   580→  const gradId = 'calRingGrad';
   581→   581→
   582→   582→  return (
   583→   583→    <div className="relative inline-flex items-center justify-center">
   584→   584→      <svg width="180" height="180" viewBox="0 0 180 180" className="-rotate-90">
   585→   585→        <defs>
   586→   586→          <linearGradient id={gradId} x1="0%" y1="0%" x2="100%" y2="0%">
   587→   587→            <stop offset="0%" stopColor="#10b981" />
   588→   588→            <stop offset="100%" stopColor="#34d399" />
   589→   589→          </linearGradient>
   590→   590→        </defs>
   591→   591→        <circle cx="90" cy="90" r={radius} fill="none" stroke="#e5e7eb" strokeWidth="12" />
   592→   592→        <motion.circle
   593→   593→          cx="90" cy="90" r={radius} fill="none"
   594→   594→          stroke={pct > 1 ? '#f43f5e' : `url(#${gradId})`}
   595→   595→          strokeWidth="12" strokeLinecap="round"
   596→   596→          strokeDasharray={circumference}
   597→   597→          initial={{ strokeDashoffset: circumference }}
   598→   598→          animate={{ strokeDashoffset }}
   599→   599→          transition={{ duration: 1, ease: 'easeOut' }}
   600→   600→        />
   601→   601→      </svg>
   602→   602→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   603→   603→        <Flame className="h-5 w-5 text-orange-500 mb-1" />
   604→   604→        <span className="text-2xl font-bold text-gray-900">{consumed}</span>
   605→   605→        <span className="text-xs text-gray-500">of {target} kcal</span>
   606→   606→        <span className="text-xs text-emerald-600 font-semibold mt-1">{remaining} left</span>
   607→   607→      </div>
   608→   608→    </div>
   609→   609→  );
   610→   610→}
